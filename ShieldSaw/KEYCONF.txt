// addslotdefault appends: Fist and Chainsaw stay in slot 1 for every other mod
addslotdefault 1 SSQ_ShieldSaw

// the desktop route, and a backup for a headset: draw / put back / recall, and throw along the aim
alias ssq_draw "netevent ssq-draw"
alias ssq_throw "netevent ssq-throw"
alias +ssq_lock "netevent ssq-lock 1"
alias -ssq_lock "netevent ssq-lock 0"
addkeysection "Ultra Badass ShieldSaw" ssq_keys
addmenukey "$SSQ_KEY_DRAW" ssq_draw
addmenukey "$SSQ_KEY_THROW" ssq_throw
addmenukey "$SSQ_KEY_LOCK" +ssq_lock
