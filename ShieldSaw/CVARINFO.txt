// RS_ShieldSawQ -- cvars. SERVER cvars decide damage and where things go, so every machine agrees.
// USER cvars are userinfo: they travel with the player, so per-player choices the sim reads
// (handedness, placement) are the same on every machine too.

// ---- the game rules (server) ------------------------------------------------
server bool  ssq_start          = true;   // every player spawns with it
server float ssq_cut_damage     = 1.0;    // thrown-disc damage scale
server float ssq_throw_speed    = 1.6;    // launch speed scale: a 6 m/s swing gives ~8 units/tic (204 u/s / 35 x 30/35 x 1.6), so ssq_speed_floor sets it; a rocket is 20
server float ssq_throw_blend    = 0.7;    // 1 = throw along the arm's own direction, 0 = along where the hand points
// ---- THE THREE TRIGGER SPEEDS, RE-PICKED FOR AN ARM ----------------------------------------
// These were chosen while the measurement still carried the player's own travel, so they had to
// be large to stop walking setting them off -- and now that only the arm counts, the same number
// asks for a shove nobody makes. Re-picked in real units at 34 map units to the metre, 35 tics a
// second, and sanity-checked against what an arm actually does:
//
//   a deliberate plate shove      1.2 - 1.5 m/s   41 - 51 u/s
//   a parry flick into a shot     0.7 - 0.9 m/s   24 - 31 u/s
//   a real throwing motion        2 m/s and up    68 u/s and up
//
// ssq_throw_min goes UP, not down: 20 was only ever met by the player's own travel, and on the
// arm alone a casual wave clears it. A throw should be a throw.
// Everything here is still a feel number -- the sliders are on the menu and these are a start.
server float ssq_throw_min      = 60.0;   // units/sec of ARM speed below which a release stows instead of throwing
server float ssq_throw_forward  = 0.35;   // how closely the throw must go where you look (cosine; 0 = any direction)
server float ssq_speed_floor    = 12.0;   // units/tic: the slowest a throw can leave at
server float ssq_mass_kg        = 5.0;    // the disc keeps arm/(arm+this) of the hand speed
server int   ssq_locks          = 5;      // targets one throw visits
server float ssq_lock_cone      = 25.0;   // half-angle from the hand's forward at release
server int   ssq_lock_button    = 2;      // what throws when let go with a flick: 0 off alt only, 1 either off button, 2 off trigger only
server float ssq_lock_mark_scale = 1.0;   // size of the ring over a locked target
server float ssq_lock_range     = 1200.0;
server float ssq_lock_keep      = 60.0;   // a lock is KEPT out to here, far wider than the cone it was acquired in
server int   ssq_lock_grace     = 105;     // tics a lock may stay out of that cone or out of sight before it drops
server int   ssq_paint_every    = 3;      // tics between targets going on: they paint one at a time as you sweep
server int   ssq_keep_marks     = 175;     // tics your marks survive a release that was not a throw
server float ssq_return_speed   = 22.0;   // units/tic on the way home
server float ssq_cruise_gain    = 0.05;   // how fast a flying disc winds up while it has a target, per tic
server float ssq_cruise_max     = 34.0;   // units/tic it will wind up to
server int   ssq_home_patience  = 105;    // tics of returning before it ignores geometry and flies straight home
server float ssq_catch_radius   = 24.0;   // hand within this of the returning disc = caught
server bool  ssq_catch          = true;   // back in hand on return
server bool  ssq_perfect        = true;   // meeting a shot with a moving plate: faster return, always staggers, rings bright
server float ssq_perfect_speed  = 25.0;   // units/sec the plate must be moving INTO the attack: a parry flick, arm only
server int   ssq_crumb_every    = 4;      // tics between breadcrumbs on the way out

// what a thrown disc does, by the victim's authored health (SpawnHealth), never by class name
server bool  ssq_tiers          = true;
server bool  ssq_bd22           = false;  // Brutal Doom v22 monster compatibility: hit tiers for its tougher monsters (slice <= 250, grind <= 1200)
server int   ssq_tier_fodder    = 70;     // at or under: bisected
server int   ssq_tier_mid       = 600;    // at or under: the disc embeds and grinds
server int   ssq_embed_tics     = 21;
server int   ssq_embed_dps      = 3;      // damage PER TIC while embedded, dealt as 3x this every third tic (not per second)
server float ssq_glance_scale   = 0.4;

// the rev (off-hand trigger held)
server int   ssq_rev_tics       = 21;     // tics from still to full speed
server int   ssq_grind_damage   = 6;      // per hit while revved and something is in the blades
server int   ssq_grind_every    = 4;      // tics between grind hits ON THE SAME BODY (per victim, not a global gate)
server float ssq_rev_throw_bonus = 0.6;   // a full-rev throw adds this fraction of damage and speed

// the block
server bool  ssq_guard          = true;
server bool  ssq_block_double   = false;  // Double Block Radius: the blocking plate is twice as wide (the shield you see is unchanged); with Gun Bonsai's VR patch loaded the plate grows by its Broadening upgrade instead
server int   ssq_store_to       = 0;      // On Manual Store: 0 the off hand's fist, 1 the weapon it held before the shield
server bool  ssq_guard_stowed   = false;  // off: a shield you cannot see does not block. Both this and ssq_stow_world default off, together
server int   ssq_yield_offhand  = 0;      // 0 pin the hands (v0.6 behaviour); 1 let switchhand take the off hand and stow the shield
server int   ssq_deflect        = 1;      // blocked projectiles: 0 cancelled, 1 return to sender
server int   ssq_deflect_aim    = 3;      // a turned shot goes: 0 back at the shooter, 1 where you LOOK, 2 mirrored off the face,
                                          // 3 off the face AND the way the plate was moving (a bash throws it along the bash)
server float ssq_deflect_swing  = 3.0;    // aim 3: how much the plate's own swing steers a turned shot
server float ssq_deflect_assist = 14.0;   // degrees of help putting a looked-at deflection onto an enemy; 0 = none
server int   ssq_sweep          = 1;      // turn shots that cross the plate BEFORE they reach you, not only ones that would hit
server float ssq_sweep_lead     = 1.0;    // tics of a shot's own path tested ahead against the plate (1..6). 1 = only what
                                          // reaches the plate this tic (owner 2026-10-31: "plate size"); was 3, which
                                          // turned shots two tics early, including ones that would have missed
server float ssq_sweep_pad      = 0.0;    // cm added to the plate's rim for the sweep and the bullet stop. 0 = the plate
                                          // itself (owner 2026-10-31); was 18, a guard 47 cm across the rim
server float ssq_sweep_boost    = 1.0;    // speed a turned shot leaves at, as a multiple of its own
server int   ssq_hitscan_guard  = 1;      // stop bullets at the muzzle when their line crosses the plate
server int   ssq_hitscan_return = 1;      // a stopped bullet goes back out, aimed by ssq_deflect_aim
server float ssq_hitscan_damage = 1.0;    // the returned shot's damage, as a multiple of the one you ate
server int   ssq_melee          = 2;      // blocked melee: 0 nothing, 1 stun, 2 stun and shoved back
server float ssq_stagger_push   = 12.0;

// the bash: shove the shield into something
server bool  ssq_bash           = true;
// 45 u/s is about 1.3 m/s of arm along the face for a snap. A steady LEAN gets there too: the
// path sum decays at 0.75 a tic, so a sustained push counts about 2.4x itself at ssq_bash_sustain
// 0.6 -- roughly 19 u/s held, which is a deliberate press and not a drift.
server float ssq_bash_speed     = 45.0;   // units/sec of arm along the face
server float ssq_bash_damage    = 1.0;
server float ssq_bash_reach     = 1.2;    // the bash opens the plate and its depth by this much, for the shove only
server bool  ssq_bash_los       = true;   // the bash and the grind cannot reach through a wall or a door
server float ssq_bash_sustain   = 0.6;    // how much of a sustained push counts as a bash (0 = fast shoves only)
server float ssq_bash_knock     = 1.35;   // how far a bashed body goes
server float ssq_bash_lift      = 2.5;    // the small lift under the shove, so they leave the floor
// WALL SLAM (owner's pick 12, 2026-10-31; Quest teardown surveys/01_MELEE.md 4.7): a body the bash throws into a
// wall hits it -- extra damage by how hard it struck, a forced pain and a short freeze (stunned), a small rebound
server bool  ssq_slam           = true;
server float ssq_slam_damage    = 1.0;    // scale on the slam's damage (2 x impact speed in units/tic, 5..40)
server float ssq_slam_min       = 3.0;    // units/tic INTO the wall below which a stop is a bump, not a slam (01_MELEE 4.7: 3.0)
server int   ssq_slam_stun      = 8;      // tics the slammed body is frozen in its pain frame
// HIT-STOP (pick 13, off by default; 01_MELEE 3.6): a bash or a perfect block freezes the monster for a beat
server bool  ssq_hitstop        = false;
server int   ssq_hitstop_tics   = 4;      // 01_MELEE 3.6: 3 is a tap, 6 a heavy, 10 reads as a bug
// PERFECT-BLOCK SLOW MOTION (pick 14, off by default; apply/05_SLOWMO.md 8): half a second through RSQ SlowMo's
// rsm_pulse. Nothing hears it when SlowMo is not loaded.
server bool  ssq_perfect_slowmo = false;
// SABER-AND-SHIELD STANCE (pick 15; surveys/07_SHIELDS_PARRY.md 4.12): RSQ Saber's lit blade in the main hand
// resting across the plate's front widens the block. Without the saber nothing changes.
server bool  ssq_brace          = true;
server float ssq_brace_widen    = 1.3;    // the block's radius while braced, as a multiple
server float ssq_brace_gap      = 15.0;   // cm in front of the face the blade may rest and still count

// ---- the plate: where the disc is, in the off controller's frame, cm -------------
// F forward along the controller, L to the controller's left, U up. Defaults measured from
// shield.md3 as MODELDEF draws it (Scale -1 1 1, Offset 0 -24 -10), followed through the stock
// QZD 17.3 HUD model chain -- the derivation is in guard.zs. The face points outward, -L.
// ssq_debug_ring draws it, which is how these get checked in a headset.
server float ssq_plate_f        = -15.4;
server float ssq_plate_l        = -7.5;   // to the controller's RIGHT: the arm sits on the plate's +L side
server float ssq_plate_u        = -5.0;
server float ssq_plate_radius   = 29.2;   // cm, measured rim; the blades add ssq_plate_blades when revved
server float ssq_plate_thick    = 4.0;    // cm either side of the face that still counts as the shield (was 25: a slab
                                          // half a metre thick; the plate is a few cm)
server float ssq_plate_blades   = 6.0;    // deployed tips measure 35.2 cm
server float ssq_shield_scale   = 1.0;    // the whole shield, model and plate together: 1.0 .. 1.5
server float ssq_roll_sign      = 1.0;    // flip if the ring rolls the wrong way

// ---- per player (user: userinfo, travels) ---------------------------------------
user bool  ssq_start_drawn      = false;  // start with the shield in hand
user bool  ssq_feedback         = true;   // block feedback and haptics
user bool  ssq_spatial          = true;   // contacts are heard where they happen, not at your feet
