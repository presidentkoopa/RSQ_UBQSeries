// RSQ Gore options (design/GORE_SYSTEM_DESIGN.md section 6, design/MENU_STANDARD.md): ONE page with the choices a
// player makes, ONE "Test and Debug" page with the levels, budgets and the debug log; Reset All at the foot covers both
// (RSQGOResetAll reads the rows). What each setting does is in CVARINFO's comments.

// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQGO_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQGORSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "RSQ Gore", "RSQGO_Options"
}
// ----------------------------------------------------------------------------------------------------

OptionMenu "RSQGO_Options"
{
	Title "RSQ GORE"
	// tip: RSQ Gore takes Brutal Doom's blood over with the same look at a fraction of the cost; Off is Brutal Doom's own gore.
	Option "RSQ Gore", "rsqgo_on", "OnOff"
	StaticText " "
	// tip: A death's meat and limbs: All every piece, Some every second one, None none.
	Option "Gore", "rsqgo_amount", "RSQGO_Amount"
	// tip: Blood sprays, mist and jets on your screen only: All is Brutal Doom's look, Some half, None a puff and one drop a hit.
	Option "Spray And Mist", "rsqgo_fx", "RSQGO_Amount"
	// tip: Floor blood left where drops land: All keeps every spot, Some fewer, None lets each fade after a few seconds.
	Option "Blood That Stays", "rsqgo_residue", "RSQGO_Amount"
	// tip: Blood drips from ceiling splats.
	Option "Ceiling Drips", "rsqgo_drips", "OnOff"
	// tip: Blood splashes on your view when you stand by a hit.
	Option "Screen Blood", "rsqgo_screen", "OnOff"
	// tip: Picks the budget set: PC, or Quest (smaller budgets and the frame-time guard).
	Option "Platform", "rsqgo_platform", "RSQGO_Platform"
	StaticText " "
	// tip: Blood level, particle budgets, the frame-time guard, residue budgets and the debug log.
	Submenu "Test and Debug", "RSQGO_Test"
	// tip: Puts every RSQ Gore setting back to its default.
	RSQGOResetAll "Reset All", "Reset all RSQ Gore settings?"
}
