// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQGY_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQGYRSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "RSQ Graveyards", "RSQGY_Options"
}
// ----------------------------------------------------------------------------------------------------

OptionValue "RSQGY_LivingModes"
{
	0, "Uniform"
	1, "Wave"
	2, "Chaos"
}

OptionValue "RSQGY_BonesBlock"
{
	0, "Something On It"
	1, "Nothing"
}

OptionMenu "RSQGY_Options"
{
	Title "RSQ GRAVEYARDS"
	StaticText "Animation", "Gold"
	// tip: Gravestones breathe with a slow squash and bounce.
	Option "Animate Gravestones", "gy_anim_enabled", "OnOff"
	// tip: A new gravestone pops in instead of just appearing.
	Option "Pop In on Spawn", "gy_popin_enabled", "OnOff", "gy_anim_enabled"
	StaticText " "
	StaticText "Grave Light", "Gold"
	// tip: Each gravestone gives off a small light.
	Option "Grave Light", "gy_light_enabled", "OnOff"
	// tip: The grave light flickers like a candle.
	Option "Candle Flicker", "gy_light_flicker", "OnOff", "gy_light_enabled"
	// tip: Each stone's light and tint breathe slowly.
	Option "Living Stones", "gy_living", "OnOff"
	StaticText " "
	StaticText "Bones", "Gold"
	// tip: A Brutal Doom or XMVE monster's Legion of Bones undead rises from its corpse.
	RSQGYIfLoaded "Bones Bridge", "gy_bones", "OnOff", "SkullZombieSpawn"
	// tip: The share of corpses that rise, rolled once at the death; a corpse that loses stays dead.
	RSQGYIfLoadedSlider "Rise Chance", "gy_bones_chance", 0, 100, 5, 0, "SkullZombieSpawn"   // % (owner 2026-11-18: 50; 2026-11-27: 15); the bridge's and LoB's own
	StaticText " "
	StaticText "Controls", "Gold"
	// tip: Removes every gravestone on this map.
	Command "Remove Map Gravestones", "gy_remove_map"   // menu buttons, not keys (owner 11-10)
	// tip: Removes every stored gravestone on every map.
	Command "Remove All Gravestones", "gy_remove_all"
	StaticText " "
	// tip: Every other setting: animation and light tuning, the Living Stones mode and the Bones rules.
	Submenu "Test and Debug", "RSQGY_Test"
	// tip: Puts every RSQ Graveyards setting back to its default; the stored stones are kept.
	RSQGYResetAll "Reset All", "Reset all RSQ Graveyards settings?", "gy_living_debug gy_bones_debug"
}

// ---- RSQ Graveyards: Test and Debug (the series' menu standard, 2026-11-24: the tuning rows off the front page) ----
OptionMenu "RSQGY_Test"
{
	Title "RSQ GRAVEYARDS: TEST AND DEBUG"
	StaticText "Animation", "Gold"
	// tip: How deep the gravestones' squash goes; 0 is perfectly still.
	Slider "Bounce Amount", "gy_bounce_amount", 0.0, 0.25, 0.01, 2, "gy_anim_enabled"
	// tip: How fast the gravestones breathe (1.5 is about a seven second breath).
	Slider "Bounce Speed", "gy_bounce_speed", 0.2, 8.0, 0.1, 1, "gy_anim_enabled"
	StaticText " "
	StaticText "Grave Light", "Gold"
	// tip: How far the grave light reaches.
	Slider "Light Radius", "gy_light_radius", 16, 256, 8, 0, "gy_light_enabled"
	// tip: How bright the grave light is.
	Slider "Light Brightness", "gy_light_bright", 32, 255, 8, 0, "gy_light_enabled"
	// tip: No grave light exists past this distance, in map units.
	Slider "Light Cull Distance", "gy_light_cull", 128, 2048, 64, 0, "gy_light_enabled"
	// tip: How the stones breathe: all together, in a wave by distance, or each at its own pace.
	Option "Living Stones Mode", "gy_living_mode", "RSQGY_LivingModes", "gy_living"
	StaticText " "
	StaticText "Bones", "Gold"
	// tip: What stops a corpse rising: something standing on it (Legion of Bones' rule, the wait starts again), or nothing.
	RSQGYIfLoaded "What Stops a Rise", "gy_bones_block", "RSQGY_BonesBlock", "SkullZombieSpawn", "gy_bones"
	// tip: Each corpse rolls its own respawn delay at the death.
	RSQGYIfLoaded "Random Delay", "gy_bones_random", "OnOff", "SkullZombieSpawn"              // each corpse its own Respawn Delay (som_lobrnd folded in)
}
