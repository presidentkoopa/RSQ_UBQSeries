version "4.14"
// TEST ONLY (never ships). Turns against the body (11-03, the owner: "i can 360 around my torso and the shield
// doesn't move and the torso doesn't turn, or does sometimes"). Every tic: the pawn's angle, both controllers'
// yaws, RSQ TacticalRig's body yaw and RSQ ShieldSaw's (its keeper's and the synced one). trt_frames 1 also prints
// every rendered frame (the view yaw the renderer used and the controller yaws as written that frame).
class RSQTRTurnProbe : EventHandler
{
	// TEST ONLY: a level change with no intermission (the tally screen waits for a press the console script cannot time)
	override void NetworkProcess(ConsoleEvent e)
	{
		if (e.Name == "trt_nextmap") level.ChangeLevel("MAP02", 0, CHANGELEVEL_NOINTERMISSION);
	}
	override void WorldTick()
	{
		let p = players[0];
		let mo = p.mo;
		if (!mo) return;
		let rig = RSQTR_Rig.Get();
		let ss = SSQ_Handler.Get();
		double ry = rig ? rig.BodyYaw(mo) : 999;
		double sy = ss ? SSQ_Handler.BodyYaw(mo) : 999;
		double ky = ss ? ss.kYaw : 999;
		// the tracking space's yaw (TEST ONLY: the rig's headset yaw is the qz_hmd cvar's fifth number): pawn less head
		double trk = 999;
		let hc = CVar.FindCVar('qz_hmd');
		if (hc)
		{
			Array<String> f;
			hc.GetString().Split(f, " ", TOK_SKIPEMPTY);
			if (f.Size() >= 5) trk = Actor.Normalize180(mo.angle - f[4].ToDouble());
		}
		Console.Printf("trk t%d %.2f", level.maptime, trk);
		Console.Printf("trt t%d a %.2f h0 %.2f h1 %.2f rig %.2f ssk %.2f ssq %.2f pitch %.1f", level.maptime, mo.angle, mo.AttackAngle, mo.OffhandAngle, ry, ky, sy, mo.pitch);
		// TRR BEGIN (dropped for PROBE=old: the recenter, 11-03 18:56)
		// RECENTER (11-03 18:56): each recenter, each hand from the chest point (metres) and in the head's frame, the holster each hand is in
		if (rig && mo.player == players[consoleplayer])
		{
			Vector3 l0 = RSQTR_Body.HeadLocal(mo, mo.AttackPos), l1 = RSQTR_Body.HeadLocal(mo, mo.OffhandPos);
			let bb = rig.GetBody();
			Console.Printf("trr t%d rec %d gap %.3f %.3f hover %d %d main %.3f %.3f %.3f off %.3f %.3f %.3f ssrec %d", level.maptime, bb ? bb.recentered : -1, rig.recGap[0], rig.recGap[1], rig.HoverRegion(0), rig.HoverRegion(1), l0.x, l0.y, l0.z, l1.x, l1.y, l1.z, ss ? ss.kRecentered : -1);
		}
		// TRR END
		// TRQ BEGIN (dropped for PROBE=old: the 11-03 keeper's queues)
		let b = rig ? rig.GetBody() : null;
		if (b && b.turns && CVar.FindCVar('trt_queues').GetBool())
			Console.Printf("trq t%d head %.2f hands %.2f owed %.2f with %d/%d hp %.1f %.1f turn %d settle %.2f", level.maptime, RSQTR_TurnKeeper.Sum(b.turns.hAmt), RSQTR_TurnKeeper.Sum(b.turns.cAmt), RSQTR_TurnKeeper.Sum(b.turns.lAmt) + RSQTR_TurnKeeper.Sum(b.turns.mAmt), b.turns.withFor, b.turns.withOn, mo.AttackPitch, mo.OffhandPitch, b.turns.tOn, b.turns.lastSettle);
		// TRQ END
	}
	override void RenderOverlay(RenderEvent e)
	{
		let c = CVar.FindCVar('trt_frames');
		if (!c || !c.GetBool()) return;
		let mo = players[consoleplayer].mo;
		if (!mo) return;
		Console.Printf("trf g%d view %.2f h0 %.2f h1 %.2f", gametic, e.ViewAngle, mo.AttackAngle, mo.OffhandAngle);
	}
}
