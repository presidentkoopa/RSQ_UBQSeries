version "4.14"
// TEST ONLY (never ships). BD22 VR dual wield and assigned holsters: logs a TRD line on every change (and once a second):
// each hand's region, both hands' weapons, the one being raised, the Rig's queued off-hand weapon, and every gun holster's
// gun and assignment (RSQTR_Rig.HolsterList). TRD_F <frame> g<gametic> every 10 frames, to match the frame dumps.
class RSQTRDualProbe : EventHandler
{
	String last;
	override void WorldTick()
	{
		let p = players[0];
		let mo = p.mo;
		if (!mo) return;
		let rg = RSQTR_Rig.Get();
		if (!rg) return;
		String now = String.Format("hover %d/%d | main %s off %s pend %s offq %s | %s",
			rg.HoverRegion(0), rg.HoverRegion(1), p.ReadyWeapon ? p.ReadyWeapon.GetClassName() : 'none',
			p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none',
			(p.PendingWeapon && p.PendingWeapon != WP_NOCHANGE) ? p.PendingWeapon.GetClassName() : 'nc',
			rg.offNext[0] != "" ? rg.offNext[0] : "-", rg.HolsterList(mo));
		// where the hips are, from the eyes, in room metres (as holsterprobe): the script's poses come from this line
		if (level.maptime == 30)
			for (int k = 0; k < 2; k++)
			{
				double f, l, u;
				rg.BodyOffsets(mo, rg.RegionAt(mo, k), f, l, u);
				Console.Printf("TRD region %d %s at %.2f forward %.2f left %.2f up from the eyes (eyes %.2f m over the floor), radius %.2f", k, RSQTR_Region.NAMES[k], f, l, u, (p.viewz - mo.floorz) * RSQTR_Body.Ps() / RSQTR_Body.Upm(), rg.RegionRadius(p, k));
			}
		if (now != last || level.maptime % 35 == 0)
		{
			Console.Printf("TRD t%d %s", level.maptime, now);
			last = now;
		}
	}
	ui int frames;
	override void RenderOverlay(RenderEvent e) { frames++; if (frames % 10 == 0) Console.Printf("TRD_F %d g%d", frames, gametic); }
}
