version "4.14"
// TEST ONLY (never ships). Tactical Rig cigarette ritual probe.
//   every tic: the cigarette's phase changes; every 5 tics while it is out: the hands against the mouth zone and the
//   tip, the Zippo, the mouth prop, smoke; flicked cigarettes in flight; RSQ_SlowMo's answer (asked by name).
//   trc_mode 1: stands the player in front of the nearest use-door and logs its ceiling (door opened or not).
class RSQTRCigProbe : EventHandler
{
	bool useWas;
	int lastPhase, lastActive, lastScale;
	Line door;
	int setup;
	double ceil0;
	override void WorldTick()
	{
		let p = players[0];
		let mo = p.mo;
		if (!mo) return;
		let c = RSQTR_Cig.Get();
		if (!c) return;
		int ph = c.phase[0];
		if (ph != lastPhase)
		{
			Console.Printf("trc: t%d phase %d -> %d  main %s off %s packs %d", level.maptime, lastPhase, ph,
				p.ReadyWeapon ? p.ReadyWeapon.GetClassName() : 'none', p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none', RSQTR_Cig.Packs(mo));
			lastPhase = ph;
		}
		bool useDown = (p.cmd.buttons & BT_USE) != 0;
		if (p.cmd.buttons & BT_OFFHANDATTACK) Console.Printf("trc: t%d off-hand fire held", level.maptime);
		if (useDown != useWas) Console.Printf("trc: t%d Use %s (phase %d, main %s, off %s)", level.maptime, useDown ? "pressed" : "released", ph, p.ReadyWeapon ? p.ReadyWeapon.GetClassName() : 'none', p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none');
		useWas = useDown;
		if (level.maptime % 35 == 0)
		{
			Vector3 h = RSQTR_Cig.Eyes(mo);
			double u = RSQTR_Body.Upm();
			Vector3 pk = c.Pocket(mo) - mo.AttackPos;
			Vector3 a = (mo.AttackPos - h) / u, o = (mo.OffhandPos - h) / u;
			Console.Printf("trc: t%d override %d main hand from the eyes (%.3f %.3f %.3f) m, off hand (%.3f %.3f %.3f) m (map x y, z unstretched x1.2), angle %.0f; packs %d, main hand %.3f m from the pocket, rsqtr_cig %d; pocket minus hand (%.1f %.1f %.1f) units", level.maptime, mo.OverrideAttackPosDir, a.x, a.y, a.z * 1.2, o.x, o.y, o.z * 1.2, mo.angle, RSQTR_Cig.Packs(mo), RSQTR_Body.RoomMetres(mo.AttackPos, c.Pocket(mo)), CVar.GetCVar("rsqtr_cig", p).GetInt(), pk.x, pk.y, pk.z);
		}
		if ((ph != 0 && level.maptime % 5 == 0) || (useDown && level.maptime % 5 == 0))
		{
			Vector3 h = RSQTR_Cig.Eyes(mo), m = RSQTR_Cig.Mouth(mo);
			Console.Printf("trc: t%d ph %d use %d usedown %d | head (%.1f %.1f %.1f) pitch %.0f mouth (%.1f %.1f %.1f) | zone: controller %.3f, drawn filter %.3f m from its axis (radius %.3f), in %d | flame %.3f m from the cigarette | zippo open %d lit %d seen %d off %s | main %s pend %s",
				level.maptime, ph, useDown, p.usedown, h.x, h.y, h.z, mo.pitch, m.x, m.y, m.z, RSQTR_Cig.ZoneDist(mo, mo.AttackPos), RSQTR_Cig.ZoneDist(mo, RSQTR_Cig.CigFilter(mo)), RSQTR_Cig.ZoneRadius(mo), RSQTR_Cig.InMouthZone(mo), RSQTR_Cig.FlameDist(mo),
				c.zipOpen[0], c.ZipBurning(0), c.zipSeen[0], p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none',
				p.ReadyWeapon ? p.ReadyWeapon.GetClassName() : 'none', p.PendingWeapon && p.PendingWeapon != WP_NOCHANGE ? p.PendingWeapon.GetClassName() : 'none');
			let mp = RSQTR_CigProp(c.mouthProp[0]);
			if (mp) Console.Printf("trc: t%d mouth cigarette at (%.1f %.1f %.1f) burning %d sprite frame %d; smoke puffs alive %d", level.maptime, mp.pos.x, mp.pos.y, mp.pos.z, mp.burning, mp.frame, Count("RSQTR_Smoke"));
		}
		// trc_marks 1: dots at the COMPUTED drawn points (green: cigarette tip, red: filter, blue: Zippo flame) and a
		// line every 5 tics of where the hands sit in the head's frame (metres forward / left / down of the eyes)
		let mk = CVar.FindCVar('trc_marks');
		if (mk && mk.GetInt() == 1)
		{
			if (!dots[0]) for (int k = 0; k < 3; k++) { dots[k] = Actor.Spawn("TRCDot", mo.pos); if (dots[k]) { dots[k].A_SetTranslation(k == 0 ? "TRCGreen" : (k == 1 ? "TRCRed" : "TRCBlue")); if (k == 2) dots[k].Scale *= 2.5; } }
			if (dots[0]) dots[0].SetOrigin(RSQTR_Cig.CigTip(mo), true);
			if (dots[1]) dots[1].SetOrigin(RSQTR_Cig.CigFilter(mo), true);
			if (dots[2]) dots[2].SetOrigin(RSQTR_Cig.Flame(mo), true);
		}
		if (level.maptime % 5 == 0 && ph != 0)
		{
			Vector3 e = RSQTR_Cig.Eyes(mo);
			Vector3 a = HeadFrame(mo, mo.AttackPos - e), f = HeadFrame(mo, RSQTR_Cig.CigFilter(mo) - e), t = HeadFrame(mo, RSQTR_Cig.CigTip(mo) - e), fl = HeadFrame(mo, RSQTR_Cig.Flame(mo) - e), o = HeadFrame(mo, mo.OffhandPos - e);
			Console.Printf("trc: t%d head frame (fwd left down, m): main controller (%.3f %.3f %.3f) drawn filter (%.3f %.3f %.3f) drawn tip (%.3f %.3f %.3f) | off controller (%.3f %.3f %.3f) drawn flame (%.3f %.3f %.3f) | hand points %.0f from the head (AttackAngle+90), pitch %.0f roll %.0f",
				level.maptime, a.x, a.y, a.z, f.x, f.y, f.z, t.x, t.y, t.z, o.x, o.y, o.z, fl.x, fl.y, fl.z, Actor.DeltaAngle(mo.angle, mo.AttackAngle + 90), mo.AttackPitch, mo.AttackRoll);
		}
		// trc_weap 1: the hands and every pistol / dual weapon owned, every 10 tics and on any change (the holster
		// pair test, 11-02)
		let wv = CVar.FindCVar('trc_weap');
		if (wv && wv.GetInt() == 1)
		{
			String inv = "";
			for (let it = mo.Inv; it; it = it.Inv)
			{
				let w = Weapon(it);
				if (!w) continue;
				String cn = w.GetClassName(); String lc = cn.MakeLower();
				if (lc.IndexOf("pistol") >= 0 || lc.IndexOf("dual") >= 0 || lc.IndexOf("smg") >= 0) inv = inv .. cn .. " ";
			}
			let rg = RSQTR_Rig.Get();
			String hol = rg ? rg.Content(mo, 0) .. "|" .. rg.Content(mo, 1) : "";
			String now = String.Format("main %s off %s pend %s | holsters L|R %s | owned %s",
				p.ReadyWeapon ? p.ReadyWeapon.GetClassName() : 'none', p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none',
				p.PendingWeapon && p.PendingWeapon != WP_NOCHANGE ? p.PendingWeapon.GetClassName() : 'none', hol, inv);
			if (now != lastWeap || level.maptime % 35 == 0) { Console.Printf("trc: t%d weap %s", level.maptime, now); lastWeap = now; }
		}
		StrayWatch(p, mo, c);
		let rp = CVar.FindCVar('trc_rep');
		if (rp && rp.GetInt() != 0) { Report(p, mo, c, String.Format("#%d", rp.GetInt())); rp.SetInt(0); }
		// flicked cigarettes
		if (level.maptime % 3 == 0)
		{
			let it = ThinkerIterator.Create("RSQTR_Butt");
			Actor a;
			while (a = Actor(it.Next()))
				Console.Printf("trc: t%d %s flying at (%.1f %.1f %.1f) vel (%.1f %.1f %.1f) pitch %.0f, %.0f units from the player, embers %d", level.maptime, a.GetClassName(), a.pos.x, a.pos.y, a.pos.z, a.vel.x, a.vel.y, a.vel.z, a.pitch, a.Distance2D(mo), Count("RSQTR_Ember"));
		}
		if (level.maptime % 35 == 0)
		{
			let it = ThinkerIterator.Create("RSQTR_ButtLying");
			Actor a;
			while (a = Actor(it.Next()))
				Console.Printf("trc: t%d %s lying at (%.1f %.1f %.1f), %.0f units from the player, lure %d", level.maptime, a.GetClassName(), a.pos.x, a.pos.y, a.pos.z, a.Distance2D(mo), RSQTR_ButtLying(a).lure);
		}
		// RSQ_SlowMo, by name
		let sit = ServiceIterator.Find("RSQSM_Service");
		Service s;
		int act = -2, sc = -2;
		while (s = sit.Next()) { act = s.GetInt("slowmo.active", "", 0, 0, null); sc = s.GetInt("scale", "", 0, 0, null); }
		if (act != lastActive || (act == 1 && level.maptime % 10 == 0 && sc != lastScale))
		{
			Console.Printf("trc: t%d slowmo service %s active %d world speed %d permille", level.maptime, act == -2 ? "absent" : "found", act, sc);
			lastActive = act; lastScale = sc;
		}
		// trc_steal 1: another mod takes the off hand (the Zippo out of it), once
		let st = CVar.FindCVar('trc_steal');
		if (st && st.GetInt() == 1)
		{
			st.SetInt(0);
			Console.Printf("trc: t%d the probe empties the off hand (was %s)", level.maptime, p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none');
			p.SetPsprite(PSP_OFFHANDWEAPON, null); p.OffhandWeapon = null;
		}
		// trc_next 1: the next level, no intermission (the inventory carries over)
		let nx = CVar.FindCVar('trc_next');
		if (nx && nx.GetInt() == 1)
		{
			nx.SetInt(0);
			Console.Printf("trc: t%d to the next level, phase %d, main %s off %s packs %d", level.maptime, ph, p.ReadyWeapon ? p.ReadyWeapon.GetClassName() : 'none', p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none', RSQTR_Cig.Packs(mo));
			level.ChangeLevel(level.NextMap, 0, CHANGELEVEL_NOINTERMISSION);
		}
		if (level.maptime == 2 || level.maptime == 40)
			Console.Printf("trc: t%d level %s start: phase %d main %s pend %s off %s packs %d", level.maptime, level.MapName, ph, p.ReadyWeapon ? p.ReadyWeapon.GetClassName() : 'none', p.PendingWeapon && p.PendingWeapon != WP_NOCHANGE ? p.PendingWeapon.GetClassName() : 'none', p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none', RSQTR_Cig.Packs(mo));
		// trc_mode 1: the door
		let cv = CVar.FindCVar('trc_mode');
		if (cv && cv.GetInt() == 1)
		{
			if (!setup && level.maptime > 10)
			{
				setup = 1;
				double best = 1e9;
				for (int k = 0; k < level.Lines.Size(); k++)
				{
					let ln = level.Lines[k];
					if (!ln.backsector || !(ln.activation & SPAC_Use)) continue;
					if (ln.special != 12 && ln.special != 11 && ln.special != 202 && ln.special != 10 && ln.special != 105) continue;
					if (ln.locknumber || ln.args[3] || (ln.special == 202 && ln.args[4])) continue;
					Vector2 mid = ln.v1.p + ln.delta * 0.5;
					double d = (mid - mo.pos.xy).Length();
					if (d < best) { best = d; door = ln; }
				}
				if (door)
				{
					Vector2 mid = door.v1.p + door.delta * 0.5;
					Vector2 nrm = (door.delta.y, -door.delta.x).Unit();
					Vector2 at = mid + nrm * 24;
					mo.SetOrigin((at.x, at.y, mo.pos.z), false);
					mo.SetZ(mo.floorz);
					mo.angle = atan2(-nrm.y, -nrm.x);
					ceil0 = door.backsector.ceilingplane.ZatPoint(mid);
					Console.Printf("trc: door line %d special %d at (%.0f %.0f), player at (%.0f %.0f) facing it, ceiling %.1f", door.Index(), door.special, mid.x, mid.y, mo.pos.x, mo.pos.y, ceil0);
				}
			}
			if (door && level.maptime % 10 == 0)
			{
				Vector2 mid = door.v1.p + door.delta * 0.5;
				double cz = door.backsector.ceilingplane.ZatPoint(mid);
				Console.Printf("trc: t%d door ceiling %.1f (%s)", level.maptime, cz, cz > ceil0 + 1 ? "OPENING/OPEN" : "shut");
			}
		}
	}
	// trc_rep 1 (11-03, rule A/B scenarios): one line of everything the cigarette could leave behind; and every tic a
	// STRAY watch: a cigarette / Zippo drawn or lying where the ritual does not put it
	String lastStray;
	int strayFor;
	String LayerOf(PlayerInfo p, int id)
	{
		let ps = p.FindPSprite(id);
		return (ps && ps.Caller) ? String.Format("%s", ps.Caller.GetClassName()) : "none";
	}
	int CountFree(Name cls)
	{
		int k = 0;
		let it = ThinkerIterator.Create(cls);
		Inventory a;
		while (a = Inventory(it.Next())) if (!a.Owner) k++;
		return k;
	}
	void Report(PlayerInfo p, PlayerPawn mo, RSQTR_Cig c, String tag)
	{
		let rig = RSQTR_Rig.Get();
		Console.Printf("trc: REP %s t%d phase %d health %d state %d | main %s pend %s off %s offNext %s | zippo open %d | inv cig %d zippo %d | packs %d | mouthprop %d flying %d lying %d world-cigweapon %d world-zippo %d | drawn main-layer %s off-layer %s",
			tag, level.maptime, c.phase[0], mo.health, p.playerstate, p.ReadyWeapon ? p.ReadyWeapon.GetClassName() : 'none',
			p.PendingWeapon && p.PendingWeapon != WP_NOCHANGE ? p.PendingWeapon.GetClassName() : 'none',
			p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none', rig ? rig.offNext[0] : "?", c.zipOpen[0],
			mo.CountInv("RSQTR_CigWeapon"), mo.CountInv("RSQTR_ZippoWeapon"), RSQTR_Cig.Packs(mo),
			Count("RSQTR_CigProp"), Count("RSQTR_Butt"), Count("RSQTR_ButtLying"), CountFree("RSQTR_CigWeapon"), CountFree("RSQTR_ZippoWeapon"),
			LayerOf(p, PSP_WEAPON), LayerOf(p, PSP_OFFHANDWEAPON));
	}
	void StrayWatch(PlayerInfo p, PlayerPawn mo, RSQTR_Cig c)
	{
		int ph = c.phase[0];
		String m = LayerOf(p, PSP_WEAPON), o = LayerOf(p, PSP_OFFHANDWEAPON);
		String s = "";
		if (CountFree("RSQTR_CigWeapon") > 0) s = s .. "a cigarette weapon lying in the world; ";
		if (CountFree("RSQTR_ZippoWeapon") > 0) s = s .. "a Zippo weapon lying in the world; ";
		if (Count("RSQTR_CigProp") > 0 && ph != 2 && ph != 3) s = s .. "a mouth cigarette with no cigarette in the mouth; ";
		if (o ~== "RSQTR_CigWeapon") s = s .. "the cigarette drawn in the OFF hand; ";
		if (m ~== "RSQTR_ZippoWeapon") s = s .. "the Zippo drawn in the MAIN hand; ";
		if (ph == 0 && (m ~== "RSQTR_CigWeapon" || o ~== "RSQTR_ZippoWeapon")) s = s .. "cigarette/Zippo drawn with no cigarette out; ";
		if (ph != 0 && ph != 1 && m ~== "RSQTR_CigWeapon") s = s .. "a cigarette drawn in the hand while it is in the mouth; ";
		strayFor = s == "" ? 0 : strayFor + 1;
		// a hand lowering a prop for a few tics after a switch is not a stray: only what lasts 8 tics is logged
		String rep = strayFor >= 8 ? s : "";
		if (rep != lastStray)
		{
			Console.Printf("trc: t%d STRAY %s", level.maptime, rep == "" ? "(none)" : rep);
			lastStray = rep;
		}
	}
	Actor dots[3];
	String lastWeap;
	// a map offset as metres forward / left / down in the head's level frame
	Vector3 HeadFrame(PlayerPawn mo, Vector3 d)
	{
		double u = RSQTR_Body.Upm();
		double c = cos(mo.angle), s = sin(mo.angle);
		return ((d.x * c + d.y * s) / u, (-d.x * s + d.y * c) / u, -d.z * RSQTR_Body.Ps() / u);
	}
	int Count(Name cls)
	{
		int k = 0;
		let it = ThinkerIterator.Create(cls);
		while (it.Next()) k++;
		return k;
	}
}

class TRCDot : Actor
{
	Default { +NOINTERACTION +NOBLOCKMAP +NOGRAVITY +FORCEXYBILLBOARD RenderStyle "Add"; Scale 0.035; Alpha 1.0; }
	States { Spawn: PUFF A -1 Bright; Stop; }
}
