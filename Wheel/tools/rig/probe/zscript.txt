version "4.10"
// TEST ONLY (never ships): logs what the wheel is doing, through its public service, every 5 tics.
//   WRQ <tic> ready=<class> off=<class> open=<0|1> hover=<class> quads=<n> strips=<n> glyphs=<n>
class wrq_Probe : EventHandler
{
	override void WorldTick()
	{
		if (level.maptime % 5) return;
		let pmo = players[consoleplayer].mo;
		if (!pmo) return;
		let it = ServiceIterator.Find("wr_RigService");
		let svc = it.Next();
		int open = svc ? svc.GetInt("IsOpen") : -1;
		Object hov = svc ? svc.GetObject("HoveredWeapon", "", 0, 0, pmo) : null;
		int quads = 0, strips = 0;
		let ti = ThinkerIterator.Create("Actor");
		Actor a;
		while (a = Actor(ti.Next()))
		{
			string n = a.GetClassName();
			if (n == "wr_BBStrip") { if (!a.bInvisible) strips++; }
			else if (n == "wr_BBQuad") { if (!a.bInvisible) quads++; }
		}
		// the series, by name: where TacticalRig holds a Rifle, whether RSQ SlowMo is running
		int hol = -2, slow = -2;
		let ti2 = ServiceIterator.Find("RSQTR_RigService"); let s2 = ti2.Next();
		if (s2) hol = s2.GetInt("rig.holster", "Rifle", 0, 0, pmo);
		let ti3 = ServiceIterator.Find("RSQSM_Service"); let s3 = ti3.Next();
		if (s3) slow = s3.GetInt("active");
		if (hol != -2 || slow != -2) Console.Printf("WRQS %d holsterRifle=%d slowmo=%d", level.maptime, hol, slow);
		if (svc && open == 1)
			Console.Printf("WRQA %d anchor=(%.0f %.0f) angle=%.0f", level.maptime, svc.GetDouble("AnchorX"), svc.GetDouble("AnchorY"), pmo.angle);
		let rw = players[consoleplayer].ReadyWeapon;
		let ow = players[consoleplayer].OffhandWeapon;
		Console.Printf("WRQ %d ready=%s off=%s open=%d hover=%s quads=%d strips=%d frozen=%d",
			level.maptime, rw ? rw.GetClassName() : 'none', ow ? ow.GetClassName() : 'none', open,
			hov ? hov.GetClassName() : 'none', quads, strips,
			(players[consoleplayer].cheats & CF_TOTALLYFROZEN) ? 1 : 0);
	}
}
