// WHAT THE DATA SHEET READS OUT OF OTHER WEAPON MODS.
//
// Its own page because these are nine switches that only matter to whoever
// has that particular mod loaded -- on the main page they would be nine rows
// of noise for everyone else. Every one is ON by default and costs nothing
// when the mod it names is absent: the sheet asks whether a field exists on
// the weapon in your hand, and a mod that is not loaded simply has no field
// to find, so the row never appears. Turning one off here is for when you
// have the mod AND would rather not see its rows.
OptionMenu wr_Compat
{
	Title "RSVR HUD - WEAPON MOD DATA"

	StaticText ""
	StaticText "Mods with a rarity ladder", gold
	// These three also COLOUR the card and the sheet title by their own
	// rarity, which is why each has a separate switch for its extra rows:
	// you may want the colour without the detail.
	Option "LegenDoom rarity",          "wr_ld_compat",    "OnOff"
	Option "  Rolled effects row",      "wr_ld_effects",   "OnOff"
	Option "DoomRL Arsenal tier",       "wr_drla_compat",  "OnOff"
	Option "  Mod Station upgrades row", "wr_drla_mods",   "OnOff"
	// Rarity, plus your character level, XP, Inferno level and rolled stats.
	Option "Doomablo rarity and level", "wr_dbl_compat",   "OnOff"
	// Rolled damage/rate/spread/pellets/crit, exotic ammo, fire modes, alt
	// modules, quirks, overheat and the run itself. Its stats are readable on
	// a weapon still on the floor, so this is the mod the inspect card and
	// the comparison table were really built for.
	Option "DOOM Infinite weapon rolls", "wr_di_compat",   "OnOff"

	StaticText ""
	StaticText "Mods with no rarity ladder", gold
	// None of these six colour anything -- they have no tier concept at all,
	// confirmed by survey rather than assumed. They only add rows.
	//
	// One switch for the whole Pandemonia family: base game durability,
	// magazine and sidegrades, Anarchy's Anarchic Sigil levelling, and
	// Insurrection's augments, durability and charge bars.
	Option "Pandemonia family",         "wr_pand_compat",  "OnOff"
	// Spell cooldown and the five player resource pools.
	Option "Guncaster resources",       "wr_gc_compat",    "OnOff"
	// Plasma rifle heat, and Unmaker demon keys.
	Option "MetaDoom heat and keys",    "wr_meta_compat",  "OnOff"
	// Damage, accuracy, rate of fire, recoil, clip and rolled level -- read
	// from the array BorderDoom's own stat lookup writes into, never from
	// that lookup itself, which has side effects on your ammo capacity.
	Option "BorderDoom weapon stats",   "wr_bd_compat",    "OnOff"
	// All four classes and their meters -- BlastMaster heat (which also
	// takes the sheet's gauge), Artificer skill and mace, the Tech Monk's
	// Striker and Kame Bomb, Past Linked's subweapon carousel and its three
	// upgrade ladders. Includes the warning the game itself only gives you
	// once: which upgrades vanish when you leave the level.
	Option "Combined Arms class meters", "wr_ca_compat",   "OnOff"
	// Nine weapon sets, one active at a time. Coverage is uneven on purpose
	// -- Whitemare, Hellbound, JPCP, Alien Vendetta and BTSX have real live
	// counters worth a row; Doom II and Aliens TC are fixed math with
	// nothing to read past which set is active.
	Option "Final Doomer set mechanics", "wr_fd_compat",   "OnOff"
	// Seven characters, 48 weapons. Magazines, mana, heat, charge, spin-up
	// and status effects. Nothing reads off a Lithium weapon on the floor --
	// that mod deletes unowned weapon actors.
	// Ours, not a third party -- a real two-way bridge, not a guess.
	// Cycle keys are under the main RSVR HUD key section, unbound by
	// default.
	Option "ModelSwapper model on card", "wr_ms_compat",  "OnOff"
	Option "Lithium weapons",           "wr_lith_compat", "OnOff"
	// Separate switch, because this is the only setting in the whole mod that
	// calls another mod's own scripts rather than reading its data. Five gates
	// sit in front of the call and the script was disassembled to prove it
	// cannot mutate anything -- but turn this off if a Lithium update lands
	// before an update to this, or if those rows ever look wrong.
	Option "  Lithium RPG stats (ACS)", "wr_lith_acs",    "OnOff"

	StaticText ""
}
