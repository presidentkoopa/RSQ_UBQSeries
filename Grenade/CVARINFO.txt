// RSQ Grenade -- settings. What each does (the menu carries no notes). Server: they change the game for everyone
// (what is given, scattered, how the grenade flies and burns). User: how YOUR hands are read.
server bool   rsqgn_on          = true;   // the whole grenade (with Brutal Doom VR Reloaded loaded it stands down, unless rsqgn_replace_bd); the barrels have their own rsqgn_barrel_on
// Replace Brutal Doom's Grenade (owner, BUGS_1.0 item 21, 2026-11-06): off, this pack stands down for BD22 VR Reloaded's own grenade (as
// before). On, it does NOT stand down: RSQ Grenade runs with BD22 VR loaded. THE HANDSHAKE: BD22 VR Reloaded reads this cvar BY NAME
// (CVar.FindCVar("rsqgn_replace_bd"): null when RSQ Grenade is not loaded) and stands its own grenade down while it is true.
server bool   rsqgn_replace_bd  = false;
server int    rsqgn_start       = 3;      // grenades you start with (a new game, or after dying); 0-8
server bool   rsqgn_scatter     = true;   // grenades scattered beside each map's items (the UAC Survival Pack's way)
server int    rsqgn_singles     = 4;      // ...at most this many single grenades a map (the pack's uac_grenamt)
server int    rsqgn_packs       = 2;      // ...at most this many packs of three a map (the pack's uac_grenpkamt)
server float  rsqgn_fuse        = 3.0;    // seconds from the spoon flying off to the blast (RS_Grenade's rsvg_fuse)
server float  rsqgn_gravity     = 0.27;   // the thrown grenade's gravity (1 = Doom's; 0.37 is earth's) (RS_Grenade)
server float  rsqgn_bounce      = 0.5;    // speed kept off a floor (RS_Grenade, BD's HandGrenade)
server float  rsqgn_wallbounce  = 0.25;   // speed kept off a wall (RS_Grenade)
server float  rsqgn_friction    = 0.82;   // ground drag, per tic on the floor (1 = ice) (RS_Grenade)
user   float  rsqgn_swing       = 2.5;    // m/s: let go faster than this and it is thrown, slower and the spoon flies (BD22 VR THROW_MIN)
user   float  rsqgn_throw       = 1.0;    // throw strength: x the hand's own speed (RS_Grenade's rsvg_throw)
user   float  rsqgn_lift        = 0.35;   // arc: this much of the throw's speed across the ground added upward (RS_Grenade)
user   bool   rsqgn_face        = true;   // pull the pin with your teeth (RS_Grenade's face route)
user   float  rsqgn_mouth_r     = 0.15;   // m: the controller, the drawn grenade or its ring this close to your mouth (lips to under the chin, from the headset) counts as at your mouth (11-03; replaces rsqgn_face_near, 0.47 m round the pawn's fixed eye)
user   float  rsqgn_face_far    = 0.71;   // m it must have been clear of your mouth before another (RS_Grenade's 24 map units)
user   int    rsqgn_face_hold   = 12;     // tics it must stay at your face (RS_Grenade)
user   bool   rsqgn_kind_alt    = true;   // the holding hand's alt fire cycles the grenade kinds (off: it cycles nothing)
server int    rsqgn_kinds_free  = 1;      // 1 All Types (the alt fire cycles every kind; with Gun Bonsai VR every kind you have learned), 0 Frag Only (with or without it)
server bool   rsqgn_shoot       = true;   // a player's shot or projectile sets off a thrown grenade on the spot (in the air: a skeet), 6 tics out of the hand, 0.35 m round it
server float  rsqgn_shot_r      = 0.35;   // m: how near a shot must pass a thrown grenade to set it off (0.10 to 0.35; never more)
server bool   rsqgn_shoot_slowmo = false; // ...only while the world is slowed (RSQ SlowMo)
server bool   rsqgn_slowmo_fast = false;  // in slow motion a thrown grenade runs faster than the world (still slow)
server float  rsqgn_slowmo_speed = 0.5;   // ...this much of the way from the world's speed up to real speed (0 the world's, 1 real time)
user   bool   rsqgn_skeet_window = true;  // Gun Bonsai VR's Skeet Window: its slow motion after a throw (nothing without the node)
server int    rsqgn_fire_self   = 0;      // Incendiary's burning ground on players: 0 Auto (yes with Brutal Doom), 1 yes, 2 no
user   bool   rsqgn_debug       = false;  // log the pin, the fuse, the flashes, the throw and the scatter to the console
// THE BARRELS (barrel.zs; owner 11-09). All SERVER: the effects are spawned in the play simulation, so every machine in
// a netgame must read the same values (a user setting here would let two machines' games differ).
server bool   rsqgn_barrel_on   = true;   // our barrel: the blink and one of the six explosions (off: the stock barrel, untouched)
server int    rsqgn_barrel_kind = 0;      // 0 Random (one of the six each time), or always 5 Snap, 6 Heavy, 7 Pillar, 8 Groundbreaker, 9 Slag Column, 10 Neon Crescent
// Replace Brutal Doom's Barrels: off, with Brutal Doom loaded (its ExplosiveBarrel1) its barrels stay its own and ours stand
// down; on, ours replace both ExplosiveBarrel and BD's ExplosiveBarrel1 (barrel.zs RSQGN_BarrelSwap says how and the limits)
server bool   rsqgn_barrel_replace_bd = false;
server float  rsqgn_barrel_debris = 1.0;  // how much of each blast's debris is thrown (0 none to 1 all the design has)
server int    rsqgn_barrel_cap  = 600;    // most barrel effect actors alive at once (Quest: a long chain reaction stays cheap); past it a blast shows its first tic only
