// TEST ONLY (never ships): the Fists tree on the headless QZD 17.3 gate. Grants Gun Bonsai VR nodes to both gloves as a
// level-up would, prints which Fists nodes each glove would be offered, lands punches through RSQ Melee's own Deliver
// (public for probes), and sends RSQ Gestures' two-hand event as Gestures would. Every line it prints starts "FP ".
class FistsProbe : EventHandler
{
	static const String NODES[] = {
		"HeavyHands", "Haymaker", "BodyBlow", "Uppercut", "Launcher", "Juggle", "Meteor", "GroundPound", "ClearOut",
		"WallSplat", "Knockdown", "StarPunch", "OneTwo", "Counterpunch", "ReturnFire",
		"ExplosiveFist", "FrostFist", "ShockFist", "FireFist",
		"Executioner", "ThreeHit", "Dempsey", "Finisher",
		"Guard", "Parry", "ReflectGuard", "AbsorbRelease", "SwatBack",
		"HumanShield", "HeadlockUppercut", "HeavierHeadlock", "Slam", "ThrowInto", "Chokehold", "Uppercut",
		"Rage", "BloodRush", "BerserkMeter", "PistolWhip", "RifleButt", "Taunt", "CriticalHits", "Headshot", "Backstab",
		"Catch", "BulletTimeFists"
	};
	bool punchHigh;
	Array<String> fxSeen;
	Array<int> fxN;
	// the look: every spark and burst spawned, and how long each lived (frames by level)
	Array<Actor> fx;
	Array<String> fxName;
	Array<int> fxBorn;

	override void WorldThingSpawned(WorldEvent e)
	{
		let a = e.Thing;
		// the trees' own art, counted by class as it spawns (fp_fxcount prints it)
		if (a)
		{
			String cn = a.GetClassName();
			if (cn.Left(6) == "RSQGN_" && (cn == "RSQGN_SmallBlast" || cn == "RSQGN_BigBlast" || cn == "RSQGN_FrostBurst" || cn == "RSQGN_Flame" || cn == "RSQGN_Puff" || cn == "RSQGN_Bloom" || cn == "RSQGN_Haze")
				|| cn == "SSQ_ChargeWave" || cn == "SSQ_DashStreak")
			{
				int i = fxSeen.Find(cn);
				if (i == fxSeen.Size()) { fxSeen.Push(cn); fxN.Push(0); }
				fxN[i]++;
			}
		}
		if (!a || !(a is "RSQML_Spark")) return;
		fx.Push(a); fxName.Push(a.GetClassName()); fxBorn.Push(level.maptime);
	}
	void FxReport()
	{
		for (int i = fx.Size() - 1; i >= 0; i--)
		{
			let a = fx[i];
			if (a && level.maptime == fxBorn[i] + 1)
				Console.Printf("FP fx t%d %s frames %d%s scale %.3f", fxBorn[i], fxName[i], a.args[0], a.bXFlip ? " MIRRORED" : "", a.Scale.x);
			if (!a) { Console.Printf("FP fx %s lived %d tics", fxName[i], level.maptime - fxBorn[i]); fx.Delete(i); fxName.Delete(i); fxBorn.Delete(i); }
		}
	}
	// the off hand on its own (the Headlock): where it is (forward, left of the pawn; up from viewz) and how it moves
	int offMode, offLeft;
	double offF, offS, offU, offVF, offVU;
	Actor held, other;
	// the Guard: 0 hands down (no sim), 1 hands up at the face, 2 the same pushed forward for PUSH tics
	int guardMode, pushLeft;
	double pushFwd;
	Actor ball;
	int swatAt, hadoAt;
	// a queued sequence of punches on a1: kinds, alternating hands from the main, every SEQ_GAP tics
	Array<int> seqKind;
	int seqNext, seqHand;
	const SEQ_GAP = 6;
	Actor a1, a2;
	int jugAt, mtAt;

	PlayerPawn P() { return players[0].mo; }
	Vector3 Ahead(double d, double side = 0)
	{
		let p = P();
		double c = cos(p.angle), s = sin(p.angle);
		return p.pos + (d * c - side * s, d * s + side * c, 0);
	}
	Actor Put(String cls, double d, double side = 0)
	{
		let a = Actor.Spawn(cls, Ahead(d, side), ALLOW_REPLACE);
		if (a) { a.angle = a.AngleTo(P()); }
		return a;
	}
	RSQ_Glove Glove(int h) { let p = P(); return RSQ_Glove(h == 0 ? p.player.ReadyWeapon : p.player.OffhandWeapon); }
	TFLV_WeaponInfo Info(int h)
	{
		let g = Glove(h);
		let st = TFLV_PerPlayerStats.GetStatsFor(P());
		return (g && st) ? st.GetInfoFor(g) : null;
	}

	override void WorldThingDamaged(WorldEvent e)
	{
		if (!e.Thing || e.Thing.player) return;
		let th = UBQS_Thrown(e.Inflictor);
		if (th) Console.Printf("FP t%d thrown saber phase %d hit %s for %d", level.maptime, th.phase, e.Thing.GetClassName(), e.Damage);
		String inf = e.Inflictor ? e.Inflictor.GetClassName() : 'none';
		Console.Printf("FP t%d DAMAGED %s by %s: %d, health now %d, z %.0f", level.maptime, e.Thing.GetClassName(), inf, e.Damage, e.Thing.health, e.Thing.pos.z - e.Thing.floorz);
	}

	// the ShieldSaw's dash: where it started, when to read it, what stood at the end of it
	int ssDashAt;
	Vector3 ssDashFrom;
	Actor ssDashImp;
	int ssDashH0;
	// a disc left stuck in a monster (fp_sscut arg1 1): read the monster again later
	int ssCutAt, ssCutH0;
	Actor ssCutMon;
	// a grenade set off by the probe: the Baron beside it, the grenade, when to read them
	int gnAt, gnBaronH0;
	Actor gnBaron, gnNade;
	Vector3 gnAtPos;
	int gnStickAt;
	int gnSkeetAt, gnWinAt;
	// a kind set off among imps: when, which kind, the imps, whether each lost its target at some point
	int gkAt, gkKind;
	int gnCookAt;
	Actor dgCorpse;
	// a thrown gun (Gun Toss, Boomerang Gun): the gun, the imp, when to read them
	Actor rgGun, rgImp;
	Actor krBit; double krD0; int krAt;
	int wpFly; Vector3 wpLast;
	int ropeT;
	RSQ_PunchHand ctSt; int ctT, ctThrowAt; Actor ctBall, ctImp; int ctH0;
	int wtTend; Actor wtShooter, wtNear, wtBall; int wtH0, wtN0; Actor wtItem1, wtItem2;
	int rgAt, rgH0;
	int dgHits, dgNext, dgDone;
	Array<Actor> gkImps;
	Array<int> gkLost, gkH0;
	override void WorldTick()
	{
		let p = P(); if (!p) return;
		int t = level.maptime;
		FxReport();
		WhipFlight();
		RopeLog(p);
		CatchRun(p);
		if (wtTend > 0)
		{
			// the whip weapon's hand kept tended and its trigger held (the mark), as weapon.zs does each tic
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			if (w) { w.tendTic[0] = level.maptime; w.markHeld[0] = true; }
			wtTend--;
		}
		if (gnAt > 0 && t == gnAt + 3)
		{
			int n = 0; double dmin = 1e9, dmax = 0;
			let it = ThinkerIterator.Create("RSQGN_Bomblet");
			Actor b;
			while (b = Actor(it.Next())) { n++; }
			Console.Printf("FP grenade blast: the Baron %d -> %d; bomblets thrown %d", gnBaronH0, gnBaron ? gnBaron.health : -1, n);
			gnBaronH0 = gnBaron ? gnBaron.health : 0;
		}
		if (gnAt > 0 && t == gnAt + 30)
		{
			int n = 0; double dmin = 1e9, dmax = 0;
			let it = ThinkerIterator.Create("RSQGN_Bomblet");
			Actor b;
			while (b = Actor(it.Next())) { n++; double d = (b.pos.xy - gnAtPos.xy).Length() / SSQ_Handler.CM(100.0); dmin = min(dmin, d); dmax = max(dmax, d); }
			if (n > 0) Console.Printf("FP grenade bomblets landed: %d, %.1f to %.1f m from the blast", n, dmin, dmax);
		}
		if (gnAt > 0 && t == gnAt + 45)
		{
			Console.Printf("FP grenade bomblets went off: the Baron %d -> %d", gnBaronH0, gnBaron ? gnBaron.health : -1);
			if (gnBaron) gnBaron.Destroy();
			gnAt = 0;
		}
		if (rgAt > 0 && (t == rgAt + 8 || t == rgAt + 40))
		{
			let rig = RSQTR_Rig.Get();
			Console.Printf("FP rig thrown gun t+%d: imp %d -> %d, frozen %d; the shotgun %s; carried %d; holster 4 holds '%s'", t - rgAt, rgH0, rgImp ? rgImp.health : -1,
				rgImp ? rgImp.freezetics : -1, rgGun ? (Inventory(rgGun).Owner ? "back with you" : "on the floor or flying") : "gone", p.CountInv("Shotgun"), rig ? rig.lastPut[4] : "?");
			if (t == rgAt + 40) { if (rgImp) rgImp.Destroy(); rgAt = 0; }
		}
		if (dgNext > 0 && t >= dgNext)
		{
			let hd = RSQDG_Hold.Get();
			if (!dgCorpse) { Console.Printf("FP grab shield: after hit %d the corpse FELL APART (hand full %d)", dgDone, hd ? hd.HandIsFull(0) : -1); dgNext = 0; dgDone = 0; }
			else if (dgDone >= dgHits)
			{
				if (hd && hd.HandIsFull(0)) hd.Drop(0);
				Console.Printf("FP grab shield: let go after %d hits -- health %d, buddha %d", dgDone, dgCorpse.health, dgCorpse.bBUDDHA);
				dgCorpse.Destroy(); dgNext = 0; dgDone = 0;
			}
			else
			{
				dgCorpse.DamageMobj(p, p, 30, 'None');
				dgDone++;
				if (dgCorpse) Console.Printf("FP grab shield: hit %d, health %d, %d left", dgDone, dgCorpse.health, hd ? hd.shieldLeft[0] : -1);
				dgNext = t + 1;
			}
		}
		if (gnCookAt > 0 && t == gnCookAt)
		{
			let h = RSQGN_Handler.Get();
			if (h) { h.armed[0] = false; h.spoon[0] = false; }
			gnCookAt = 0;
		}
		if (gkAt > 0)
		{
			for (int i = 0; i < gkImps.Size(); i++) if (gkImps[i] && !gkImps[i].target) gkLost[i] = 1;
			if (t == gkAt + 6 || t == gkAt + 40)
			{
				String s = "";
				for (int i = 0; i < gkImps.Size(); i++)
				{
					let m = gkImps[i];
					if (!m) { s = s .. " [gone]"; continue; }
					let dz = m.FindInventory("RSQ_Dazed");
					s = s .. String.Format(" [%d->%d frozen %d%s target %s%s%s lost %d]", gkH0[i], m.health, m.freezetics,
						m.Translation == Translation.GetID('Ice') ? " ICE" : "", m.target ? m.target.GetClassName() : 'none',
						m.bFriendly ? " FRIENDLY" : "", m.FindInventory("RSQGN_Infight") ? " INFIGHT" : "", gkLost[i]);
				}
				Console.Printf("FP grenade kind %d t+%d:%s", gkKind, t - gkAt, s);
			}
			if (t == gkAt + 40)
			{
				if (gkKind == 3 && gkImps.Size() > 0 && gkImps[0] && gkImps[0].health > 0)
				{
					int h0 = gkImps[0].health;
					gkImps[0].DamageMobj(p, p, 25, 'None');
					Console.Printf("FP grenade freeze: a 25 hit on a frozen imp: %d -> %d, frozen after %d", h0, gkImps[0] ? gkImps[0].health : -1, gkImps[0] ? gkImps[0].freezetics : -1);
				}
				for (int i = 0; i < gkImps.Size(); i++) if (gkImps[i]) gkImps[i].Destroy();
				gkImps.Clear(); gkAt = 0;
			}
		}
		if (gnSkeetAt > 0 && t == gnSkeetAt)
		{
			let g = RSQGN_Thrown(gnNade);
			Console.Printf("FP grenade skeet: %s, its blast x%.2f", !g || g.boomed ? "SHOT, IT BURST IN THE AIR" : "still flying", g ? g.dmgMult : -1.0);
			if (g && !g.boomed) g.Destroy();
			gnSkeetAt = 0;
		}
		if (gnWinAt > 0 && t == gnWinAt)
		{
			let g = RSQGN_Thrown(gnNade);
			Console.Printf("FP grenade skeet window: age %d, full bright %d", g ? g.age : -1, g ? g.bBright : -1);
			if (g) g.Destroy();
			gnWinAt = 0;
		}
		if (gnStickAt > 0 && t == gnStickAt)
		{
			let g = RSQGN_Thrown(gnNade);
			Console.Printf("FP grenade sticky: stuck to %s%s", g && g.stuckTo ? g.stuckTo.GetClassName() : 'nothing', g && g.stuckWall ? " (a wall)" : "");
			if (g && g.stuckTo) { g.stuckTo.SetOrigin(g.stuckTo.pos + (0, 48, 0), false); }
		}
		if (gnStickAt > 0 && t == gnStickAt + 3)
		{
			let g = RSQGN_Thrown(gnNade);
			if (g && g.stuckTo) Console.Printf("FP grenade sticky: the imp moved 48; the grenade %.1f units from it (it rides along)", (g.pos.xy - g.stuckTo.pos.xy).Length());
			if (g) { if (g.stuckTo) g.stuckTo.Destroy(); g.Destroy(); }
			gnStickAt = 0;
		}
		if (ssCutAt > 0 && t >= ssCutAt)
		{
			ssCutAt = 0;
			Console.Printf("FP shieldsaw embedded grind: %s %d -> %d over the embed", ssCutMon ? ssCutMon.GetClassName() : 'gone', ssCutH0, ssCutMon ? ssCutMon.health : -1);
		}
		if (ssDashAt > 0 && t >= ssDashAt)
		{
			ssDashAt = 0;
			let g = SSQ_Guard(p.FindInventory("SSQ_Guard"));
			Console.Printf("FP shieldsaw dash: moved %.2f m (%.0f units), the imp at the end %d -> %d, cooldown %d", (p.pos.xy - ssDashFrom.xy).Length() / SSQ_Handler.CM(100.0),
				(p.pos.xy - ssDashFrom.xy).Length(), ssDashH0, ssDashImp ? ssDashImp.health : -1, g ? g.dashCool : -1);
		}
		if (t == 3) { p.GiveInventory("RSQ_Glove", 1); p.GiveInventory("RSQ_GloveOff", 1); p.player.PendingWeapon = Weapon(p.FindInventory("RSQ_Glove")); }
		if (t == 30) { let o = Weapon(p.FindInventory("RSQ_GloveOff")); if (o && p.player.OffhandWeapon != o) p.player.PendingWeapon = o; }
		if (t == 60) Console.Printf("FP t60 ready %s off %s", p.player.ReadyWeapon ? p.player.ReadyWeapon.GetClassName() : 'none', p.player.OffhandWeapon ? p.player.OffhandWeapon.GetClassName() : 'none');
		let sim = RSQML_Sim.Get();
		if (sim)
		{
			if (guardMode > 0 && pushLeft > 0) pushFwd += 2.0;
			for (int h = 0; h < 2; h++)
			{
				sim.simOn[h] = guardMode > 0;
				if (guardMode <= 0) continue;
				Vector2 f = (cos(p.angle), sin(p.angle)), s = (-f.y, f.x);
				double side = h == 0 ? -4.0 : 4.0;
				Vector2 xy = p.pos.xy + f * pushFwd + s * side;
				sim.simHand[h] = (xy.x, xy.y, p.player.viewz - 2);
			}
			if (pushLeft > 0) { pushLeft--; if (pushLeft == 0) Console.Printf("FP t%d push done, hands %.0f forward", t, pushFwd); }
			if (offMode > 0)
			{
				if (offLeft > 0) { offF += offVF; offU += offVU; offLeft--; }
				Vector2 f2 = (cos(p.angle), sin(p.angle)), s2 = (-f2.y, f2.x);
				Vector2 xy2 = p.pos.xy + f2 * offF + s2 * offS;
				sim.simOn[1] = true;
				sim.simHand[1] = (xy2.x, xy2.y, p.player.viewz + offU);
			}
		}
		if (swatAt > 0 && t == swatAt) { swatAt = 0; HandAcross(0); }
		if (hadoAt > 0 && t == hadoAt) { hadoAt = 0; HandAcross(1); }
		if (seqKind.Size() > 0 && t >= seqNext)
		{
			int k = seqKind[0]; seqKind.Delete(0);
			if (a1 && a1.health > 0) Punch(seqHand, a1, k);
			seqHand = 1 - seqHand;
			seqNext = t + SEQ_GAP;
		}
		if (jugAt > 0 && t == jugAt)
		{
			jugAt = 0;
			if (a1 && a1.health > 0) { Console.Printf("FP juggle: off-hand jab, launch.age %d, z %.0f", Svc().GetInt("launch.age", "", 0, 0, a1), a1.pos.z - a1.floorz); Punch(1, a1, 0); }
		}
		if (mtAt < 0 && a1 && a2 && a1.pos.z - a1.floorz > 90) { a2.SetOrigin((a1.pos.x, a1.pos.y, a2.floorz), false); a2.vel = (0, 0, 0); }
		if (mtAt > 0 && t == mtAt)
		{
			mtAt = -1;
			if (a1 && a2) { a1.vel.xy = (0, 0); a2.SetOrigin((a1.pos.x, a1.pos.y, a2.floorz), false); a2.vel = (0, 0, 0); Console.Printf("FP meteor: second imp moved under the first (first at z %.0f)", a1.pos.z - a1.floorz); }
		}
	}

	Service Svc() { return ServiceIterator.Find("RSQML_Service").Next(); }

	// CATCH, through Melee's own CatchTick with a scripted hand: the palm held still 40 ahead at chest height, an imp's
	// ball arriving at it; every tic the hand is passed in still until ctThrowAt (0: never), then thrown at the imp at 4 m/s
	void CatchRun(PlayerPawn p)
	{
		if (ctT <= 0 || !ctSt) return;
		let gl = RSQML_Gloves.Get();
		let g = Glove(0);
		if (!gl || !g) { ctT = 0; return; }
		ctT++;
		ctSt.prevHand = ctSt.hand;
		if (ctThrowAt > 0 && ctT == ctThrowAt)
		{
			Vector3 to = ctImp ? (ctImp.pos + (0, 0, ctImp.height * 0.5) - ctSt.hand) : (cos(p.angle), sin(p.angle), 0);
			ctSt.vel = to.Unit() * 136;     // 4 m/s, map units a second
		}
		else ctSt.vel = (0, 0, 0);
		gl.CatchTick(p, g, 0, ctSt);
		Actor held = gl.caught[p.PlayerNumber() * 2];
		if (ctT % 10 == 0 || ctT == ctThrowAt + 1)
			Console.Printf("FP catch t%d: ball %s, %s; the imp %d -> %d", ctT, ctBall ? (ctBall.bMissile ? "flying" : "burst") : "gone",
				held ? String.Format("held at the palm (%.1f from it)", (held.pos + (0, 0, held.height * 0.5) - ctSt.hand).Length()) : "not held",
				ctH0, ctImp ? ctImp.health : -1);
		if (ctT > 160) ctT = 0;
	}

	// a hand across the ball's path: 0 an open-palm swat (Melee's Swat), 1 a fist (Melee's ReturnFire at level 2)
	void HandAcross(int how)
	{
		let p = P(); let g = Glove(0);
		if (!ball || !g) { Console.Printf("FP across: no ball or glove"); return; }
		Vector3 v = ball.vel; Vector2 s = (-v.y, v.x); double l = s.Length(); s = l > 0.01 ? s / l : (1, 0);
		let st = new("RSQ_PunchHand");
		st.prevHand = ball.pos + (s.x * 8, s.y * 8, 0);
		st.hand = ball.pos - (s.x * 8, s.y * 8, 0);
		st.vel = (-s.x * 120, -s.y * 120, 0);
		Console.Printf("FP t%d %s across %s at %.0f from you", level.maptime, how == 0 ? "swat" : "fist", ball.GetClassName(), ball.Distance3D(p));
		if (how == 0) RSQML_Gloves.Get().Swat(p, g, 0, st); else RSQML_Gloves.Get().ReturnFire(p, g, 0, st, 2);
		Console.Printf("FP after: %s target %s, heading %.0f deg off the imp", ball.GetClassName(), ball.target ? ball.target.GetClassName() : 'none', a1 ? Actor.absangle(VectorAngle(ball.vel.x, ball.vel.y), ball.AngleTo(a1)) : -1.0);
	}

	// kind 0 jab, 2 uppercut, through Melee's own Deliver
	void Punch(int h, Actor v, int kind)
	{
		let p = P(); let g = Glove(h);
		if (!g || !v) { Console.Printf("FP punch: no glove %d or no target", h); return; }
		Vector2 f = (cos(p.angle), sin(p.angle));
		Vector3 dir, path;
		if (kind == 2) { dir = (f.x * 0.3, f.y * 0.3, 0.95); path = (f.x * 0.05, f.y * 0.05, 0.3); }
		else if (kind == 1) { Vector2 s = (-f.y, f.x); dir = (s.x * 0.95 + f.x * 0.3, s.y * 0.95 + f.y * 0.3, 0); path = (s.x * 0.3, s.y * 0.3, 0); }
		else { dir = (f.x, f.y, 0); path = (f.x * 0.3, f.y * 0.3, 0); }
		Vector3 at = v.pos + (-f.x * v.radius, -f.y * v.radius, v.height * (punchHigh ? 0.92 : 0.7));
		RSQML_Gloves.Get().Deliver(p, g, h, v, 7.0, dir.Unit(), at, path);
		Console.Printf("FP punch %s kind asked %d -> landed kind %d, %s health %d", h ? "L" : "R", kind, g.lastKind, v.GetClassName(), v.health);
	}

	// the ShieldSaw and Grenade trees' events (a function of their own: NetworkProcess ran out of registers)
	// THE WHIP'S REACH FLIGHT, sampled every tic after fp_wp 5: how far the tip really moved, against the speed the sim
	// will read from it ((wp - wq) x SUB)
	void WhipFlight()
	{
		if (wpFly <= 0) return;
		let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
		if (!w) { wpFly = 0; return; }
		Vector3 tip = w.wp[RSQWP_Whip.NN];
		double step = (tip - wpLast).Length();
		double v = ((w.wp[RSQWP_Whip.NN] - w.wq[RSQWP_Whip.NN]) * RSQWP_Whip.SUB).Length();
		Console.Printf("FP whip flight t%d: reach tics left %d, the tip moved %.1f, the sim reads %.1f (room units a tic)", level.maptime, w.reachTic[0], step, v);
		wpLast = tip;
		if (++wpFly > 14) wpFly = 0;
	}

	// THE ROPE SWING after a snag (fp_wpsnag 2): where you are every 4 tics, and whether the rope still holds you
	void RopeLog(PlayerPawn p)
	{
		if (ropeT <= 0) return;
		if (ropeT % 4 == 1)
		{
			let c = RSQWP_Core.Get();
			let h = c ? c.hooks[0] : null;
			Console.Printf("FP rope t%d: you at (%.0f %.0f %.0f) moving (%.1f %.1f %.1f), rope %s", level.maptime, p.pos.x, p.pos.y, p.pos.z,
				p.vel.x, p.vel.y, p.vel.z, h ? String.Format("held, %.0f long", (h.pos - p.pos).Length()) : "gone");
		}
		if (++ropeT > 120) ropeT = 0;
	}

	// THE WHIP'S EVENTS (fp_wp*, fp_wt*): their own function, TreeEvents is at the register limit
	void WhipEvents(ConsoleEvent e, PlayerPawn p)
	{
		if (false) {}
		else if (e.Name ~== "fp_wp")
		{
			// THE WHIP'S FIXES (11-02). arg0 0: start the right-hand whip (arg1 mode, 1 Assist): its length and lasso reach;
			// 1: the crack's damage at five tip speeds; 2: the lasso's held piece for an off hand 0.3 / 0.6 / 1.2 m out;
			// 3: put arg1 links in the coil (then 4 reads how long it has been out enough to crack); 5: Assist's throw
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			if (!w) { Console.Printf("FP whip: no RSQWP_Whip"); return; }
			int a = e.Args[0];
			if (a == 0)
			{
				w.Start(0, p, 0, e.Args[1]);
				Console.Printf("FP whip started: mode %d, length %.1f room units (%.2f m), lasso reach %.0f (%.1f m), coil %.1f -> %.1f",
					w.wmode[0], w.wlen[0], w.wlen[0] / 34.0, w.LassoReach(0), w.LassoReach(0) / 34.0, w.cp[0], w.cgoal[0]);
			}
			else if (a == 1)
			{
				static const double SP[] = { 20, 30, 35, 43, 50 };
				String s = "";
				double full = RSQWP_Util.Opt("rsqwp_whip_damage", 12);
				for (int k = 0; k < 5; k++)
				{
					double hk = w.CrackK(0, SP[k]);
					s = s .. String.Format(" %.0f->%d", SP[k], max(1, int(full * (0.25 + 0.75 * hk) + 0.5)));
				}
				Console.Printf("FP whip crack damage by tip speed (full %.0f, fullcrack %.0f):%s", full, RSQWP_Util.Num("rsqwp_whip_hardcrack", 0), s);
			}
			else if (a == 2)
			{
				Vector3 b, d;
				[b, d] = w.Handle(0, p, 0);
				Vector3 tip = b + d * RSQWP_Whip.HANDLEN;
				static const double M[] = { 0.3, 0.6, 1.2 };
				String s = "";
				for (int k = 0; k < 3; k++)
				{
					w.lassoP1[0] = tip + (0, 0, -1) * M[k] * 34.0;
					s = s .. String.Format(" %.1f m->%d links", M[k], w.LassoHeld(0, b, d));
				}
				Console.Printf("FP whip lasso held piece:%s", s);
			}
			else if (a == 3) { w.cp[0] = e.Args[1]; w.cgoal[0] = e.Args[1]; }
			else if (a == 4) Console.Printf("FP whip: %.1f links coiled, out enough to crack for %d tics", w.cp[0], w.fullTics[0]);
			else if (a == 5)
			{
				w.Start(0, p, 0, 1);
				w.Throw(0, p, 0);
				wpLast = w.wp[RSQWP_Whip.NN];
				wpFly = 1;
				Console.Printf("FP whip: Assist's throw, %d tics of flight", w.reachTic[0]);
			}
		}
		else if (e.Name ~== "fp_wpsnag")
		{
			// SNAG on WHIPTEST (Whip/tools/rigtest/whipprobe/maps): arg0 1: stand at the west ledge's lip facing the
			// beam, start the whip all out and crack it just under the beam's west edge (z 72); 2: step off into the pit
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			if (!w) return;
			if (e.Args[0] == 1)
			{
				p.SetOrigin((496, 512, 0), false);
				p.angle = 0;
				w.Start(0, p, 0, 0);
				w.cp[0] = 0; w.cgoal[0] = 0;
				w.Crack(0, p, 0, (534, 512, 70), (1, 0, 0) * 30);
				let c = RSQWP_Core.Get();
				let h = c ? c.hooks[0] : null;
				Console.Printf("FP whip snag: a crack under the beam's edge -> %s (whip state %d)", h ? String.Format("ROPE at (%.0f %.0f %.0f)", h.pos.x, h.pos.y, h.pos.z) : "no rope", w.wstate[0]);
			}
			else if (e.Args[0] == 2) { p.SetOrigin((530, 512, 0), true); p.vel = (4, 0, 0); ropeT = 1; }   // just past the lip, over the pit (a still player on the ground is stopped dead)
		}
		else if (e.Name ~== "fp_wt")
		{
			// THE WHIP TREE. arg0 0: give the whip weapon and put it in hand; 1: offers, levels and every read; 2: grant arg1's
			// node (+arg2 levels)
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			if (!w) return;
			if (e.Args[0] == 0) { p.GiveInventory("RSQWP_WhipWeapon", 1); p.player.PendingWeapon = Weapon(p.FindInventory("RSQWP_WhipWeapon")); return; }
			let wpn = Weapon(p.FindInventory("RSQWP_WhipWeapon"));
			let st = TFLV_PerPlayerStats.GetStatsFor(p);
			let info = (wpn && st) ? st.GetInfoFor(wpn) : null;
			if (!info) { Console.Printf("FP whip tree: no record (carried: %s)", wpn ? "yes" : "no"); return; }
			if (e.Args[0] == 2)
			{
				String cls = "RSQ_Upgrade_" .. RSQWP_Whip.GBN[clamp(e.Args[1], 0, RSQWP_Whip.G_N - 1)];
				info.upgrades.Add(cls, max(e.Args[2], 1));
				Console.Printf("FP whip tree granted %s", cls);
				return;
			}
			String on = "", have = "";
			for (int k = 0; k < RSQWP_Whip.G_N; k++)
			{
				Class<TFLV_Upgrade_BaseUpgrade> c = "RSQ_Upgrade_" .. RSQWP_Whip.GBN[k];
				if (!c) { on = on .. " ?" .. RSQWP_Whip.GBN[k]; continue; }
				if (TFLV_Upgrade_BaseUpgrade(new(c)).IsSuitableForWeapon(info)) on = on .. " " .. RSQWP_Whip.GBN[k];
				int lv = int(info.upgrades.Level("RSQ_Upgrade_" .. RSQWP_Whip.GBN[k]));
				if (lv > 0) have = have .. String.Format(" %s%d", RSQWP_Whip.GBN[k], lv);
			}
			Console.Printf("FP whip tree [stage %d] has:%s | offered:%s", e.Args[1], have, on);
			String rd = "";
			for (int k = 0; k < RSQWP_Whip.G_N; k++) rd = rd .. String.Format(" %d", RSQWP_Whip.Gb(p, k));
			w.Start(0, p, 0, 0);
			Console.Printf("FP whip tree reads:%s | length %.2f m, lasso reach %.1f m, light mass %.0f, snag above %.0f, swing cap %.0f; lasso %d, snag %d, mantle %d, keep speed %d, crack shot %d",
				rd, w.wlen[0] / 34.0, w.LassoReach(0) / 34.0, RSQWP_Whip.LightMass(p), RSQWP_Whip.SnagAbove(p), RSQWP_Whip.SwingCap(p),
				RSQWP_Whip.Unlock(p, RSQWP_Whip.G_LASSO, "rsqwp_whip_lasso", true), RSQWP_Whip.Unlock(p, RSQWP_Whip.G_SNAG, "rsqwp_whip_snag", true),
				RSQWP_Whip.Unlock(p, RSQWP_Whip.G_MANTLE, "rsqwp_mantle", true), RSQWP_Whip.Unlock(p, RSQWP_Whip.G_MOMENTUM, "rsqwp_carry", true),
				RSQWP_Whip.Unlock(p, RSQWP_Whip.G_CRACKSHOT, "rsqwp_whip_fireball", false));
			w.Halt(0);
		}
		else if (e.Name ~== "fp_wtcrack")
		{
			// a crack at tip speed 43 on a 1000-health imp 60 ahead, by the whip's own Hit
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			let m = Put("DoomImp", 60);
			if (!w || !m) return;
			m.health = 1000;
			w.Start(0, p, 0, 0);
			w.Hit(0, p, 0, m, m.pos + (0, 0, 30), 43, true);
			let d = RSQ_Dazed(m.FindInventory("RSQ_Dazed"));
			Console.Printf("FP whip crack at 43: %d damage, frozen %d tics, held in pain %d tics", 1000 - m.health, m.freezetics, d ? d.staggerUntil - level.maptime : 0);
			w.Halt(0);
			m.Destroy();
		}
		else if (e.Name ~== "fp_wtmissile")
		{
			// an imp 400 ahead shoots; its ball, 60 ahead and coming at you, is cracked (the whip's own CrackMissile); a
			// second 1000-health imp stands 40 beside the ball. fp_wtmissiled reports a few tics later
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			if (!w) return;
			Vector3 f = (cos(p.angle), sin(p.angle), 0), s = (-sin(p.angle), cos(p.angle), 0);
			wtShooter = Put("DoomImp", 400); wtNear = Put("DoomImp", 60, 40);
			if (!wtShooter || !wtNear) return;
			wtShooter.health = 1000; wtShooter.freezetics = 40; wtNear.health = 1000; wtNear.freezetics = 40;
			wtH0 = 1000; wtN0 = 1000;
			wtBall = Actor.Spawn("DoomImpBall", p.pos + f * 60 + (0, 0, 32));
			if (!wtBall) return;
			wtBall.target = wtShooter;
			wtBall.vel = -f * wtBall.Speed;
			bool sent = w.CrackMissile(p, wtBall, s * 10);
			Console.Printf("FP whip cracked the ball: %s", sent ? String.Format("sent away, flying %.0f deg off the shooter's bearing", wtBall ? abs(Actor.DeltaAngle(VectorAngle(wtBall.vel.x, wtBall.vel.y), p.angle)) : -1.0) : "blown up");
		}
		else if (e.Name ~== "fp_wtmissiled")
		{
			Console.Printf("FP whip after the crack: the shooter %d -> %d, the imp beside the ball %d -> %d", wtH0, wtShooter ? wtShooter.health : -1, wtN0, wtNear ? wtNear.health : -1);
			if (wtShooter) wtShooter.Destroy();
			if (wtNear) wtNear.Destroy();
		}
		else if (e.Name ~== "fp_wtyank")
		{
			// a wrapped monster (arg0 0 imp, 1 arachnotron 600, 2 mancubus 1000; arg1 1: lassoed) 150 ahead, yanked as hard as
			// a pull goes; with arg0 0 and arg1 1, two more imps stand by it for Bola
			static const String YC[] = { "DoomImp", "Arachnotron", "Fatso" };
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			let m = Put(YC[clamp(e.Args[0], 0, 2)], 150);
			if (!w || !m) return;
			Actor b1 = null, b2 = null;
			if (e.Args[0] == 0 && e.Args[1] == 1) { b1 = Put("DoomImp", 150, 40); b2 = Put("DoomImp", 150, -40); }
			w.Start(0, p, 0, 0);
			w.BeginWrap(0, p, 0, m, e.Args[1] == 1);
			int n = (w.tied[0] ? 1 : 0) + (w.tied[1] ? 1 : 0);
			int held = w.wrapLen[0];
			w.Yank(0, p, 0, m, RSQWP_Whip.PULL_FULL);
			let d = RSQ_Dazed(m.FindInventory("RSQ_Dazed"));
			Console.Printf("FP whip yank %s (mass %d, lassoed %d): held %d tics, %d tied with it; flies at %.1f a tic toward you%s, staggered %d tics",
				m.GetClassName(), m.Mass, e.Args[1], held, n, m.vel.xy.Length(), b1 ? String.Format(" (the tied ones %.1f, %.1f)", b1.vel.xy.Length(), b2 ? b2.vel.xy.Length() : -1.0) : "",
				d ? max(0, d.staggerUntil - level.maptime) : 0);
			w.Halt(0);
			m.Destroy(); if (b1) b1.Destroy(); if (b2) b2.Destroy();
		}
		else if (e.Name ~== "fp_wttarzan")
		{
			// swinging (no rope needed for the test) at arg0 map units a tic into a 1000-health imp just ahead
			let c = RSQWP_Core.Get();
			let m = Put("DoomImp", 40);
			if (!c || !m) return;
			m.health = 1000;
			p.vel = (cos(p.angle), sin(p.angle), 0) * e.Args[0];
			c.Tarzan(p);
			let d = RSQ_Dazed(m.FindInventory("RSQ_Dazed"));
			Console.Printf("FP whip tarzan at %d a tic (cap %.0f): %d damage, knocked down %d", e.Args[0], RSQWP_Whip.SwingCap(p), 1000 - m.health, d ? d.downUntil > level.maptime : 0);
			p.vel = (0, 0, 0);
			m.Destroy();
		}
		else if (e.Name ~== "fp_wtqm")
		{
			// QUICK MARK (the whip weapon in hand, +attack held by the run): arg0 1 a medikit 100 ahead and a stimpack 150 ahead; 2 crack (it flies to
			// the mark); 3 report where both went
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			if (!w) return;
			if (e.Args[0] == 1)
			{
				wtItem1 = Put("Medikit", 100); wtItem2 = Put("Stimpack", 150);
				p.health = p.player.health = 10;     // the whip weapon in hand marks while +attack is held (the run holds it)
			}
			else if (e.Args[0] == 2)
			{
				Console.Printf("FP whip marks: %s and %s", w.lockOn[0] ? String.Format("%s", w.lockOn[0].GetClassName()) : "nothing", w.lock2[0] ? String.Format("%s", w.lock2[0].GetClassName()) : "nothing");
				w.Crack(0, p, 0, p.pos + (0, 0, 40), (cos(p.angle), sin(p.angle), 0) * 30);
			}
			else
			{
				Console.Printf("FP whip after one crack: health %d; the medikit %s, the stimpack %s", p.health,
					(!wtItem1 || Inventory(wtItem1).Owner) ? "taken" : String.Format("%.0f away", wtItem1.Distance3D(p)),
					(!wtItem2 || Inventory(wtItem2).Owner) ? "taken" : String.Format("%.0f away", wtItem2.Distance3D(p)));
				p.health = p.player.health = 5000;
			}
		}
		else if (e.Name ~== "fp_wtsnag")
		{
			// SNAG 2 on WHIPTEST: you at the lip, the hand raised arg0 units, a crack just under the beam's edge (z 72)
			let w = RSQWP_Whip(EventHandler.Find("RSQWP_Whip"));
			let c = RSQWP_Core.Get();
			if (!w || !c) return;
			if (c.hooks[0]) c.Drop(0);
			p.SetOrigin((496, 512, 0), false);
			p.angle = 0;
			w.Start(0, p, 0, 0);
			w.cp[0] = 0; w.cgoal[0] = 0;
			p.player.viewz = p.pos.z + p.player.viewheight + e.Args[0];   // the hand (body fallback: the eyes - 8) raised arg0
			Vector3 hp0 = RSQWP_Util.HandPos(p, 0);
			double hz = hp0.z;
			w.Crack(0, p, 0, (534, 512, 70), (1, 0, 0) * 30);
			Console.Printf("FP whip snag with the hand at z %.0f (the edge %.0f above it, needs over %.0f): %s", hz, 70 - hz, RSQWP_Whip.SnagAbove(p), c.hooks[0] ? "ROPE" : "no rope");
			if (c.hooks[0]) c.Drop(0);
			w.Halt(0);
		}
	}

	void TreeEvents(ConsoleEvent e, PlayerPawn p)
	{
		if (e.Name ~== "fp_ss")
		{
			// THE SHIELDSAW TREE. arg0 1: print its offers and levels (arg1 = stage); 2: grant arg1's node (+arg2 levels)
			static const String SSN[] = { "Lockstep", "Ricochet", "HuntersReturn", "MeatGrinder", "WallBounces", "Cleaver",
				"Broadening", "Bulwark", "Rebound", "BackPlate", "ShieldWall", "Reflect",
				"BatteringRam", "WallSlam", "BashHitStop", "Charge", "ShieldSize", "ShieldDash",
				"HungryBlades", "RevShield", "BuzzsawReturn", "Spartan" };
			let saw = Weapon(p.FindInventory("SSQ_ShieldSaw"));
			let st = TFLV_PerPlayerStats.GetStatsFor(p);
			let info = (saw && st) ? st.GetInfoFor(saw) : null;
			if (!info) { Console.Printf("FP shieldsaw: no record (carried: %s)", saw ? "yes" : "no"); return; }
			if (e.Args[0] == 2) { String cls = "RSQ_Upgrade_" .. SSN[clamp(e.Args[1], 0, SSN.Size() - 1)]; info.upgrades.Add(cls, max(e.Args[2], 1)); Console.Printf("FP shieldsaw granted %s", cls); return; }
			String on = "", have = "";
			for (int i = 0; i < SSN.Size(); i++)
			{
				Class<TFLV_Upgrade_BaseUpgrade> c = "RSQ_Upgrade_" .. SSN[i];
				if (!c) { on = on .. " ?" .. SSN[i]; continue; }
				if (TFLV_Upgrade_BaseUpgrade(new(c)).IsSuitableForWeapon(info)) on = on .. " " .. SSN[i];
				int lv = int(info.upgrades.Level("RSQ_Upgrade_" .. SSN[i]));
				if (lv > 0) have = have .. String.Format(" %s%d", SSN[i], lv);
			}
			Console.Printf("FP shieldsaw [stage %d] has:%s | offered:%s", e.Args[1], have, on);
		}
		else if (e.Name ~== "fp_ssthrow")
		{
			// how many locks a throw takes, and the walls it may bounce off: through Begin (the hand throw's own step) and
			// through LaunchAlong (the One-Handed throw)
			let saw = Weapon(p.FindInventory("SSQ_ShieldSaw"));
			Vector3 dir = (cos(p.angle), sin(p.angle), 0);
			let d1 = SSQ_Disc(Actor.Spawn("SSQ_Disc", p.pos + (0, 0, 40) + dir * 24));
			let d2 = SSQ_Disc(Actor.Spawn("SSQ_Disc", p.pos + (0, 0, 40) - dir * 24));
			if (!d1 || !d2) return;
			d1.Claim(saw); d1.Begin(p, dir, 12, 0, dir);
			d2.Claim(saw); d2.LaunchAlong(p, dir, 12, 90, 0);
			Console.Printf("FP shieldsaw throw: locks %d, wall bounces %d (hand throw) / %d (one-handed)", SSQ_Disc.WantLocks(p), d1.bounces, d2.bounces);
			d1.Destroy(); d2.Destroy();
		}
		else if (e.Name ~== "fp_ssguard")
		{
			// the shield where it is now (on the back, headless): arg0 0 a fireball onto the plate along its own normal,
			// through the guard's own AbsorbDamage; 1 an imp's 10-damage bullet at it, through the guard's own StopsHitscan
			let h = SSQ_Handler.Get();
			let g = SSQ_Guard(p.FindInventory("SSQ_Guard"));
			if (!g || !h) { Console.Printf("FP shieldsaw guard: none"); return; }
			bool has = g.Plate();
			Console.Printf("FP shieldsaw guard: state %d, plate %s (held %d), back plate %d, turns %d, shield wall %d ms", h.StateOf(0),
				has ? "yes" : "no", g.plateHeld, SSQ_Guard.BackPlate(p), has && g.PlateTurns(), SSQ_Guard.ShieldWallMs(p));
			if (!has) return;
			Vector3 C = g.plateC, n = g.plateN;
			Vector3 mid = p.pos + (0, 0, p.height * 0.5);
			Vector3 away = ((C - mid) dot n) >= 0 ? n : -n;
			let imp = Actor.Spawn("DoomImp", C + away * (e.Args[1] > 0 ? e.Args[1] : 160) - (0, 0, 28));
			if (!imp) return;
			imp.freezetics = 70; imp.bNoGravity = true;
			if (e.Args[0] == 0)
			{
				let ball = Actor.Spawn("DoomImpBall", C + away * 20);
				if (!ball) return;
				ball.target = imp; ball.vel = -away * 10;
				int before = 0, after = 0;
				let it = ThinkerIterator.Create("DoomImpBall");
				Actor a;
				while (a = Actor(it.Next())) if (a.target == p) before++;
				int nd = 20;
				g.AbsorbDamage(20, 'None', nd, ball, imp, 0);
				it = ThinkerIterator.Create("DoomImpBall");
				while (a = Actor(it.Next())) if (a.target == p) after++;
				Console.Printf("FP shieldsaw guard: fireball onto the plate: 20 damage -> %d, turned back: %d", nd, after - before);
				ball.Destroy();
			}
			else
			{
				Vector3 A = imp.pos + (0, 0, imp.height * 0.5);
				Vector3 d = C - A;
				double dist = d.Length();
				int h0 = imp.health;
				bool stopped = g.StopsHitscan(imp, A, d / dist, dist + 200, 10);
				Console.Printf("FP shieldsaw guard: a 10-damage bullet at the plate: %s, the imp %d -> %d (imp %.0f from the plate, sees the player %d, in the level %d, player angle %.0f, out (%.2f %.2f %.2f))", stopped ? "STOPPED" : "not stopped", h0, imp.health,
					dist, imp.CheckSight(p), level.IsPointInLevel(imp.pos), p.angle, away.x, away.y, away.z);
			}
			imp.Destroy();
		}
		else if (e.Name ~== "fp_ssbash")
		{
			// SSQ_Guard.Bash on a frozen-free monster in front: arg0 monster (fp_sscut's list), arg1 how hard (units/sec)
			static const String BM[] = { "LostSoul", "Demon", "BaronOfHell", "DoomImp", "HellKnight", "Revenant" };
			let saw = SSQ_ShieldSaw(p.FindInventory("SSQ_ShieldSaw"));
			let m = Put(BM[clamp(e.Args[0], 0, BM.Size() - 1)], 64);
			if (!m || !saw) return;
			Vector3 f = (cos(p.angle), sin(p.angle), 0);
			int h0 = m.health;
			let hh = SSQ_Handler.Get();
			if (hh) hh.mBraced[0] = e.Args[2] == 1;	// arg2 1: the blade resting on the shield (Spartan 2)
			SSQ_Guard.Bash(p, saw, m, p.pos + f * 30 + (0, 0, 40), f, e.Args[1]);
			if (hh) hh.mBraced[0] = false;
			Console.Printf("FP shieldsaw bash %s at %d u/s: %d -> %d, frozen %d tics (hit-stop), thrown %.1f u/tic", m.GetClassName(), e.Args[1], h0, m.health, m.freezetics, m.vel.xy.Length());
			m.Destroy();
		}
		else if (e.Name ~== "fp_ssslam")
		{
			// SSQ_Handler.Slam: an imp hits a wall at arg0 units a tic
			let h = SSQ_Handler.Get();
			let m = Put("DoomImp", 96);
			if (!m || !h) return;
			int h0 = m.health;
			h.Slam(m, 0, (-cos(p.angle), -sin(p.angle)), e.Args[0]);
			Console.Printf("FP shieldsaw wall slam at %d u/tic: %d -> %d, stunned %d tics", e.Args[0], h0, m.health, m.freezetics);
			m.Destroy();
		}
		else if (e.Name ~== "fp_sscharge")
		{
			// SSQ_Guard.Charge with the stick at arg0 tenths of m/s forward: imps at 0.6, 1.2 (0.3 aside), 1.8 and 2.4 m
			let saw = SSQ_ShieldSaw(p.FindInventory("SSQ_ShieldSaw"));
			if (!saw) return;
			double m = SSQ_Handler.CM(100.0);
			Vector3 f = (cos(p.angle), sin(p.angle), 0);
			Vector3 C = p.pos + f * 20 + (0, 0, 32);
			Actor i1 = Put("DoomImp", 20 + 0.6 * m), i2 = Put("DoomImp", 20 + 1.2 * m, 0.3 * m), i3 = Put("DoomImp", 20 + 1.8 * m), i4 = Put("DoomImp", 20 + 2.4 * m);
			p.vel.xy = f.xy * (e.Args[0] / 10.0) * m / TICRATE;

				if (!info) { Console.Printf("FP tree %d: no record", h); continue; }

				for (int i = 0; i < NODES.Size(); i++)
