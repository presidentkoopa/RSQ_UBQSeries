// Gun Bonsai VR / Offhand Patch -- added cvars.
// These are additions only; every stock bonsai_* cvar is untouched and still
// lives in Gun Bonsai's own CVARINFO.

// Draw a HUD for each hand instead of a single one.
// Mainhand goes top-right, offhand top-left (reflected for a left-hander:
// QuestZDoom's Control Scheme).
user bool bonsai_vr_dual_hud = true;

// When dual HUD is OFF, which hand the single HUD follows.
// 0 = mainhand, 1 = offhand, 2 = whichever earned XP most recently.
// Always falls through to the hand that actually holds a gun, so this can
// never blank the HUD on its own.
user int bonsai_vr_hud_hand = 0;

// Per-hand HUD anchors, used only when bonsai_vr_dual_hud is on.
// X/Y follow the same convention as bonsai_hud_x/y: 0..1 is a fractional
// anchor, negative is absolute pixels from the right/bottom edge, and >1 is an
// absolute pixel position.
//
// Mirror is a bitmask -- 1 = horizontal, 2 = vertical -- and also selects which
// corner-specific frame texture is drawn. The defaults make the two HUDs exact
// horizontal reflections of each other, so they read as a matched pair rather
// than as two copies of the same widget.
user float bonsai_vr_hud_main_x = 0.99;
user float bonsai_vr_hud_main_y = 0.02;
user int bonsai_vr_hud_main_mirror = 3;

user float bonsai_vr_hud_off_x = 0.01;
user float bonsai_vr_hud_off_y = 0.02;
user int bonsai_vr_hud_off_mirror = 2;

// Colour Cycle, per hand (owner 11-06, from his fork's HUD: Gun Bonsai's own level-up cycle, ca.ancilla.bonsai/HUD.zsc
// GetShinyColour / CycleOr). On: that hand's gauge frame, weapon XP bar and player XP bar cycle through the colours all
// the time, 34 tics apart so the colour ripples across. Off: Gun Bonsai's colours, as before. A waiting level-up cycles
// whatever the switch. Speed scales the rate (0 = still), Spread the gap between frame and bars, Intensity the
// brightness; 1.0 / 1.0 / 1.0 is Gun Bonsai's own cycle. The main set also colours the single HUD.
user bool  bonsai_vr_hud_main_cycle = true;
user float bonsai_vr_hud_main_cyclespeed = 1.0;
user float bonsai_vr_hud_main_gradient = 1.0;
user float bonsai_vr_hud_main_intensity = 1.0;
user bool  bonsai_vr_hud_off_cycle = true;
user float bonsai_vr_hud_off_cyclespeed = 1.0;
user float bonsai_vr_hud_off_gradient = 1.0;
user float bonsai_vr_hud_off_intensity = 1.0;

// The LegenDoom HUD, direction D (owner 11-11 15:30; rsq/ui/legendoom.zs). Each row off brings LegenDoom's own back:
// Pickup Card off: LegenDoom's pickup screen shows as LegenDoom draws it, no card. Gauge Row off: no row under the gauge.
user bool rsqgb_ld_card = true;
user bool rsqgb_ld_row = true;

// (rsq_hud_shield, the ShieldSaw's third gauge, is gone since 11-11: the owner removed it. A value left in an old ini
// is ignored.)

// (bonsai_vr_warn_binding, the startup warning about Gun Bonsai's upgrade binding mode, is gone since 0.99a: measured
// on the rig in modes 0, 1 and 2, with two different weapons and with two weapons merged into one class, each hand's
// XP went to its own record and no record piled up -- WeaponInfo.IsLiveInAnotherHand stops the thrash it warned of.)

// (2026-11-25: Unlock RSQ Features, the Force, Star Punch, the hook XP, Style XP, the stock pruning and the Berserk gauge are

// GBOVR Menus and HUD (owner 11-05: "the only hud we want is the Mockup ... GBOVR Menus and HUD : On / Off"):
// 1 On = the recreation of the owner's mockups (the status screen, the level-up cards, the bottom row, the XP bars'
// breathing), 0 Off = Gun Bonsai's own screens and gauges. The name is the old Screen Skin's, so a saved value carries:
// settings version 1 (rsq_ui_cfgver) turns every saved look (Gun Bonsai 0, Hex Rail 2, Terminal 3) into On.
server int rsq_ui_skin = 1;
server int rsq_ui_cfgver = 0;
// The screens: HUD inset (fraction of each side kept clear, for the headset's edge of view),
// HUD size, and the bottom row (per-hand ammo, health, face, armour).
server float rsq_ui_inset = 0.04;
server float rsq_ui_hud_scale = 1.5;   // 1.5 from 11-03: 1.0 read too small in the headset
server bool rsq_ui_bottom = true;
// What the skins changed in other mods' user cvars, to put back on skin 0 (-1 = nothing saved).
server int rsq_ui_saved_scorehide = -1;
server int rsq_ui_saved_sbartype = -1;
