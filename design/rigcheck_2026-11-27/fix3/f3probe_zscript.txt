version "4.14"
// Agent A test probe (rig checks 2026-11-27). Never ships.
class AProbe : EventHandler
{
	int every; String watch; int watchEvery; bool logInput; bool logDmg;
	override void OnRegister() { logInput = true; logDmg = true; }
	override bool InputProcess(InputEvent e)
	{
		if (logInput && (e.Type == InputEvent.Type_KeyDown || e.Type == InputEvent.Type_KeyUp))
			Console.Printf("AP KEY %s %d", e.Type == InputEvent.Type_KeyDown ? "down" : "up", e.KeyScan);
		return false;
	}
	ui int rfCount;
	ui String lastMenu;
	override void UiTick()
	{
		let m = Menu.GetCurrentMenu();
		String n = "none";
		if (m)
		{
			n = m.GetClassName();
			let om = OptionMenu(m);
			if (om && om.mDesc) n = n .. String.Format(" %s sel %d", om.mDesc.mMenuName, om.mDesc.mSelectedItem);
			let lm = ListMenu(m);
			if (lm && lm.mDesc) n = n .. String.Format(" %s sel %d", lm.mDesc.mMenuName, lm.mDesc.mSelectedItem);
		}
		if (n != lastMenu) { Console.Printf("AP MENU %s gametic %d gamestate %d", n, gametic, gamestate); lastMenu = n; }
	}
	override void RenderOverlay(RenderEvent e)
	{
		rfCount++;
		Console.Printf("AP RF %d t%d g%d", rfCount, level.maptime, gametic);
		Screen.DrawText(SmallFont, Font.CR_GREEN, 4, 4, String.Format("t%d", level.maptime), DTA_VirtualWidth, 160, DTA_VirtualHeight, 120);
	}
	PlayerPawn P() { return players[consoleplayer].mo; }
	void Hands(String why)
	{
		let pmo = P(); if (!pmo) return;
		let pl = pmo.player;
		Console.Printf("AP HANDS %s t%d ready %s off %s pending %s hp %d", why, level.maptime,
			pl.ReadyWeapon ? pl.ReadyWeapon.GetClassName() : 'none', pl.OffhandWeapon ? pl.OffhandWeapon.GetClassName() : 'none',
			pl.PendingWeapon && pl.PendingWeapon != WP_NOCHANGE ? pl.PendingWeapon.GetClassName() : 'none', pmo.health);
	}
	void Pose(String why)
	{
		let pmo = P(); if (!pmo) return;
		Console.Printf("AP POSE %s t%d main %.1f %.1f %.1f ang %.1f pit %.1f  off %.1f %.1f %.1f ang %.1f pit %.1f  player %.1f %.1f %.1f yaw %.1f pitch %.1f vz %.1f",
			why, level.maptime, pmo.AttackPos.x, pmo.AttackPos.y, pmo.AttackPos.z, pmo.AttackAngle, pmo.AttackPitch,
			pmo.OffhandPos.x, pmo.OffhandPos.y, pmo.OffhandPos.z, pmo.OffhandAngle, pmo.OffhandPitch,
			pmo.pos.x, pmo.pos.y, pmo.pos.z, pmo.angle, pmo.pitch, pmo.player.viewz);
	}

	Service Svc(String n) { let it = ServiceIterator.Find(n); return it ? it.Next() : null; }
	void Rig(String why)
	{
		let pmo = P(); if (!pmo) return;
		let s = Svc("RSQTR_RigService"); let b = Svc("RSQTR_BodyService");
		if (!s) { Console.Printf("AP RIG no service"); return; }
		String o = "";
		for (int r = 0; r < 14; r++)
		{
			if (r == 2) continue;
			String ow = s.GetString("rig.owner", "", r, 0, pmo);
			double x = s.GetDouble("region.x", "", r, 0, pmo), y = s.GetDouble("region.y", "", r, 0, pmo), z = s.GetDouble("region.z", "", r, 0, pmo);
			Vector3 a = pmo.AttackPos - (x, y, z), b = pmo.OffhandPos - (x, y, z);
			double dm = (a.x, a.y, a.z * 1.2).Length() / 34.0, doff = (b.x, b.y, b.z * 1.2).Length() / 34.0;
			Vector2 rel = Actor.RotateVector((x - pmo.pos.x, y - pmo.pos.y), -pmo.angle) / 34.0;
			o = o .. String.Format(" [%d %s own=%s m%.2f o%.2f st %.3f %.3f %.3f]", r, s.GetString("rig.regionname", "", r), ow, dm, doff, -rel.y, (z - pmo.pos.z) * 1.2 / 34.0, -rel.x);
		}
		String rw = pmo.player.ReadyWeapon ? pmo.player.ReadyWeapon.GetClassName() : 'none';
		Console.Printf("AP RIG %s t%d hand0 in %d hand1 in %d body.yaw %.1f ready %s holster %d nade.at %d,%d pouch.at %d,%d%s", why, level.maptime,
			s.GetInt("rig.region", "", 0, 0, pmo), s.GetInt("rig.region", "", 1, 0, pmo), b ? b.GetDouble("body.yaw", "", 0, 0, pmo) : -1,
			rw, s.GetInt("rig.holster", rw, 0, 0, pmo), s.GetInt("nade.at", "", 0, 0, pmo), s.GetInt("nade.at", "", 1, 0, pmo), s.GetInt("pouch.at", "", 0, 0, pmo), s.GetInt("pouch.at", "", 1, 0, pmo), o);
	}
	void SvcAsk(Array<String> w, ConsoleEvent e)
	{
		// a_svc:Service:kind:request:stringArg  (kind i/s/d) intArg = Args[0]
		if (w.Size() < 4) return;
		let s = Svc(w[1]); if (!s) { Console.Printf("AP SVC %s missing", w[1]); return; }
		String sa = w.Size() > 4 ? w[4] : "";
		if (w[2] == "i") Console.Printf("AP SVC t%d %s %s(%s,%d) = %d", level.maptime, w[1], w[3], sa, e.Args[0], s.GetInt(w[3], sa, e.Args[0], 0, P()));
		else if (w[2] == "d") Console.Printf("AP SVC t%d %s %s(%s,%d) = %.3f", level.maptime, w[1], w[3], sa, e.Args[0], s.GetDouble(w[3], sa, e.Args[0], 0, P()));
		else Console.Printf("AP SVC t%d %s %s(%s,%d) = %s", level.maptime, w[1], w[3], sa, e.Args[0], s.GetString(w[3], sa, e.Args[0], 0, P()));
	}
	void Inv(String why)
	{
		let pmo = P(); if (!pmo) return;
		String s = "";
		for (let i = pmo.Inv; i; i = i.Inv) s = s .. String.Format(" %s=%d", i.GetClassName(), i.Amount);
		Console.Printf("AP INV %s t%d%s", why, level.maptime, s);
	}
	override void WorldTick()
	{
		if (every > 0 && level.maptime % every == 0) { Hands("tick"); Pose("tick"); }
		if (level.maptime < 50) { let pmo = P(); let b = Svc("RSQTR_BodyService"); if (pmo) Console.Printf("AP EARLY t%d g%d yaw %.1f att %.1f body %.1f", level.maptime, gametic, pmo.angle, pmo.AttackAngle, b ? b.GetDouble("body.yaw", "", 0, 0, pmo) : -1); }
		if (watch != "" && watchEvery > 0 && level.maptime % watchEvery == 0)
		{
			let it = ThinkerIterator.Create("Actor");
			Actor a;
			while (a = Actor(it.Next()))
			{
				String cn = a.GetClassName();
				if (cn.IndexOf(watch) < 0) continue;
				Console.Printf("AP W t%d %s pos %.1f %.1f %.1f vel %.1f %.1f %.1f hp %d", level.maptime, cn, a.pos.x, a.pos.y, a.pos.z, a.vel.x, a.vel.y, a.vel.z, a.health);
			}
		}
	}
	override void WorldThingSpawned(WorldEvent e)
	{
		if (watch != "" && e.Thing) { String cn = e.Thing.GetClassName(); if (cn.IndexOf(watch) >= 0)
			Console.Printf("AP SPAWN t%d %s at %.1f %.1f %.1f vel %.1f %.1f %.1f", level.maptime, e.Thing.GetClassName(), e.Thing.pos.x, e.Thing.pos.y, e.Thing.pos.z, e.Thing.vel.x, e.Thing.vel.y, e.Thing.vel.z); }
	}
	override void WorldThingDied(WorldEvent e)
	{
		if (e.Thing) Console.Printf("AP DIED t%d %s by %s inflictor %s", level.maptime, e.Thing.GetClassName(), e.Thing.target ? e.Thing.target.GetClassName() : 'none', e.Inflictor ? e.Inflictor.GetClassName() : 'none');
	}
	override void WorldThingDamaged(WorldEvent e)
	{
		if (logDmg && e.Thing) Console.Printf("AP DMG t%d %s took %d (%s) from %s via %s hp %d", level.maptime, e.Thing.GetClassName(), e.Damage, e.DamageType, e.DamageSource ? e.DamageSource.GetClassName() : 'none', e.Inflictor ? e.Inflictor.GetClassName() : 'none', e.Thing.health);
	}
	override void NetworkProcess(ConsoleEvent e)
	{
		Array<String> w; e.Name.Split(w, ":");
		String n = w[0];
		let pmo = P();
		if (n ~== "a_hands") Hands("asked");
		else if (n ~== "a_pose") Pose("asked");
		else if (n ~== "a_inv") Inv("asked");
		else if (n ~== "a_rig") Rig(w.Size() > 1 ? w[1] : "asked");
		else if (n ~== "a_svc") SvcAsk(w, e);
		else if (n ~== "a_every") every = e.Args[0];
		else if (n ~== "a_input") logInput = e.Args[0] != 0;
		else if (n ~== "a_dmglog") logDmg = e.Args[0] != 0;
		else if (n ~== "a_watch") { watch = w.Size() > 1 ? w[1] : ""; watchEvery = e.Args[0] > 0 ? e.Args[0] : 1; }
		else if (n ~== "a_spawn" && pmo && w.Size() > 1)
		{
			// a_spawn:Class fwd left [angle-relative facing 180 = toward player]
			Vector2 f = Actor.AngleToVector(pmo.angle, e.Args[0]);
			Vector2 l = Actor.AngleToVector(pmo.angle + 90, e.Args[1]);
			let a = Actor.Spawn(w[1], (pmo.pos.x + f.x + l.x, pmo.pos.y + f.y + l.y, pmo.floorz + e.Args[2]));
			if (a) { a.angle = pmo.angle + 180; Console.Printf("AP SPAWNED t%d %s at %.1f %.1f %.1f hp %d", level.maptime, a.GetClassName(), a.pos.x, a.pos.y, a.pos.z, a.health); }
			else Console.Printf("AP SPAWN FAILED %s", w[1]);
		}
		else if (n ~== "a_tele" && pmo)
		{
			pmo.SetOrigin((e.Args[0], e.Args[1], pmo.floorz), false);
			pmo.angle = e.Args[2];
			Console.Printf("AP TELE t%d %.1f %.1f %.1f yaw %.1f", level.maptime, pmo.pos.x, pmo.pos.y, pmo.pos.z, pmo.angle);
		}
		else if (n ~== "a_hurt" && pmo) { pmo.DamageMobj(null, null, e.Args[0], 'None'); Console.Printf("AP HURT t%d %d hp %d", level.maptime, e.Args[0], pmo.health); }
		else if (n ~== "a_count" && w.Size() > 1)
		{
			int c = 0; let it = ThinkerIterator.Create("Actor"); Actor a;
			while (a = Actor(it.Next())) { String cn = a.GetClassName(); if (cn.IndexOf(w[1]) < 0) continue; c++; Console.Printf("AP C %s pos %.1f %.1f %.1f hp %d target %s", a.GetClassName(), a.pos.x, a.pos.y, a.pos.z, a.health, a.target ? a.target.GetClassName() : 'none'); }
			Console.Printf("AP COUNT t%d %s %d", level.maptime, w[1], c);
		}
		else if (n ~== "a_vis")
		{
			String o = "";
			Array<String> cls; Array<int> vis, all;
			let it = ThinkerIterator.Create("Actor"); Actor a;
			while (a = Actor(it.Next()))
			{
				String cn = a.GetClassName();
				if (cn.IndexOf("RSQTR_") != 0) continue;
				int k = cls.Find(cn);
				if (k == cls.Size()) { cls.Push(cn); vis.Push(0); all.Push(0); }
				all[k]++;
				if (!a.bINVISIBLE && a.Alpha > 0.01) { vis[k]++; o = o .. String.Format(" %s(f%d a%.2f)", cn, a.frame, a.Alpha); }
			}
			String t = "";
			for (int i = 0; i < cls.Size(); i++) t = t .. String.Format(" %s %d/%d", cls[i], vis[i], all[i]);
			Console.Printf("AP VIS t%d hp %d visible/all:%s", level.maptime, pmo ? pmo.health : -1, t);
			Console.Printf("AP VISON t%d%s", level.maptime, o);
		}
		else if (n ~== "a_echo") Console.Printf("AP MARK t%d %s", level.maptime, w.Size() > 1 ? w[1] : "");
	}
}
