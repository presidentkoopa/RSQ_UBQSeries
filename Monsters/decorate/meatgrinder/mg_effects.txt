ACTOR MG_Bullet : FastProjectile
{
	Radius 1
	Height 1
	Speed 250
	Damage (random (10, 10))
	+MISSILE
	+BLOODSPLATTER
	+ACTIVATEIMPACT
	+ACTIVATEPCROSS
	Renderstyle Add
	+BRIGHT
	Scale 0.2
	// The tracer: HF's MG ships no TRAC sprite, so the owner's streak model draws it
	// (MODELDEF.txt, models/meatgrinder/tracer/). MGTR A is the 1x1 placeholder frame
	// the model binds to; Add + fullbright because the skin is RGB on black.
	States
	{
	Spawn:
		MGTR A 1 Bright
		MGTR A -1 Bright
		Loop
	Death:
		TNT1 A 0
		TNT1 A 0 A_SpawnItem("BulletPuff")
		Stop
	}
}
