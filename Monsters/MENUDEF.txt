OptionMenu "RSQMN_Options"
{
	Title "RSQ MONSTERS"
	// tip: How tough every monster is, by colour. Changes the monsters on the map now. Beating a Boss raises it.
	Option "Monster Tier", "rsq_mon_worldtier", "RSQMN_TierNames"
	// tip: Which monsters the maps get: vanilla, Brutal Doom and XMVE on the colour tiers, or ColorfulHell.
	Option "Monster Set", "rsq_monster_set", "RSQMN_MonsterSet"
	// tip: Monsters are drawn from painted art in their tier colour where it exists; off uses the tint only.
	Option "Painted Colour Art", "rsq_mon_art", "OnOff"
	// tip: A gem showing each monster's tier floats above it (ColorfulHell's colour-blind aid).
	Option "Tier Icons", "rs_ch_colorblind", "OnOff"
	// rsq_mon_bdtier: Brutal Doom / XMVE / plain monsters' colours AND the World Tier everywhere (off: it counts as 1 for
	// ColorfulHell spawns and RSQ Miniboss's Boss too, and a Boss won or lost does not move it). Was "Colour Ladder On BD
	// Monsters" (11-16 bugfix N-MO2: the name said BD monsters only)
	// tip: Monsters take the Monster Tier's colour; off, every monster stays as its mod ships it.
	Option "Tiers", "rsq_mon_bdtier", "OnOff"
	// tip: When a monster changes tier: keep the health it lost, full health on an instant change, or always full.
	Option "On Tier Change", "rsq_mon_retier_health", "RSQMN_RetierHealth"

	StaticText " "
	StaticText "Boss", 1
	// tip: Where a miniboss fight's Boss comes from: the monster set or ColorfulHell.
	Option "Boss From", "rsq_mon_bf_boss", "RSQMN_BfBoss"
	// tip: Where the Boss's thralls come from: as the Boss, ColorfulHell, or the map's own monsters.
	Option "Thralls From", "rsq_mon_bf_thralls", "RSQMN_BfThralls"

	StaticText " "
	StaticText "ColorfulHell", 1
	// tip: ColorfulHell's extra colours (Cyan, Brown, Abyss, FireBlu, Gray): off, on, or half as often.
	RSQMNGroup "Extra Colours", "rsq_mon_ch_extras", "RSQMN_Extras"
	// tip: Black bosses spawn; off, they become Red.
	Option "Black Boss", "rs_ch_blackboss", "OnOff"
	// tip: White bosses spawn; off, they become Black.
	Option "White Boss", "rs_ch_whiteboss", "OnOff"
	// tip: How often a Black or White boss is an EX boss: never, rare, half or always.
	Option "EX Bosses", "rs_ch_exboss", "RSQMN_EXBoss"
	// tip: ColorfulHell's drops all at once: standard, scarce, generous or none.
	RSQMNGroup "Drops", "rsq_mon_ch_drops", "RSQMN_Drops"

	// RESERVED: the "Second Colour" section goes here (design/SECOND_COLOUR.md; its own build, owner 11-16 09:58).
