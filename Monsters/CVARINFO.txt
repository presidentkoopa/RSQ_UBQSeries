//                      1 ColorfulHell, 4 Map's Own Monsters (an old 2 or 3: ColorfulHell)

// only and the share on a flashed one (Hackfraud's rule), 2 always full health.
server int rsq_mon_retier_health = 1;

// Brutal Doom's, XMVE's and plain Doom's monsters on the colour ladder, in place (RS_BDTier.zs; the rule is
// design/BD_XMVE_TIERING.md), and the World Tier. rsq_mon_bdtier is the page's "World Tier Ladder": off, the kits stand
// down to their entry colour and the World Tier counts as 1 everywhere (ColorfulHell spawns, RSQ Miniboss's Boss).
// rsq_mon_worldtier is the World Tier: 1 = every monster as its mod ships it, each step one colour up. RSQ Miniboss
// raises it for a Boss won and lowers it for a Boss lost. A savegame holds it (QuestZDoom keeps every server cvar in a save
// and puts it back on load), so loading a save brings back the World Tier it was made at; a new game starts from this
// setting as it stands.
server bool rsq_mon_bdtier = true;
server int  rsq_mon_worldtier = 1;
server bool rsq_mon_bdtier_debug = false;    // RSBD log lines, for the rig
// The painted colour art (RS_BDArt.zs; the art itself is pk3/RSQ_BDXMTiers_<Colour>.pk3, one per colour):

server bool rsq_mon_art = true;
server bool rsq_mon_art_debug = false;       // RSART log lines: what the art driver draws and its time a tic
server bool rsq_mon_plus_debug = false;      // RSPLUSEX log lines: the White Arch-vile EX's armour steal and its count (RS_PlusEX.zs)
