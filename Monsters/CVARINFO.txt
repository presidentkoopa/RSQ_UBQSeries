// --- Threshold marks & boss bracket ---
// Both OFF by default; Default style stays 1:1 with upstream.
//
// Marks are LEARNED, not read from a table. CheckThreshold's fraction is
// a literal at each call site inside the monsters, so nothing stores it,
// and exposing it would mean editing monster files for a health-bar
// feature. Instead the bar watches ThresholdFired() and records the
// health fraction the tic a gate trips, remembered per monster CLASS.
// Consequence: the first Baron you fight teaches the system, every Baron
// after shows the mark before it enrages. First of a kind shows nothing.
user bool rs_hpbar_marks     = false;
// 0 Off, 1 Auto (health at or above the cutoff), 2 Always
user int  rs_hpbar_bossframe = 0;
// The Auto cutoff. 1000 splits our ladder cleanly -- T10 is 115, T11 is
// 2000 -- so it catches the genuine long fights and nothing else.
user int  rs_hpbar_boss_hp   = 1000;

// --- No More Hitscans (zscript/monsters/RS_NoHitscan.zs) -----
// Server-scope, not user: whether monster bullets travel is a rule of
// the match, not a client preference.
server bool  rs_nohitscan_enabled = true;
// Units per tic. 80 crosses a 2048-unit shot in ~26 tics; lower is more
// dodgeable, higher approaches the old instant behaviour.
server float rs_nohitscan_speed   = 80.0;
server float rs_nohitscan_alpha   = 0.5;

// --- Only when damaged ---
// Bars appear the first time a monster is hit, not from the moment it
// spawns. ON by default: this is an owner decision, not upstream's --
// upstream shows a bar on everything from the start.
//
// Measured against the REAL ceiling. The hand-edit this replaces used
// SpawnHealth() and so had the main bug all over again: an overhealed
// monster never counted as damaged until chewed below its class default.
//
// It is also the single biggest performance lever here. In a large map
// most monsters are never touched, and every one of those now costs zero
// actors, zero thinking and zero draw calls.
user bool rs_hpbar_onlydamaged = true;

// tier tables keep CH's real value; this is the floor applied at use.
