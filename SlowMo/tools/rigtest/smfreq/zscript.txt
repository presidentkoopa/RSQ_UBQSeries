version "4.14"
// TEST ONLY. Never ships (SlowMo's build packs only its listed lumps; tools/ is not one). How often slow
// motion starts by itself in a fight (owner, 2026-11-02: "it's firing all the fuckin time").
//   smf_mode 1  THE FIGHT, the same script before and after: god mode, the pawn stands still; every 4 s up to
//               3 monsters spawn in sight (imps, zombiemen, shotgunners, a demon; ALLOW_REPLACE so Brutal Doom's
//               own come); every 25 tics the pawn hits the nearest one it can see (DamageMobj, 25-70); every
//               2 s it fires a real rocket at it (Brutal Doom's when loaded: splash, gibs, airborne kills);
//               every 8 s a magazine reload is faked; every 10 s a zombieman steps up and is shoved
//               (rsqml_buttshove, Melee's own event: the deliberate path).
//   smf_mode 2  MELEE SHOVE only: a zombieman 50 units ahead, every 3 s rsqml_buttshove.
//   smf_mode 3  TIME SCALE: no monsters; `netevent rsm_on`, then every 35 tics the clock's scale and a
//               test fireball's measured speed against its full speed (the actual time scale).
//   smf_mode 5  HAND POSES (set from the console script with qz_rhand / qz_lhand): logs every gesture toggle and
//               the hands' gap each second, so a two-hand grip or a hand reload that toggles slow motion shows.
//   smf_mode 6  OTHER MODS' rsm_pulse, as they send it: every 3 s RSQ Score's big-moment pulse (600 35) and,
//               1.5 s later, a perfect parry's (500 35), for smf_secs.
//   smf_mode 7  PER-TRIGGER SWITCHES: five 300-tic phases; each sends Score's pulse (600 35), a Gun Bonsai VR
//               Adrenaline lv1 pulse (450 55), the cigarette's (rsqtr_cig_secs x 1000, rsqtr_cig_depth) and a real
//               Melee shove. Phase 0 all on; then Score, Gun Bonsai VR, Cigarette, Melee Shove switched off in turn
//               (only that one). Set rsqsm_auto_cooldown 0 and rsqsm_debug 1 in the console script.
//   smf_mode 4  THE NAMED DOOR (another mod's call): rsqsm_start:<who> and the service, then the cooldown.
// Lines: T_EVT every slow-mo door event seen; T_START when the clock leaves full speed (with the events of
// that tic); T_SUM totals at the end.
class RSQSMF_Probe : EventHandler
{
	int starts, slowTics, kills, spawned, reloads, evPulse, evFear, evOther, lastStart, rockets, shoves;
	bool wasHeading;
	Array<Actor> mons;
	String tickEvents;
	int tickEventsAt;
	Actor ball;
	Vector3 ballLast;

	static int Mode() { let c = CVar.FindCVar("smf_mode"); return c ? c.GetInt() : 1; }
	static int Secs() { let c = CVar.FindCVar("smf_secs"); return c ? c.GetInt() : 90; }

	static int MbWindow()
	{
		let it = ServiceIterator.Find("RSQMB_Service");
		let s = it.Next();
		return s ? s.GetInt("window") : -1;
	}

	override void NetworkProcess(ConsoleEvent e)
	{
		String n = e.Name.MakeLower();
		if (n == "rsm_pulse") evPulse++;
		else if (n == "rsqml_fear") evFear++;
		else if (n.Left(4) == "rsm_" || n == "rsqsm_burst" || n.Left(11) == "rsqsm_start") evOther++;
		else return;
		Console.Printf("T_EVT t%d %s %d %d %d (player %d) mbwindow %d", level.maptime, e.Name, e.Args[0], e.Args[1], e.Args[2], e.Player, MbWindow());
		if (tickEventsAt != level.maptime) { tickEventsAt = level.maptime; tickEvents = ""; }
		tickEvents.AppendFormat("%s %d %d; ", e.Name, e.Args[0], e.Args[1]);
	}

	override void WorldThingDied(WorldEvent e)
	{
		if (e.Thing && e.Thing.bISMONSTER) kills++;
	}

	Actor SpawnIn(PlayerPawn mo, Class<Actor> cls)
	{
		for (int tries = 0; tries < 12; tries++)
		{
			double a = mo.angle + frandom[smf](-70, 70);
			double d = frandom[smf](220, 700);
			Vector3 at = (mo.pos.xy + (cos(a), sin(a)) * d, 0);
			let m = Actor.Spawn(cls, at, ALLOW_REPLACE);
			if (!m) continue;
			m.SetOrigin((m.pos.xy, m.floorz), false);
			if (!m.TestMobjLocation() || !level.IsPointInLevel(m.pos) || !mo.CheckSight(m)) { m.Destroy(); continue; }
			m.target = mo;
			mons.Push(m);
			spawned++;
			return m;
		}
		return null;
	}

	Actor Nearest(PlayerPawn mo)
	{
		Actor best = null; double bd = 1e9;
		let it = ThinkerIterator.Create("Actor");
		Actor a;
		while (a = Actor(it.Next()))
		{
			if (!a.bISMONSTER || a.health <= 0 || a.bFRIENDLY) continue;
			double d = mo.Distance3D(a);
			if (d < bd && d < 1500 && mo.CheckSight(a)) { bd = d; best = a; }
		}
		return best;
	}

	override void WorldTick()
	{
		let p = players[consoleplayer];
		if (!p || !p.mo) return;
		let mo = p.mo;
		int t = level.maptime;
		int m = Mode();
		p.cheats |= CF_GODMODE;
		mo.vel.xy = (0, 0);

		let c = RSQSM_Clock.Get();
		if (c)
		{
			if (c.scale < 0.9999) slowTics++;
			if (c.target < 0.9999 && !wasHeading)
			{
				starts++;
				Console.Printf("T_START t%d (%.1f s since the last) on %d pulseUntil %d target %.2f by: %s", t, (t - lastStart) / 35.0, c.on, c.pulseUntil, c.target,
					tickEventsAt == t ? tickEvents : "(no event this tic: SlowMo's own trigger)");
				lastStart = t;
			}
			wasHeading = c.target < 0.9999;
		}

		if (m == 1 && t > 10 && t < Secs() * TICRATE)
		{
			if (t % 140 == 20)
			{
				static const String cls[] = { "DoomImp", "ZombieMan", "ShotgunGuy", "DoomImp", "Demon", "ZombieMan" };
				int alive = 0;
				for (int i = 0; i < mons.Size(); i++) if (mons[i] && mons[i].health > 0) alive++;
				for (int k = 0; k < 3 && alive < 8; k++, alive++) SpawnIn(mo, cls[random[smf](0, cls.Size() - 1)]);
			}
			if (t % 25 == 0)
			{
				let best = Nearest(mo);
				if (best) best.DamageMobj(mo, mo, random[smf](25, 70), 'None');
			}
			if (t % 70 == 35)
			{
				let best = Nearest(mo);
				if (best) { let r = mo.SpawnMissile(best, "Rocket"); if (r) rockets++; }
			}
			if (t % 280 == 140)
			{
				let w = p.ReadyWeapon;
				if (w && w.Ammo1 && w.Ammo2 && w.Ammo1 != w.Ammo2)
				{
					w.Ammo1.Amount = max(0, w.Ammo1.Amount - 3);
					w.Ammo2.Amount = min(w.Ammo2.MaxAmount, w.Ammo2.Amount + 3);
					reloads++;
					Console.Printf("T_RELOAD t%d %s", t, w.GetClassName());
				}
			}
			Shove(mo, t, 350, 200);
		}
		if (m == 2 && t > 10 && t < Secs() * TICRATE) Shove(mo, t, 105, 20);
		if (m == 3) TimeScale(mo, c, t);
		if (m == 4) NamedDoor(c, t);
		if (m == 6 && t > 10 && t < Secs() * TICRATE)
		{
			if (t % 105 == 0) EventHandler.SendNetworkEvent("rsm_pulse", 600, 35);
			if (t % 105 == 52) EventHandler.SendNetworkEvent("rsm_pulse", 500, 35);
		}
		if (m == 7) Switches(mo, t);
		if (m == 5 && t % 35 == 0)
		{
			Vector3 a = mo.AttackPos, b = mo.OffhandPos;
			Console.Printf("T_HANDS t%d vr %d gap %.3f m main-off lateral %.3f m on %d", t, mo.OverrideAttackPosDir, RSQSM_Clock.Metres(a - b),
				((a.x - b.x) * sin(mo.angle) - (a.y - b.y) * cos(mo.angle)) / 34.0, c ? c.on : -1);
		}
		if (t == Secs() * TICRATE + 70)
			Console.Printf("T_SUM %d s: slow-mo starts %d (%.1f a minute), slowed %.1f s (%.0f %%), rsm_pulse events %d, rsqml_fear %d, other doors %d, kills %d, spawned %d, rockets %d, shoves %d, faked reloads %d",
				Secs(), starts, starts * 60.0 / Secs(), slowTics / 35.0, slowTics * 100.0 / (Secs() * TICRATE), evPulse, evFear, evOther, kills, spawned, rockets, shoves, reloads);
	}

	void Shove(PlayerPawn mo, int t, int every, int at)
	{
		Vector2 fwd = (cos(mo.angle), sin(mo.angle));
		if (t % every == at)
		{
			let z = Actor.Spawn("ZombieMan", (mo.pos.xy + fwd * 50, mo.floorz), NO_REPLACE);
			if (z) { z.angle = z.AngleTo(mo); z.bDORMANT = false; }
		}
		if (t % every == at + 10)
		{
			int yaw = int(round(mo.angle)) % 360; if (yaw < 0) yaw += 360;
			EventHandler.SendNetworkEvent("rsqml_buttshove", 0, 350, yaw * 1000 + 90);
			shoves++;
			Console.Printf("T_SHOVE t%d", t);
		}
	}

	void Switches(PlayerPawn mo, int t)
	{
		static const String sw[] = { "", "rsqsm_trig_score", "rsqsm_trig_gbvr", "rsqsm_trig_smoke", "rsqsm_trig_shove" };
		int ph = t / 300, k = t % 300;
		if (ph > 4) return;
		if (k == 1)
		{
			for (int i = 1; i < sw.Size(); i++) CVar.FindCVar(sw[i]).SetInt(i == ph ? 0 : 1);
			Console.Printf("T_PHASE %d t%d: %s off", ph, t, ph ? sw[ph] : "nothing");
		}
		if (k == 40) EventHandler.SendNetworkEvent("rsm_pulse", 600, 35);
		if (k == 110) EventHandler.SendNetworkEvent("rsm_pulse", 450, 55);
		if (k == 180)
		{
			let cs = CVar.GetCVar("rsqtr_cig_secs", players[consoleplayer]);
			let cd = CVar.GetCVar("rsqtr_cig_depth", players[consoleplayer]);
			if (cs && cd) EventHandler.SendNetworkEvent("rsm_pulse", int(cs.GetFloat() * 1000), cd.GetInt());
		}
		Shove(mo, k, 300, 230);
	}

	// the actual time scale: a test fireball (no target, no gravity, constant speed) flying away from the pawn
	void TimeScale(PlayerPawn mo, RSQSM_Clock c, int t)
	{
		if (t == 20) EventHandler.SendNetworkEvent("rsm_on");
		if (t == 20 || (ball && ball.bDestroyed)) ball = null;
		if (!ball && t >= 20 && t % 35 == 0)
		{
			ball = Actor.Spawn("TimeProbeBall", mo.pos + (0, 0, 40), NO_REPLACE);
			if (ball) { ball.vel = (cos(mo.angle + 180), sin(mo.angle + 180), 0) * ball.Speed; ballLast = ball.pos; }
			return;
		}
		if (ball && c && t % 35 == 17)
		{
			double moved = (ball.pos.xy - ballLast.xy).Length();
			Console.Printf("T_SCALE t%d rsm_depth %.2f clock scale %.4f target %.2f | test ball moved %.3f units/tic, full speed %.1f -> actual time scale %.4f | actors slowed %d",
				t, CVar.FindCVar("rsm_depth").GetFloat(), c.scale, c.target, moved, ball.Speed, moved / ball.Speed, c.act.Size());
		}
		if (ball) ballLast = ball.pos;
		if (t == 400) { EventHandler.SendNetworkEvent("rsm_off"); }
	}

	void NamedDoor(RSQSM_Clock c, int t)
	{
		if (t == 20) EventHandler.SendNetworkEvent("rsqsm_start:rigprobe", 1000, 0, 0);          // deliberate, the slider's speed
		if (t == 120) EventHandler.SendNetworkEvent("rsqsm_start:rigprobe.auto", 1000, 50, 1);   // automatic, its own 50 %
		if (t == 220) EventHandler.SendNetworkEvent("rsqsm_start:rigprobe.auto", 1000, 50, 1);   // automatic again: cooldown
		if (t == 260) EventHandler.SendNetworkEvent("rsqsm_start:rigprobe", 1000, 10, 0);        // deliberate: never blocked
		if (t == 360)
		{
			let it = ServiceIterator.Find("RSQSM_Service");
			let s = it.Next();
			int r1 = s ? s.GetInt("start", "rigprobe.svc", 700, 0.40) : -9;
			Console.Printf("T_SVC t%d start (700 ms, 0.40) -> %d", t, r1);
		}
		if (t == 420)
		{
			let it = ServiceIterator.Find("RSQSM_Service");
			let s = it.Next();
			int r2 = s ? s.GetInt("start.auto", "rigprobe.svc", 700, 0) : -9;
			Console.Printf("T_SVC t%d start.auto (700 ms, slider) -> %d", t, r2);
		}
		if (c && t % 10 == 0 && t < 520)
			Console.Printf("T_CLK t%d scale %.3f target %.2f pulseUntil %d", t, c.scale, c.target, c.pulseUntil);
	}
}

class TimeProbeBall : Actor
{
	Default { Speed 10; Radius 2; Height 2; +NOGRAVITY +NOBLOCKMAP +NOINTERACTION +DONTSPLASH }
	States { Spawn: TNT1 A -1; Stop; }
}
