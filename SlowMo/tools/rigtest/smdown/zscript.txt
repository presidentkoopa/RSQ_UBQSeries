class RSQSMD_Probe : EventHandler
{
	Array<Actor> blk;
	Array<Actor> z;
	int t0;
	bool started;
	Actor crate, crate2;
	static clearscope int Mode() { let c = CVar.FindCVar("dn_mode"); return c ? c.GetInt() : 0; }
	static void SetF(String n, double v) { let c = CVar.FindCVar(n); if (c) c.SetFloat(v); else Console.Printf("T_NOCVAR %s", n); }
	static void SetI(String n, int v) { let c = CVar.FindCVar(n); if (c) c.SetInt(v); else Console.Printf("T_NOCVAR %s", n); }

	Actor Near(Vector3 at)
	{
		let it = ThinkerIterator.Create("Actor");
		Actor a, best;
		double bd = 1e9;
		while (a = Actor(it.Next()))
		{
			if (!a.bISMONSTER || !a.bCOUNTKILL || a.health <= 0 || a.player || z.Find(a) < z.Size()) continue;
			double d = (a.pos.xy - at.xy).Length();
			if (d < bd) { bd = d; best = a; }
		}
		return best;
	}
	Service Svc(String n)
	{
		let it = ServiceIterator.Find(n);
		Service s;
		while (s = it.Next()) if (s.GetClassName() == n) return s;
		return null;
	}
	void Report(int t, String why)
	{
		let pmo = players[consoleplayer].mo;
		let p = players[consoleplayer];
		let d = RSQSM_Down.Get();
		let c = RSQSM_Clock.Get();
		let sc = Svc("RS_ScoreService");
		int n = consoleplayer;
		let ar = BasicArmor(pmo.FindInventory("BasicArmor"));
		let w = RSQSM_DownWatch(pmo.FindInventory("RSQSM_DownWatch"));
		String tg = "";
		for (int i = 0; i < d.tMon.Size(); i++)
			if (d.tPn[i] == n) tg = tg .. String.Format(" [%s icon %d col %06X %s%s]", d.tMon[i] ? String.Format("%s", d.tMon[i].GetClassName()) : "gone", d.tIcon[i], d.tCol[i],
				(d.tDead[i] != 0) ? "dead" : "alive", (d.tSpawned[i] != 0) ? " spawned" : "");
		Console.Printf("T_STATE t%d (maptime %d) %s: health %d/%d armour %d, DOWN %d (clock %.2f s, pool %.2f s, bar %.2f s, owed %d, downs %d, bank %.0f/%.0f), invul %d, buddha %d, world %.2f, view %.1f, crawl x%.2f, Score lives %d, targets:%s",
			t, level.maptime, why, pmo.health, pmo.GetMaxHealth(true), ar ? ar.Amount : 0, d.downOn[n], d.clk[n] / 35.0, d.pool[n] / 35.0, d.Elapsed(n) / 35.0, d.owed[n],
			w ? w.downs : -1, d.bankHp[n], d.bankAr[n], d.Invul(n), pmo.bBUDDHA, c.scale, p.viewheight, d.Crawl(n), sc ? sc.GetInt("lives", "", 0, 0, pmo) : -9, tg);
	}
	void God(bool on)
	{
		let p = players[consoleplayer];
		if (on) p.cheats |= CF_GODMODE; else p.cheats &= ~CF_GODMODE;
	}
	void Hit(int t, Actor src, int hp, int dmg, String why)
	{
		let pmo = players[consoleplayer].mo;
		if (hp > 0) pmo.A_SetHealth(hp);
		God(false);
		Console.Printf("T_HIT t%d %s: health %d, %d from %s", t, why, pmo.health, dmg, src ? String.Format("%s", src.GetClassName()) : "nothing");
		pmo.DamageMobj(src, src, dmg, 'None');
		Console.Printf("T_AFTER t%d health %d", t, pmo.health);
		God(true);
	}
	void Shoot(int t, Actor a, int dmg, String why)
	{
		let pmo = players[consoleplayer].mo;
		if (!a) { Console.Printf("T_SHOOT t%d %s: none", t, why); return; }
		int was = a.health;
		a.DamageMobj(pmo, pmo, dmg, 'None');
		Console.Printf("T_SHOOT t%d %s: %s health %d -> %d (%d asked)", t, why, a.GetClassName(), was, a.health, dmg);
	}
	Actor Target(int k, bool spawnedOnly)
	{
		let d = RSQSM_Down.Get();
		int j = 0;
		for (int i = 0; i < d.tMon.Size(); i++)
		{
			if (d.tPn[i] != consoleplayer || d.tDead[i] != 0 || !d.tMon[i]) continue;
			if (spawnedOnly && (d.tSpawned[i] == 0)) continue;
			if (j == k) return d.tMon[i];
			j++;
		}
		return null;
	}
	void Inv(int t, String why)
	{
		let pmo = players[consoleplayer].mo;
		static const String names[] = { "RifleAmmo", "Clip2", "BDPistolAmmo", "Clip1", "ShotgunAmmo", "AmmoShell", "RSQTR_CigPack", "GrenadeAmmo", "RSQGN_Ammo" };
		String s = "";
		for (int i = 0; i < names.Size(); i++)
		{
			Name nm = names[i];
			class<Inventory> ic = nm;
			let it = ic ? pmo.FindInventory(ic) : null;
			s = s .. String.Format(" %s=%s", names[i], it ? String.Format("%d/%d", it.Amount, it.MaxAmount) : "-");
		}
		let p = players[consoleplayer];
		Console.Printf("T_INV t%d %s:%s; ready %s offhand %s", t, why, s, p.ReadyWeapon ? String.Format("%s", p.ReadyWeapon.GetClassName()) : "-",
			p.OffhandWeapon ? String.Format("%s", p.OffhandWeapon.GetClassName()) : "-");
	}
	void SetAmmo(String n, int v) { let pmo = players[consoleplayer].mo; Name nm = n; class<Inventory> c = nm; if (c) pmo.SetInventory(c, v); }
	// the view is the rig's head (its yaw may be put back each frame), so the MONSTER is moved round the player instead
	void Place(int t, Actor a, double ang, double dist, String why)
	{
		let pmo = players[consoleplayer].mo;
		if (!a) { Console.Printf("T_PLACE t%d %s: none", t, why); return; }
		double aa = pmo.angle + ang;
		Vector2 xy = pmo.pos.xy + (cos(aa), sin(aa)) * dist;
		a.SetOrigin((xy, pmo.floorz), false);
		a.angle = a.AngleTo(pmo);
		Console.Printf("T_PLACE t%d %s: %s moved to %.0f degrees off the view, %.0f units (fits %d)", t, why, a.GetClassName(), ang, dist, a.TestMobjLocation());
	}

	ui int frames;

	{
		frames++;
		if (Mode() != 0) Console.Printf("T_FRAME %d maptime %d", frames, level.maptime);
	}

	void Lines(int t, String why)
	{
		let d = RSQSM_Down.Get();
		int n = consoleplayer;
		String s = "";
		for (int i = 0; i < d.tMon.Size(); i++)
			if (d.tPn[i] == n) s = s .. String.Format(" [\"%s\" icon %d %s]", d.tName[i], d.tIcon[i], d.tDead[i] == 0 ? "alive" : (d.tDead[i] == 1 ? "DEAD" : "gone"));
		for (int i = 0; i < d.spPn.Size(); i++) if (d.spPn[i] == n) s = s .. String.Format(" [waiting %s icon %d]", d.spCls[i], d.spIcon[i]);
		String pl = " (HUD: drawn by downhud.zs from these)";
		Console.Printf("T_LINES t%d %s: targets%s; plate lines%s; bar elapsed %.2f s, refill at %d (sweep %d tics), clock %.2f s",
			t, why, s, pl, d.Elapsed(n) / 35.0, d.refillTic[n], RSQSM_Down.REFILL_TICS, d.clk[n] / 35.0);
	}
	void Block(int t, bool on)
	{
		let pmo = players[consoleplayer].mo;
		if (!on) { int k = blk.Size(); for (int i = 0; i < blk.Size(); i++) if (blk[i]) blk[i].Destroy(); blk.Clear(); Console.Printf("T_BLOCK t%d %d blockers removed", t, k); return; }
		static const double DISTS[] = { 640, 512, 384, 288, 192 };
		static const double ANGS[] = { 0, 12, -12, 24, -24, 36, -36, 48, -48, 60, -60, 30, -30 };
		for (int i = 0; i < DISTS.Size(); i++)
			for (int j = 0; j < ANGS.Size(); j++)
			{
				double aa = pmo.angle + ANGS[j];
				Vector2 xy = pmo.pos.xy + (cos(aa), sin(aa)) * DISTS[i];
				let b = Actor.Spawn("RSQSMD_Blocker", (xy, Actor.ONFLOORZ), NO_REPLACE);
				if (b) blk.Push(b);
			}
		Console.Printf("T_BLOCK t%d %d blockers on every in-view spawn spot (angle %.0f)", t, blk.Size(), pmo.angle);
	}

	//   t100 Target Lines Name (the lines show Score's monster.name: Champions' colour words); t200 back to Icon
	void Champs(int t, String why)
	{
		let d = RSQSM_Down.Get();
		String s = "";
		for (int i = 0; i < d.tMon.Size(); i++)
		{
			let m = d.tMon[i];
			if (d.tPn[i] != consoleplayer || !m) continue;
			int icons = 0, fires = 0;
			let it = ThinkerIterator.Create("Actor");
			Actor a;
			while (a = Actor(it.Next()))
			{
				String cn = a.GetClassName();
				if (a.master == m && cn ~== "champion_Icon") icons++;
				if (cn ~== "champion_ArchvileFire" && (a.pos.xy - m.pos.xy).Length() < 4) fires++;
			}
			s = s .. String.Format(" [%s%s: %s; health %d (spawn %d), translation %d, champion_Icon %d, champion_ArchvileFire %d, colour %06X, name \"%s\"]",
				m.GetClassName(), d.tSpawned[i] ? " spawned" : " map's", RSQSM_Down.ChampWhat(m) != "" ? RSQSM_Down.ChampWhat(m) : "no Champions inventory",
				m.health, m.GetSpawnHealth(), m.Translation, icons, fires, d.tCol[i], d.tName[i]);
		}
		int hid = 0;
		let it2 = ThinkerIterator.Create("Actor");
		Actor b;
		while (b = Actor(it2.Next())) if (b.bISMONSTER && b.bINVISIBLE && b.health > 0) hid++;
		Console.Printf("T_CHAMP t%d %s:%s; hidden living monsters left %d; candidates waiting %d; Target Lines %d", t, why, s, hid, d.cdMon.Size(),
			CVar.GetCVar("rsqsm_down_lines", players[consoleplayer]).GetInt());
	}

	void Marks(int t, String why)
	{
		let d = RSQSM_Down.Get();
		String s = "";
		if (d.plate)
			for (int i = 0; i < d.plate.marks.Size(); i++)
			{
				let m = d.plate.marks[i];
				if (!m || !m.mon) continue;
				int tv = 0;
				double tal = 0;
				for (int k = 0; k < 6; k++) if (m.tC[k] && !m.tC[k].bInvisible) { tv++; tal = m.tC[k].alpha; }
				double off = 0;
				let pmo = players[consoleplayer].mo;
				Vector3 to = m.mon.pos + (0, 0, m.mon.height * 0.5) - (pmo.pos + (0, 0, players[consoleplayer].viewheight));
				Vector3 gz = (cos(pmo.angle) * cos(pmo.pitch), sin(pmo.angle) * cos(pmo.pitch), -sin(pmo.pitch));
				off = acos(clamp((to / max(1.0, to.Length())) dot gz, -1.0, 1.0));
				s = s .. String.Format(" [%s icon %d %.0f deg off: mode %d (%s), edge direction %.0f, trail %d of 6 drawn (alpha %.2f), crosshair %s frame %d]", m.mon.GetClassName(), m.icon, off,
					m.mode, m.mode == 1 ? "edge arrows (HUD)" : (m.mode == 2 ? "trail" : "none"), m.eAng, tv, tal,
					(m.xhair && !m.xhair.bInvisible) ? "drawn" : "hidden", min(16, level.maptime - m.lockTic));
			}
		Console.Printf("T_MARKS t%d %s:%s", t, why, s);
	}
