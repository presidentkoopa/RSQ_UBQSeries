version "4.14"
// TEST ONLY. Never ships. RSQ SlowMo's DOWN v3 (design/NEAR_DEATH.md) + RSQ Score's life crate + Graves Rise's cap, on the
// rig, the full load order, in ONE run (dn_mode 1), in god mode (god is dropped only for the probe's own lethal hits).
// Every line starts T_ for grep. It links the RSQ classes it reads (test only: loaded with all of them).
// Timeline (tics from the start, MAP01):
//   A  t40 lethal by z0 (in the fight) -> DOWN, owed 1, z0 the target; t50 z1 hit (time bought); t58 a monster hit on
//      the downed player (held at 1 HP); t62 z1 killed (bits fly into the plate and bank); t80 z0 killed -> TOP tier:
//      bank, invulnerable 5 s, slow-mo 3 s, Revival Fire, +1 cigarette, +1 grenade, magazines free
//   B  t270 lethal by z2 (after the top tier's 5 s invulnerability) -> DOWN 2, owed 2: z2 + a spawned copy in view; t340
//      the spawned one killed (cracked slot); t375 the tall step breaks; t430 z2 killed -> MIDDLE tier
//   C  t540 z3 moved behind the player, t545 lethal by z3 (behind) -> DOWN 3, owed 3: z3 behind (turn marker,
//      U-turn), two spawned in view; t600 z3 moved to his left, t640 in front (the icon flies on and assembles); t700 / t720 the
//      spawned ones killed, t775 z3 killed -> BOTTOM tier
//   D  t900 a Score life given; t905 lethal by z4 -> DOWN 4, owed 4; t960 every target moved off view (right, left, behind): marker styles A t965,
//      B t990, D t1015, C t1040; nobody killed: the clock runs out ~t1220 -> the Score life -> spawned targets despawn in
//      teleport fog, z4 loses its boost, the count goes back
//   E  t1260 a Bossfight payoff (RSQ Miniboss's own BfPayoff) -> the life crate where the boss stood, moved 80 ahead for the
//      frame; Distance Grab's rule asked; t1450 breathing red; t1612 it runs out; t1620 a second crate taken (+1 life)
//   F  t1640 Graves Rise: 30 tombstones, Graveyard's RaiseGraves with a budget of 5 (no cap: 30 risers, Legion of Bones
//      loaded); the Bones Bridge (gy_bones on from t5): the BD monsters killed above rise as LoB undead after LoB's delay
class RSQSMD_Probe : EventHandler
{
	Array<Actor> z;
	int t0;
	bool started;
	Actor crate, crate2;
	static clearscope int Mode() { let c = CVar.FindCVar("dn_mode"); return c ? c.GetInt() : 0; }
	static void SetF(String n, double v) { let c = CVar.FindCVar(n); if (c) c.SetFloat(v); else Console.Printf("T_NOCVAR %s", n); }
	static void SetI(String n, int v) { let c = CVar.FindCVar(n); if (c) c.SetInt(v); else Console.Printf("T_NOCVAR %s", n); }

	Actor Near(Vector3 at)
	{
		let it = ThinkerIterator.Create("Actor");
		Actor a, best;
		double bd = 1e9;
		while (a = Actor(it.Next()))
		{
			if (!a.bISMONSTER || !a.bCOUNTKILL || a.health <= 0 || a.player || z.Find(a) < z.Size()) continue;
			double d = (a.pos.xy - at.xy).Length();
			if (d < bd) { bd = d; best = a; }
		}
		return best;
	}
	Service Svc(String n)
	{
		let it = ServiceIterator.Find(n);
		Service s;
		while (s = it.Next()) if (s.GetClassName() == n) return s;
		return null;
	}
	void Report(int t, String why)
	{
		let pmo = players[consoleplayer].mo;
		let p = players[consoleplayer];
		let d = RSQSM_Down.Get();
		let c = RSQSM_Clock.Get();
		let sc = Svc("RS_ScoreService");
		int n = consoleplayer;
		let ar = BasicArmor(pmo.FindInventory("BasicArmor"));
		let w = RSQSM_DownWatch(pmo.FindInventory("RSQSM_DownWatch"));
		String tg = "";
		for (int i = 0; i < d.tMon.Size(); i++)
			if (d.tPn[i] == n) tg = tg .. String.Format(" [%s icon %d col %06X %s%s]", d.tMon[i] ? String.Format("%s", d.tMon[i].GetClassName()) : "gone", d.tIcon[i], d.tCol[i],
				(d.tDead[i] != 0) ? "dead" : "alive", (d.tSpawned[i] != 0) ? " spawned" : "");
		Console.Printf("T_STATE t%d (maptime %d) %s: health %d/%d armour %d, DOWN %d (clock %.2f s, pool %.2f s, bar %.2f s, owed %d, downs %d, bank %.0f/%.0f), invul %d, buddha %d, world %.2f, view %.1f, crawl x%.2f, Score lives %d, targets:%s",
			t, level.maptime, why, pmo.health, pmo.GetMaxHealth(true), ar ? ar.Amount : 0, d.downOn[n], d.clk[n] / 35.0, d.pool[n] / 35.0, d.Elapsed(n) / 35.0, d.owed[n],
			w ? w.downs : -1, d.bankHp[n], d.bankAr[n], d.Invul(n), pmo.bBUDDHA, c.scale, p.viewheight, d.Crawl(n), sc ? sc.GetInt("lives", "", 0, 0, pmo) : -9, tg);
	}
	void God(bool on)
	{
		let p = players[consoleplayer];
		if (on) p.cheats |= CF_GODMODE; else p.cheats &= ~CF_GODMODE;
	}
	void Hit(int t, Actor src, int hp, int dmg, String why)
	{
		let pmo = players[consoleplayer].mo;
		if (hp > 0) pmo.A_SetHealth(hp);
		God(false);
		Console.Printf("T_HIT t%d %s: health %d, %d from %s", t, why, pmo.health, dmg, src ? String.Format("%s", src.GetClassName()) : "nothing");
		pmo.DamageMobj(src, src, dmg, 'None');
		Console.Printf("T_AFTER t%d health %d", t, pmo.health);
		God(true);
	}
	void Shoot(int t, Actor a, int dmg, String why)
	{
		let pmo = players[consoleplayer].mo;
		if (!a) { Console.Printf("T_SHOOT t%d %s: none", t, why); return; }
		int was = a.health;
		a.DamageMobj(pmo, pmo, dmg, 'None');
		Console.Printf("T_SHOOT t%d %s: %s health %d -> %d (%d asked)", t, why, a.GetClassName(), was, a.health, dmg);
	}
	Actor Target(int k, bool spawnedOnly)
	{
		let d = RSQSM_Down.Get();
		int j = 0;
		for (int i = 0; i < d.tMon.Size(); i++)
		{
			if (d.tPn[i] != consoleplayer || d.tDead[i] != 0 || !d.tMon[i]) continue;
			if (spawnedOnly && (d.tSpawned[i] == 0)) continue;
			if (j == k) return d.tMon[i];
			j++;
		}
		return null;
	}
	void Inv(int t, String why)
	{
		let pmo = players[consoleplayer].mo;
		static const String names[] = { "RifleAmmo", "Clip2", "BDPistolAmmo", "Clip1", "ShotgunAmmo", "AmmoShell", "RSQTR_CigPack", "GrenadeAmmo", "RSQGN_Ammo" };
		String s = "";
		for (int i = 0; i < names.Size(); i++)
		{
			Name nm = names[i];
			class<Inventory> ic = nm;
			let it = ic ? pmo.FindInventory(ic) : null;
			s = s .. String.Format(" %s=%s", names[i], it ? String.Format("%d/%d", it.Amount, it.MaxAmount) : "-");
		}
		let p = players[consoleplayer];
		Console.Printf("T_INV t%d %s:%s; ready %s offhand %s", t, why, s, p.ReadyWeapon ? String.Format("%s", p.ReadyWeapon.GetClassName()) : "-",
			p.OffhandWeapon ? String.Format("%s", p.OffhandWeapon.GetClassName()) : "-");
	}
	void SetAmmo(String n, int v) { let pmo = players[consoleplayer].mo; Name nm = n; class<Inventory> c = nm; if (c) pmo.SetInventory(c, v); }
	// the view is the rig's head (its yaw may be put back each frame), so the MONSTER is moved round the player instead
	void Place(int t, Actor a, double ang, double dist, String why)
	{
		let pmo = players[consoleplayer].mo;
		if (!a) { Console.Printf("T_PLACE t%d %s: none", t, why); return; }
		double aa = pmo.angle + ang;
		Vector2 xy = pmo.pos.xy + (cos(aa), sin(aa)) * dist;
		a.SetOrigin((xy, pmo.floorz), false);
		a.angle = a.AngleTo(pmo);
		Console.Printf("T_PLACE t%d %s: %s moved to %.0f degrees off the view, %.0f units (fits %d)", t, why, a.GetClassName(), ang, dist, a.TestMobjLocation());
	}

	ui int frames;
	override void RenderOverlay(RenderEvent e)
	{
		frames++;
		if (Mode() != 0) Console.Printf("T_FRAME %d maptime %d", frames, level.maptime);
	}
	override void WorldTick()
	{
		let p = players[consoleplayer];
		if (!p || !p.mo) return;
		let pmo = p.mo;
		if (Mode() == 0) return;
		if (!started) { started = true; t0 = level.maptime; Console.Printf("T_START %s maptime %d", level.mapname, t0); }
		int t = level.maptime - t0 + 1;
		Vector2 fwd = (cos(pmo.angle), sin(pmo.angle));
		Vector2 side = (fwd.y, -fwd.x);
		if (t == 5)
		{
			God(true);
			SetI("rsqsm_debug", 1); SetI("rs_score_debug", 1); SetI("rsqgb_kr_debug", 1); SetI("rsqmb_debug", 1); SetI("gy_bones_debug", 1); SetI("gy_bones", 1);
			SetI("rsqsm_down", 1); SetI("rsqsm_down_redmap", 1); SetI("rsqsm_down_lowview", 1); SetI("rsqsm_down_marker", 2);
			SetI("rsqgb_kr_enabled", 1); SetI("rsqgb_kr_chance", 100); SetI("rsqgb_kr_magnet", 0);
			SetI("rsqtr_cig", 1); SetI("rsqthp_dmg", 1);
		}
		static const double SD[] = { 150, 260, 200, 170, 230, 230 };
		static const double SL[] = { -40, 60, 40, 0, -50, 50 };
		if (t == 8)
			for (int i = 0; i < 6; i++) Actor.Spawn(i == 1 ? "DoomImp" : "ZombieMan", (pmo.pos.xy + fwd * SD[i] + side * SL[i], pmo.floorz), ALLOW_REPLACE);
		if (t == 30)
		{
			Console.Printf("T_PLAYER at %.0f %.0f angle %.0f", pmo.pos.x, pmo.pos.y, pmo.angle);
			for (int i = 0; i < 6; i++)
			{
				let a = Near((pmo.pos.xy + fwd * SD[i] + side * SL[i], 0));
				z.Push(a);
				if (a) { a.A_ClearTarget(); a.bFRIENDLY = false; }
				Console.Printf("T_SPAWN monster %d: %s health %d", i, a ? String.Format("%s", a.GetClassName()) : "none", a ? a.health : 0);
			}
		}
		if (z.Size() < 6) return;
		if (t == 34)
		{
			SetAmmo("RifleAmmo", 5); SetAmmo("BDPistolAmmo", 2); SetAmmo("RSQTR_CigPack", 1); SetAmmo("GrenadeAmmo", 1);
			Inv(t, "before DOWN (magazines low, 1 cigarette, 1 grenade)");
			Report(t, "before");
		}
		// ---- A: top tier
		if (t == 40) { Hit(t, z[0], 40, 100, "A: a lethal hit by z0"); Report(t, "A: after the lethal hit"); }
		if (t == 50) { Shoot(t, z[1], 20, "A: z1 hit (an Imp: time bought)"); Report(t, "A: after a hit"); }
		if (t == 58) { Hit(t, z[3], 0, 80, "A: a monster hit while DOWN"); Report(t, "A: held"); }
		if (t == 62) { Shoot(t, z[1], 5000, "A: z1 killed (not a target: its bits bank)"); }
		if (t == 76) Report(t, "A: bits banked?");
		// test only: the top tier's Revival Fire blast (384 at 256 units) would kill the test monsters the later steps need
		if (t == 78) for (int i = 2; i < 6; i++) if (z[i]) z[i].bINVULNERABLE = true;
		if (t == 80) { Shoot(t, z[0], 5000, "A: the target killed"); }
		if (t == 95) for (int i = 2; i < 6; i++) if (z[i]) z[i].bINVULNERABLE = false;
		if (t == 82) { Report(t, "A: UP (top)"); Inv(t, "after UP (top)"); }
		if (t == 100) Report(t, "A: invulnerable");
		if (t == 110) { Hit(t, z[3], 0, 30, "A: a hit while invulnerable"); }
		// ---- B: middle tier, two owed
		if (t == 270) { Hit(t, z[2], 30, 100, "B: a lethal hit by z2 (DOWN 2)"); Report(t, "B: after the lethal hit"); }
		if (t == 290) Report(t, "B: two targets");
		if (t == 340) { Shoot(t, Target(0, true), 5000, "B: the spawned target killed"); Report(t, "B: one left"); }
		if (t == 380) Report(t, "B: the tall step broke");
		if (t == 430) { Shoot(t, z[2], 5000, "B: z2 killed"); }
		if (t == 432) { Report(t, "B: UP (middle)"); Inv(t, "after UP (middle)"); }
		// ---- C: bottom tier, three owed, the killer behind
		if (t == 540) Place(t, z[3], 180, 170, "C: z3 behind the player");
		if (t == 545) { Hit(t, z[3], 30, 100, "C: a lethal hit by z3, now behind (DOWN 3)"); Report(t, "C: after the lethal hit"); }
		if (t == 560) Report(t, "C: z3 behind, two spawned");
		if (t == 600) Place(t, z[3], 90, 170, "C: z3 to the player's left");
		if (t == 640) Place(t, z[3], 0, 170, "C: z3 in front (in view)");
		if (t == 700) { Shoot(t, Target(0, true), 5000, "C: a spawned target killed"); }
		if (t == 720) { Shoot(t, Target(0, true), 5000, "C: the other spawned target killed"); Report(t, "C: z3 left"); }
		if (t == 775) { Shoot(t, z[3], 5000, "C: z3 killed"); }
		if (t == 777) { Report(t, "C: UP (bottom)"); Inv(t, "after UP (bottom)"); }
		// ---- D: fail with a Score life, the marker styles
		if (t == 900) { let sc = Svc("RS_ScoreService"); Console.Printf("T_LIFE t%d givelife -> %d", t, sc ? sc.GetInt("givelife", "", 1, 0, pmo) : -9); }
		if (t == 905) { Hit(t, z[4], 30, 100, "D: a lethal hit by z4 (DOWN 4), one Score life held"); Report(t, "D: after the lethal hit"); }
		if (t == 930) Report(t, "D: four targets");
		if (t == 960) { Place(t, z[4], -100, 200, "D: z4 to the right"); Place(t, Target(0, true), 140, 220, "D: spawned 1 behind-left"); Place(t, Target(1, true), 75, 220, "D: spawned 2 left"); Place(t, Target(2, true), 180, 220, "D: spawned 3 behind"); }
		if (t == 965) { SetI("rsqsm_down_marker", 0); Console.Printf("T_MARKER t%d A Framed Triangle", t); }
		if (t == 990) { SetI("rsqsm_down_marker", 1); Console.Printf("T_MARKER t%d B Framed Fat Arrow", t); }
		if (t == 1015) { SetI("rsqsm_down_marker", 3); Console.Printf("T_MARKER t%d D Triangle With Afterimages", t); }
		if (t == 1040) { SetI("rsqsm_down_marker", 2); Console.Printf("T_MARKER t%d C Triple Chevron", t); }
		if (t == 1100) { Place(t, Target(0, true), 0, 260, "D: spawned 1 back in view"); }
		if (t == 1200 || t == 1230) Report(t, "D: the clock");
		if (t == 1240) Report(t, "D: after the fail");
		// ---- E: the Bossfight's life crate
		if (t == 1260)
		{
			let h = RSQMB_Handler.Get();
			let b = RSQMB_Boss(new("RSQMB_Boss"));
			b.mb = z[5];
			if (h && b.mb) { Console.Printf("T_BOSS t%d a Bossfight payoff for %s (Miniboss's BfPayoff)", t, b.mb.GetClassName()); h.BfPayoff(b, pmo); }
			else Console.Printf("T_BOSS t%d no Miniboss handler", t);
			b.Destroy();
		}
		if (t == 1262)
		{
			let it = ThinkerIterator.Create("RSQ_ScoreLifeCrate");
			crate = Actor(it.Next());
			if (crate)
			{
				Console.Printf("T_CRATE t%d at %.0f %.0f %.0f translation %d", t, crate.pos.x, crate.pos.y, crate.pos.z, crate.Translation);
				let pol = RSQDG_Policy.Get();
				if (pol) { let r = pol.Decide(crate, pmo, p); Console.Printf("T_GRAB t%d Distance Grab's rule for the crate: %s (%s)", t, r ? "ALLOWED" : "refused", r ? r.why : pol.WhyNot(crate, pmo, p)); }
				crate.SetOrigin((pmo.pos.xy + fwd * 80, pmo.floorz), false);
				crate.vel = (0, 0, 0);
				Console.Printf("T_CRATE t%d moved 80 ahead of the player for the frames (test only)", t);
			}
			else Console.Printf("T_CRATE t%d none", t);
		}
		if (t == 1400 || t == 1500 || t == 1600) Console.Printf("T_CRATE t%d %s alpha %.2f", t, crate ? "there" : "gone", crate ? crate.alpha : 0);
		if (t == 1615) Console.Printf("T_CRATE t%d %s", t, crate ? "STILL THERE" : "gone (10 s)");
		if (t == 1618)
		{
			crate2 = Actor.Spawn("RSQ_ScoreLifeCrate", (pmo.pos.xy + fwd * 60, pmo.floorz), NO_REPLACE);
			let sc = Svc("RS_ScoreService");
			int was = sc ? sc.GetInt("lives", "", 0, 0, pmo) : -9;
			bool took; Actor res;
			if (crate2) [took, res] = Inventory(crate2).CallTryPickup(pmo);
			Console.Printf("T_CRATE2 t%d taken %d: Score lives %d -> %d", t, took, was, sc ? sc.GetInt("lives", "", 0, 0, pmo) : -9);
		}
		// ---- F: Graves Rise without a cap
		if (t == 1640)
		{
			for (int i = 0; i < 30; i++)
			{
				let dth = gy_Death.of((pmo.pos.xy + fwd * (300 + (i / 6) * 40) + side * ((i % 6) * 40 - 100), pmo.floorz), "test stone").setKind(1);
				gy_EventHandler.spawnStone(dth);
			}
			let gh = gy_BonesHandler(EventHandler.Find("gy_BonesHandler"));
			Array<int> kinds; kinds.Push(0);
			int made = gh ? gh.RaiseGraves(pmo.pos, 0, 5, kinds, pmo) : -1;
			Console.Printf("T_GRAVES t%d 30 stones, budget 5: %d risers planned (the old cap was 24), Legion of Bones loaded %d", t, made, gy_Bones.LobLoaded() ? 1 : 0);
		}
		if (t == 1700 || t == 1790)
		{
			let it = ThinkerIterator.Create("Actor");
			Actor a;
			int k = 0;
			while (a = Actor(it.Next())) if (a.bISMONSTER && a.health > 0) { String cn = a.GetClassName(); if (cn.IndexOf("Skull") >= 0 || cn.IndexOf("Skeleton") >= 0 || cn.IndexOf("Undead") >= 0) k++; }
			Console.Printf("T_UNDEAD t%d living Legion of Bones undead (Skull / Skeleton / Undead in the name): %d", t, k);
		}
	}
}
