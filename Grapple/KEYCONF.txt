// RSQ Grapple 2.0: one bind, Fire Hook. No default bind (the owner binds his own keys).
// `+rsqgr_hook` is no engine action, so the key falls through to these aliases (key-up sends `-rsqgr_hook`). The press
// is a LOCAL `event` first (zscript/rsqgr/input.zs): the off hand's pose is read on the machine that holds the
// controller and packed into the network event every machine acts on.
addkeysection "RSQ Grapple" RSQGrapple
addmenukey "Fire Hook" "+rsqgr_hook"
alias "+rsqgr_hook" "event rsqgr_presskey"
alias "-rsqgr_hook" "event rsqgr_releasekey"
