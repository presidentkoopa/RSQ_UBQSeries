// RSQ Grapple 2.0 options (the series' menu pass, owner 11-04): ONE page with the options, ONE "Test and Debug" page
// with the raw number and the debug log; Reset All at the foot covers both (RSQGRResetAll reads the rows).
// What each setting does is in CVARINFO's comments.

// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQGR_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQGRRSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "RSQ Grapple", "RSQGR_Options"
}
// ----------------------------------------------------------------------------------------------------

OptionMenu "RSQGR_Options"
{
	Title "RSQ GRAPPLE"
	// tip: In the air, the hook snaps to the valid target nearest your hook hand's line.
	Option "Aim Assist", "rsqgr_aim_assist", "OnOff"
	// tip: One double jump in any jump or fall until you land, with no slow motion.
	Option "Double Jump Always", "rsqgr_dj_always", "OnOff"
	StaticText " "
	StaticText "Controls", 1
	// tip: Which hand fires the hook.
	Option "Hook Hand", "rsqgr_hand", "RSQGR_HookHand"
	// tip: While the hook is ready, a small dot shows where the claw would bite.
	Option "Aim Dot", "rsqgr_aim_dot", "OnOff"
	// tip: The key or button that fires the hook.
	Control "Fire Hook", "+rsqgr_hook"
	StaticText " "
	// tip: Aim trims, range and the debug log.
	Submenu "Test and Debug", "RSQGR_Test"
	// tip: Puts every RSQ Grapple setting back to its default.
	RSQGRResetAll "Reset All", "Reset all RSQ Grapple settings?"
}

// ---- Test and Debug: aim trims, the raw number and the debug log -----------------------------------
OptionMenu "RSQGR_Test"
{
	Title "RSQ GRAPPLE: TEST AND DEBUG"
	StaticText "Old Pull", 1
	StaticText " "
	StaticText "Aim", 1
	// tip: Turns the claw's line left (+) or right (-) of the hook hand's, in degrees.
	Slider "Aim Trim Sideways", "rsqgr_aim_yaw", -15, 15, 0.5, 1
	// tip: Turns the claw's line up (+) or down (-), in degrees.
	Slider "Aim Trim Vertical", "rsqgr_aim_pitch", -15, 15, 0.5, 1
	// tip: The claw rides the line a gun in the hook hand fires, its roll included.
	Option "Firing Line", "rsqgr_fireline", "OnOff"
	// tip: The firing line uses Brutal Doom VR's bore offset, as its guns and lasers do.
	Option "Gun Bore", "rsqgr_bore", "OnOff"
	StaticText " "
	StaticText "Tuning", 1
	// tip: How far the hook reaches, in map units.
	Slider "Range", "rsqgr_range", 256, 2048, 32, 0
	// tip: Prints every bite, kick and slow-motion event to the console.
	Option "Debug Log", "rsqgr_debug", "OnOff"
}

OptionValue "RSQGR_HookHand"
{
	0, "Off Hand"
	1, "Main Hand"
}
