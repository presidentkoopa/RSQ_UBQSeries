// RSQ Grapple 2.0 options (the series' menu pass, owner 11-04): ONE page with the options, ONE "Test and Debug" page
// with the raw number and the debug log; Reset All at the foot covers both (RSQGRResetAll reads the rows).
// What each setting does is in CVARINFO's comments.

// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQGR_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQGRRSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "RSQ Grapple", "RSQGR_Options"
}
// ----------------------------------------------------------------------------------------------------

OptionMenu "RSQGR_Options"
{
	Title "RSQ GRAPPLE"
	Option "Aim Assist", "rsqgr_aim_assist", "OnOff"
	Option "Scorpion Pull", "rsqgr_scorpion", "OnOff"
	StaticText " "
	StaticText "Controls", 1
	Control "Fire Hook", "+rsqgr_hook"
	StaticText " "
	Submenu "Test and Debug", "RSQGR_Test"
	RSQGRResetAll "Reset All", "Reset all RSQ Grapple settings?"
}

// ---- Test and Debug: the raw number and the debug log ---------------------------------------------
OptionMenu "RSQGR_Test"
{
	Title "RSQ GRAPPLE: TEST AND DEBUG"
	Slider "Range", "rsqgr_range", 256, 2048, 32, 0
	Option "Debug Log", "rsqgr_debug", "OnOff"
}
