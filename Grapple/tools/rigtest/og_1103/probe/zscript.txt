version "4.14"
// TEST ONLY: logs the player's velocity every tic, the tic-to-tic change, and where each claw is (OG HookShot or RSQGR_Hook).
class GRV_Probe : EventHandler
{
	Vector3 pv; bool wasGround; int lastFire;
	Actor mon; Vector3 mpv; Array<Actor> extra;
	static const Name mcls[] = { "DoomImp", "ZombieMan", "Demon", "Cyberdemon", "Cacodemon" };
	override void OnRegister() { SetOrder(1000); }   // after RSQGR_Core: sees this tic's thrust
	override void NetworkProcess(ConsoleEvent e)
	{
		let pmo = players[e.Player].mo; if (!pmo) return;
		if (e.Name == "grv_tp")
		{
			pmo.SetOrigin((e.Args[0], e.Args[1], 0), false); pmo.angle = e.Args[2]; pmo.vel = (0,0,0); pv = (0,0,0);
			Console.Printf("V_TP t%d pos(%.1f %.1f %.1f)", level.maptime, pmo.pos.x, pmo.pos.y, pmo.pos.z);
		}
		else if (e.Name == "grv_sum")
		{
			if (mon) mon.Destroy();
			mon = Actor.Spawn(mcls[clamp(e.Args[0], 0, 4)], (pmo.pos.x + e.Args[1], pmo.pos.y, e.Args[2]));
			if (mon) { mon.angle = 180; mpv = (0,0,0); Console.Printf("V_SUM t%d %s at (%.1f %.1f %.1f) mass %d", level.maptime, mon.GetClassName(), mon.pos.x, mon.pos.y, mon.pos.z, mon.Mass); }
		}
		else if (e.Name == "grv_sumk")
		{
			let m2 = Actor.Spawn(mcls[clamp(e.Args[0], 0, 4)], (pmo.pos.x + e.Args[1], pmo.pos.y, e.Args[2]));
			if (m2) { m2.angle = 180; extra.Push(m2); Console.Printf("V_SUMK t%d %s at (%.1f %.1f)", level.maptime, m2.GetClassName(), m2.pos.x, m2.pos.y); }
		}
		else if (e.Name == "grv_og")
		{
			String cn0 = "HookShooter"; Class<Inventory> c = cn0;
			if (c) { pmo.GiveInventory(c, 1); lastFire = level.maptime; Console.Printf("V_OGFIRE t%d atk(%.1f %.1f %.1f) pos(%.1f %.1f %.1f)", level.maptime, pmo.AttackPos.x, pmo.AttackPos.y, pmo.AttackPos.z, pmo.pos.x, pmo.pos.y, pmo.pos.z); }
			else Console.Printf("V_OGFIRE none");
		}
	}
	override void WorldTick()
	{
		if (CVar.FindCVar("grv_log").GetInt() < 1) return;
		let pmo = players[consoleplayer].mo; if (!pmo) return;
		Vector3 dv = pmo.vel - pv;
		String hk = "";
		let it = ThinkerIterator.Create("Actor");
		Actor a;
		while (a = Actor(it.Next()))
		{
			String cn = a.GetClassName();
			if (cn == "HookShot" || cn == "RSQGR_Hook")
				hk = hk .. String.Format(" %s(%.2f %.2f %.2f st%d)", cn, a.pos.x, a.pos.y, a.pos.z, a.CurState ? a.CurState.Tics : -1);
		}
		bool g = pmo.pos.z <= pmo.floorz + 0.5;
		Console.Printf("V t%d pos(%.2f %.2f %.2f) vel(%.3f %.3f %.3f) dv(%.3f %.3f %.3f) |vxy| %.2f km%d gnd%d%s", level.maptime, pmo.pos.x, pmo.pos.y, pmo.pos.z, pmo.vel.x, pmo.vel.y, pmo.vel.z, dv.x, dv.y, dv.z, pmo.vel.xy.Length(), pmo.player.keepmomentum, g, hk);
		pv = pmo.vel;
		if (mon && !mon.bDestroyed)
		{
			Vector3 md = mon.vel - mpv;
			Console.Printf("M t%d %s pos(%.2f %.2f %.2f) vel(%.3f %.3f %.3f) dv(%.3f %.3f %.3f) |vxy| %.2f d2 %.1f hp %d tics %d blk %s", level.maptime, mon.GetClassName(), mon.pos.x, mon.pos.y, mon.pos.z, mon.vel.x, mon.vel.y, mon.vel.z, md.x, md.y, md.z, mon.vel.xy.Length(), (mon.pos.xy - pmo.pos.xy).Length(), mon.health, mon.tics, mon.BlockingMobj ? mon.BlockingMobj.GetClassName() .. String.Format("(solid%d hp%d)", mon.BlockingMobj.bSolid, mon.BlockingMobj.health) : "-");
			mpv = mon.vel;
		}
		for (int k = 0; k < extra.Size(); k++)
			if (extra[k] && !extra[k].bDestroyed)
				Console.Printf("MX t%d %d %s pos(%.2f %.2f) |vxy| %.2f hp %d tics %d", level.maptime, k, extra[k].GetClassName(), extra[k].pos.x, extra[k].pos.y, extra[k].vel.xy.Length(), extra[k].health, extra[k].tics);
	}
}
