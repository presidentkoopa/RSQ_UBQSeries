version "4.14"
class GRP_Probe : EventHandler
{
	static const Name cls[] = { "ZombieMan", "DoomImp", "Demon", "Cacodemon", "BaronOfHell", "Cyberdemon", "ExplosiveBarrel", "Revenant", "Fatso", "LostSoul", "Arachnotron", "ShotgunGuy" };
	double lastz; int airT;
	int stick;
	int pushT; double pushX;   // QoL tests: the pawn pushed east at pushX u/tic for pushT tics (a single vel is wiped on the ground)   // QoL GR1 (2026-10-31): a forwardmove written into the player's cmd every tic before the rope reads it
	override void OnRegister() { SetOrder(-1000); }   // before RSQGR_Core, so the rope sees this tic's stick
	override void NetworkProcess(ConsoleEvent e)
	{
		let pmo = players[e.Player].mo; if (!pmo) return;
		if (e.Name == "grp_push") { pushX = e.Args[0]; pushT = e.Args[1]; return; }
		if (e.Name == "grp_stick") { stick = e.Args[0]; Console.Printf("T_STICK t%d forwardmove %d", level.maptime, stick); return; }
		if (e.Name == "grp_vel")
		{
			pmo.vel = (e.Args[0], e.Args[1], e.Args[2]);
			Console.Printf("T_VEL t%d vel(%.1f %.1f %.1f)", level.maptime, pmo.vel.x, pmo.vel.y, pmo.vel.z);
		}
		else if (e.Name == "grp_tp")
		{
			pmo.SetOrigin((e.Args[0], e.Args[1], pmo.floorz), false);
			pmo.SetOrigin((e.Args[0], e.Args[1], pmo.CurSector.floorplane.ZatPoint((e.Args[0], e.Args[1]))), false);
			pmo.angle = e.Args[2]; pmo.vel = (0,0,0);
			Console.Printf("T_TP t%d pos(%.1f %.1f %.1f) ang %.1f ceil %.1f", level.maptime, pmo.pos.x, pmo.pos.y, pmo.pos.z, pmo.angle, pmo.ceilingz);
		}
		else if (e.Name == "grp_summon")
		{
			int i = clamp(e.Args[0], 0, cls.Size() - 1);
			Vector2 f = Actor.AngleToVector(pmo.angle + e.Args[2], e.Args[1]);
			let a = Actor.Spawn(cls[i], (pmo.pos.xy + f, pmo.pos.z));
			if (a) { a.angle = a.AngleTo(pmo); Console.Printf("T_SUM t%d %s at (%.1f %.1f %.1f) mass %d boss %d", level.maptime, a.GetClassName(), a.pos.x, a.pos.y, a.pos.z, a.Mass, a.bBOSS); }
		}
		else if (e.Name == "grp_dir")
		{
			for (int h = 0; h < 2; h++)
			{
				double yaw = RSQGR_Util.HandYaw(pmo, h), pit = RSQGR_Util.HandPitch(pmo, h);
				Vector3 my = RSQGR_Util.MapDir(yaw, pit);
				Vector3 eng = (0,0,0);
				if (pmo.OverrideAttackPosDir) eng = h == 0 ? pmo.AttackDir(pmo, pmo.Angle, pmo.Pitch) : pmo.OffhandDir(pmo, pmo.Angle, pmo.Pitch);
				Vector3 hp = RSQGR_Util.HandPos(pmo, h);
				Console.Printf("T_DIR t%d h%d ovr%d yaw %.2f pit %.2f roll %.2f my(%.3f %.3f %.3f) eng(%.3f %.3f %.3f) hand-off(%.1f %.1f %.1f) pang %.1f", level.maptime, h, pmo.OverrideAttackPosDir, yaw, pit, h == 0 ? pmo.AttackRoll : pmo.OffhandRoll, my.x, my.y, my.z, eng.x, eng.y, eng.z, hp.x - pmo.pos.x, hp.y - pmo.pos.y, hp.z - pmo.pos.z, pmo.angle);
			}
		}
		else if (e.Name == "grp_ceil")
		{
			// Ceiling_LowerByValue (tag, speed, value) / Ceiling_RaiseByValue
			level.ExecuteSpecial(e.Args[0] ? 41 : 40, pmo, null, false, 2, 8, e.Args[1]);
			Console.Printf("T_CEIL t%d %s by %d", level.maptime, e.Args[0] ? "raise" : "lower", e.Args[1]);
		}
		else if (e.Name == "grp_hurt")
		{
			pmo.DamageMobj(null, null, 10000, 'None');
		}
	}
	override void RenderOverlay(RenderEvent e)
	{
		if (CVar.FindCVar("grp_log").GetInt() != 4) return;
		let pmo = players[consoleplayer].mo; if (!pmo) return;
		Console.Printf("T_RO g%d frac %.2f atk.z %.2f view.z %.2f pmo.z %.2f viewz %.2f  atk-view %.2f atk-pmo %.2f atkxy-pmo %.2f %.2f", gametic, e.FracTic, pmo.AttackPos.z, e.ViewPos.z, pmo.pos.z, pmo.player.viewz, pmo.AttackPos.z - e.ViewPos.z, pmo.AttackPos.z - pmo.pos.z, pmo.AttackPos.x - pmo.pos.x, pmo.AttackPos.y - pmo.pos.y);
	}
	override void WorldTick()
	{
		if (pushT > 0 && players[consoleplayer].mo) { pushT--; players[consoleplayer].mo.vel.x = pushX; }
		if (stick != 0 && players[consoleplayer].mo) players[consoleplayer].cmd.forwardmove = stick;
		if (CVar.FindCVar("grp_log").GetInt() == 6 && level.maptime % 5 == 0)
		{
			let it = level.CreateSectorTagIterator(2);
			int si = it.Next();
			if (si >= 0) Console.Printf("T_ALCOVE t%d ceiling %.1f", level.maptime, level.sectors[si].ceilingplane.ZatPoint((320, 1100)));
		}
		if (CVar.FindCVar("grp_log").GetInt() == 6) return;
		if (CVar.FindCVar("grp_log").GetInt() < 1) return;
		let pmo = players[consoleplayer].mo; if (!pmo) return;
		if (CVar.FindCVar("grp_log").GetInt() == 5)
		{
			Console.Printf("T_KM t%d pos(%.1f %.1f %.1f) vel(%.2f %.2f %.2f) keepmomentum %d speed %.2f onground %d", level.maptime, pmo.pos.x, pmo.pos.y, pmo.pos.z, pmo.vel.x, pmo.vel.y, pmo.vel.z, pmo.player.keepmomentum, pmo.Speed, pmo.player.onground);
			return;
		}
		if (CVar.FindCVar("grp_log").GetInt() >= 2)
		{
			let it = ThinkerIterator.Create("Actor");
			Actor a;
			while (a = Actor(it.Next()))
				if (a.bISMONSTER && a.Health > 0)
					Console.Printf("T_M t%d %s pos(%.1f %.1f %.1f) vel(%.2f %.2f %.2f) tics %d", level.maptime, a.GetClassName(), a.pos.x, a.pos.y, a.pos.z, a.vel.x, a.vel.y, a.vel.z, a.tics);
		}
		if (CVar.FindCVar("grp_log").GetInt() == 3)
		{
			let it = ThinkerIterator.Create("Actor");
			Actor a; int n = 0; double mn = 1e9, mx = 0; Actor hk;
			while (a = Actor(it.Next()))
			{
				String cn = a.GetClassName();
				if (cn == "LoreShot2" || cn == "RSQGR_Link")
				{
					n++; double d = (a.pos - pmo.AttackPos).Length(); mn = min(mn, d); mx = max(mx, d);
				}
				if (cn == "HookShot") hk = a;
			}
			Console.Printf("T_TR t%d links %d nearest %.1f farthest %.1f hook %s", level.maptime, n, mn, mx, hk ? String.Format("%.1f", (hk.pos - pmo.AttackPos).Length()) : "-");
		}
		Console.Printf("T_P t%d pos(%.1f %.1f %.1f) vel(%.2f %.2f %.2f) fz %.1f cz %.1f hp %d h %.1f vh %.1f", level.maptime, pmo.pos.x, pmo.pos.y, pmo.pos.z, pmo.vel.x, pmo.vel.y, pmo.vel.z, pmo.floorz, pmo.ceilingz, pmo.health, pmo.height, pmo.player.viewheight);
	}
}
