// TEST ONLY, never ships (2026-11-03): needs Gun Bonsai + RSQ_GunBonsaiVR. "gbp_offers" prints, for every RSQ hook perk
// Gun Bonsai has registered, whether it would be offered now (IsSuitableForPlayer, what GenerateUpgradesForPlayer asks).
class GBP_Probe : EventHandler
{
	override void NetworkProcess(ConsoleEvent e)
	{
		let pmo = players[e.Player].mo; if (!pmo) return;
		if (e.Name == "gbp_weapon") { Console.Printf("GBP t%d ready %s", level.maptime, pmo.player.ReadyWeapon ? pmo.player.ReadyWeapon.GetClassName() : 'none'); return; }
		if (e.Name != "gbp_offers") return;
		let reg = TFLV_Upgrade_Registry.GetRegistry();
		let stats = TFLV_PerPlayerStats.GetStatsFor(pmo);
		if (!reg || !stats) { Console.Printf("GBP no registry/stats"); return; }
		String yes = "", no = "";
		for (uint i = 0; i < reg.upgrades.Size(); i++)
		{
			let u = reg.upgrades[i];
			if (!(u is "RSQ_HookPerk")) continue;
			String n = u.GetClassName(); n.Replace("RSQ_Upgrade_", "");
			if (u.IsSuitableForPlayer(stats)) yes = yes .. " " .. n; else no = no .. " " .. n;
		}
		Console.Printf("GBP t%d offered:%s | not:%s", level.maptime, yes, no);
	}
}
