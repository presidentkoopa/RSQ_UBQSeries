// RSQ Top HP Bars. Server cvars, like the rest of the series (single-player VR; the bars are play-scope
// actors, so every machine must see the same values). Defaults are coldcite's (size 0.3, height 0.1, not full bright, stealth followed).
server bool   rsqthp_on       = true;   // bars on/off
server int    rsqthp_scheme   = 0;      // 0 Classic (coldcite's A), 1 Red (coldcite's B), 2 Protanopia, 3 Deuteranopia,
                                      // 4 Tritanopia, 5 High Contrast. RSQ_Accessibility's rsqa_colorblind, when
                                      // loaded and not Off, overrides this.
server float  rsqthp_scale    = 0.3;    // bar size (coldcite's Scale)
server float  rsqthp_far      = 2.0;    // size multiplier reached at range: 1 = off; grows from 256 to 1024 units
server float  rsqthp_offset   = 0.1;    // height above the sprite, as a fraction of its height (coldcite's)
server bool   rsqthp_bright   = false;  // ignore sector light (High Contrast always does)
server bool   rsqthp_stealth  = true;   // follow the monster's translucency (coldcite's Allow Stealth)
server bool   rsqthp_overheal = true;   // strip above the bar for health over spawn health
server bool   rsqthp_markers  = true;   // hooked / stunned / locked / gripped icons from the series' mods, and the frozen /
                                      // held icon right of the bar (owner's picks 36, 37: on with the markers)
server bool   rsqthp_aimed    = false;  // Aimed Only (owner's pick 35, an option): a bar only over the monster a hand's gun
                                      // points at (and half a second after the aim leaves it)
// QoL round (owner's picks, 2026-10-31). HP6-HP9 are always on (bar.zs); these four are options, off by default:
server bool   rsqthp_chunk    = false;  // HP1 Damage Chunk: what your hit took off stays lit half a second, then drains
server bool   rsqthp_ally     = false;  // HP2 Ally Colour: friendly monsters (allies, Brutal Doom's marines) get a blue bar
server bool   rsqthp_nearest  = false;  // HP3 Nearest 24: only the 24 bars nearest you are drawn
server bool   rsqthp_limit    = false;  // HP4 Distance Limit: no bars farther than rsqthp_limitdist
server float  rsqthp_limitdist = 1536;  // HP4: that distance, map units (Tuning)
// Damage Numbers (owner, 2026-11-03): damnums by Xaser Acheron folded in. Its own cvars, renamed, same scopes and defaults
// (dam_enabled, dam_shootable, dam_spray are server; dam_fontclass, dam_translation, dam_physics, dam_usetypes are user).
server bool   rsqthp_dmg         = true;   // Damage Numbers on/off (dam_enabled)
server bool   rsqthp_dmg_all     = false;  // Numbers For: 0 monsters only, 1 all shootable things (dam_shootable; never your own body)
server bool   rsqthp_dmg_spray   = false;  // Shotgun Spray: one number per hit instead of one per target per tic (dam_spray)
user   string rsqthp_dmg_font    = "RSQTHP_DamNumFontClassic"; // Number Font: Classic (damnums' Doom default) or Small (dam_fontclass)
user   string rsqthp_dmg_colour  = "RSQTHP_DamDefault";        // Number Colour: Default = the font's (red) (dam_translation)
user   int    rsqthp_dmg_physics = 0;      // Number Motion: 0 Toss, 1 Float (dam_physics)
user   bool   rsqthp_dmg_types   = true;   // Damage Type Colours: fire orange, plasma cyan, ... (dam_usetypes)
