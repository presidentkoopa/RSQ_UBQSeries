// RSQ Whip -- MENUDEF. The series' menu (MENU_PASS_BRIEF.md, owner 11-04): ONE page with every option under short
// headers, On/Off first, no Control rows (no keys since 11-10); ONE sub menu, "Test and Debug", with the numbers and the
// debug log. Reset All (RSQWPResetAll) resets every row on both pages; KEYCONF's rsqwp_reset alias (tools/gen_reset.py)
// does the same from the console.
// 2026-11-27 (owner: "what i want is to do a big wide swing ... and if i crack down to the ground the impact sends out a
// shockwave", "i want to tone this thing down"): the whip has two moves and neither has a setting (their numbers are
// constants in zscript/rsqwp/lash.zs). Whip Style, Whip Reach, Stun and every row of the crack, the mark, the lock, the
// wrap, the yank, the lasso, the rope, the swinging, the climbing and the grip keys went, with their OptionValue blocks
// (RSQWP_Styles, RSQWP_WhipAims, RSQWP_Knock). Their settings stay in CVARINFO, marked RETIRED.
// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQWP_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQWPRSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "RSQ Whip", "RSQWP_Options"
}

OptionMenu "RSQWP_Options"
{
	Title "RSQ WHIP"
	StaticText "Whip", "Gold"
	// tip: Turns the whip on or off; while off nobody is given it.
	Option "Whip", "rsqwp_on", "OnOff"
	// tip: Every player carries the whip weapon in slot 1.
	Option "Whip in Slot 1", "rsqwp_whip_weapon", "OnOff"
	// tip: Which hand holds the whip.
	Option "Whip In", "rsqwp_whip_hand", "RSQWP_WhipHand"
	// tip: The whip hangs coiled at your hip while you own it and hold it in neither hand.
	Option "Show Whip", "rsqwp_show", "OnOff"
	StaticText " "
	// tip: The hand on the handle, how the thong moves, and the debug log.
	Submenu "Test and Debug", "RSQWP_Test"
	// tip: Puts every RSQ Whip setting back to its default.
	RSQWPResetAll "Reset All", "Reset all RSQ Whip settings?"
}
