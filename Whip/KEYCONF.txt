// The whip weapon in slot 1 (Weapon.SlotNumber is ignored when a player class -- Brutal Doom's -- lists its own slots;
// addslotdefault appends, so Fist and Chainsaw stay, as RSQ ShieldSaw does it)
addslotdefault 1 RSQWP_WhipWeapon

// Two binds, one per hand, as their own commands -- never +user1-4 (Brutal Doom Quest uses all four).
// `+name` falls through to alias lookup and key-up sends `-name`: a press and a release. The press goes
// through a local `event` first so the hand's pose is read on the machine that owns it (input.zs).
// No default controller binds: the owner binds them himself.
addkeysection "RSQ Whip" RSQWhip
addmenukey "Whip (Main)" "+rsqwp_main"
addmenukey "Whip (Off)"  "+rsqwp_off"
addmenukey "Grab Whip (Main)" "+rsqwp_grip_main"
addmenukey "Grab Whip (Off)"  "+rsqwp_grip_off"
alias "+rsqwp_main" "event rsqwp_press 0"
alias "-rsqwp_main" "event rsqwp_release 0"
alias "+rsqwp_off"  "event rsqwp_press 1"
alias "-rsqwp_off"  "event rsqwp_release 1"
alias "+rsqwp_grip_main" "event rsqwp_grip 0"
alias "-rsqwp_grip_main" "event rsqwp_ungrip 0"
alias "+rsqwp_grip_off"  "event rsqwp_grip 1"
alias "-rsqwp_grip_off"  "event rsqwp_ungrip 1"
