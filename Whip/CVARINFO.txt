// RSQ_Whip settings. All SERVER cvars (the host's world).

server bool   rsqwp_on            = true;   // the whip is available on its binds
server int    rsqwp_whip_mode     = 0;      // 0 Real: hold the bind, the whip is out and you crack it; 1 Assist: a press throws a lash for you
server int    rsqwp_whip_aim      = 1;      // 0 Free: the crack hits what the tip touches; 1 Assist: what is nearest the tip; 2 Lock: other hand points, alt-fire holds, the crack flies to it
server float  rsqwp_whip_reach    = 128;    // the whip's length, room units from the handle (128 = 3.8 m, a long bullwhip)
server float  rsqwp_whip_crack    = 20;     // how fast the tip must go to crack, room units a tic (lower = easier)
server float  rsqwp_whip_assist   = 40;     // Assist aim's reach round the crack, map units
server int    rsqwp_whip_damage   = 12;     // a crack's damage (a sting does part of it)
server bool   rsqwp_whip_stun     = true;   // a crack knocks a monster into its Pain state (breaks off its attack)
server bool   rsqwp_whip_snag     = true;   // a lash that catches an edge (beam, lintel, ledge lip) wraps round it: a rope
server float  rsqwp_whip_above    = 16;     // how far above the hand an edge must be to snag
server bool   rsqwp_whip_lines    = true;   // the crack throws a switch (any use line) it lands on
server bool   rsqwp_whip_fetch    = true;   // the crack flicks a loose pickup near it to you
server bool   rsqwp_whip_leash    = true;   // Lock: a locked crack on a monster with the bind still held catches it and pulls it in
server bool   rsqwp_whip_weapon   = true;   // every player carries the whip weapon (slot 1)
server float  rsqwp_whip_prio     = 256;    // the whip weapon: switches, keys and powerups this close (map units, 256 = 8 m) are marked first
server float  rsqwp_whip_light    = 400;    // the whip weapon's pull: up to this mass a monster is pulled off its feet to you
server float  rsqwp_whip_heavy    = 1000;   // from this mass (and bosses) it cannot be pulled: the yank swings you to it
server bool   rsqwp_whip_lasso    = true;   // the whip weapon: grab it with the off hand (two-hand grip, SteamVR) and whirl it as a lasso
server noarchive int rsqwp_lasso_test = 0;  // test only: 1 acts as if the off hand held the two-hand grip (the rig has no SteamVR latch)

// on the rope (a snag)
server float  rsqwp_reel_speed    = 12;     // holding the bind takes in this much rope per tic
server float  rsqwp_min_len       = 32;     // shortest rope (reeled all the way in)
server float  rsqwp_max_speed     = 30;     // speed cap on the rope
server float  rsqwp_pump          = 0.25;   // how hard the movement stick leans a swing
server float  rsqwp_arm_pump      = 0.12;   // hauling down on the whip in the air adds this x the haul to the swing; 0 off
server float  rsqwp_drag          = 0.002;  // air drag on the swing, share of its speed lost per tic
server float  rsqwp_rope_friction = 0.025;  // and a constant loss (u/tic per tic) that brings it to rest
server float  rsqwp_attach        = 0.6;    // where the rope meets the body, fraction of height
server float  rsqwp_hop           = 6;      // lift when letting go at the top of a wall
server bool   rsqwp_mantle        = true;   // reel up to a ledge lip, or lay a free hand on it with Grab: up and over
server bool   rsqwp_carry         = true;   // keep the swing's speed on landing (player.keepmomentum)
server float  rsqwp_rope_width    = 1.0;    // the thong's thickness (radius, map units)
server int    rsqwp_rope_segs     = 12;     // most pieces a slack line is drawn in

// climbing hand over hand
server int    rsqwp_climb           = 1;     // 1: a free hand on Grab Rope climbs the whip, 0 off
server bool   rsqwp_climb_free_only = true;  // only an empty hand (fist) may take hold, never a gun hand
server float  rsqwp_grab_reach      = 16;    // how near the rope the hand must be to take hold (map units)
server float  rsqwp_climb_cap       = 8;     // most the body moves per tic by the hand (u/tic); faster slips
server bool   rsqwp_grab_dg         = true;  // RSQ_DistanceGrab's grab binds take hold of the rope too

// a leashed monster
server float  rsqwp_mass_max      = 1000;   // heavier than this (or a boss) = an anchor, it does not come
server float  rsqwp_slide_speed   = 10;     // a leashed monster slides to you this fast (scaled by slow motion)
server float  rsqwp_slide_gap     = 16;     // and stops this far from you
server float  rsqwp_retract_speed = 64;     // a let-go line coming back to the hand

server int    rsqwp_debug         = 0;      // 1: log events (T_WP lines), 2: every tic, 3: cost (T_WPWMS)
