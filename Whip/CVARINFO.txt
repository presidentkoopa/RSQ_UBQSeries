server float  rsqwp_whip_above    = 16;     // how far above the hand an edge must be to snag

Replaced rsqwp_swing_reach (9 m) and rsqwp_whip_prio (7.5 m); monsters were 15 m (+33%)
server bool   rsqwp_swing_ohfire  = true;   // Aim With Off-Hand Fire: whip in the main hand, off hand empty or a fist -- holding off-hand fire aims the ray (swing.zs)
server bool   rsqwp_swing_beamwins = true;  // Beam Wins Over Lock-On: while the off-hand beam is held (or a gold spot let go of within 1 s), the crack goes there and nothing else -- no mark, lock-on, edge snag or Assist reach (swing.zs)

server bool   rsqwp_swing_grip     = true;  // Off-Hand Grip Starts Swing: snagged from the ground, an empty off hand closing on the whip takes the rope up and starts the swing
server bool   rsqwp_swing_jump     = true;  // Jump Starts Swing: snagged, a jump lifts you and pushes you on where you face, keeping all your speed (once a rope)
server float  rsqwp_swing_jumplift = 10;    // Jump Lift: the jump start's upward speed, map units a tic (the engine's jump is 8; never stacked on it)

server bool   rsqwp_whip_lines    = true;   // the crack throws a switch (any use line) it lands on
server bool   rsqwp_whip_fetch    = true;   // the crack flicks a loose pickup near it to you
server bool   rsqwp_whip_leash    = true;   // Lock: a locked crack on a monster with the bind still held catches it and pulls it in
server bool   rsqwp_whip_weapon   = true;   // every player carries the whip weapon (slot 1)
user   int    rsqwp_whip_hand     = 0;      // Whip In: 0 the main hand, 1 the off hand (one whip, in that hand)

user   bool   rsqwp_show          = true;   // Show Whip (owner 11-04): the whip, while you own it and hold it in neither hand, hangs coiled at the main-hand hip: drawn by the Rig at its whip holster when the Rig is loaded (it reads this by name), else by Whip itself (hip.zs) at the same spot
server noarchive bool rsqwp_handtest = false; // test only: the world handle drawn beside the hand's (to line them up on the rig)
server float  rsqwp_whip_light    = 400;    // the whip weapon's pull: up to this mass a monster is pulled off its feet to you
server float  rsqwp_whip_heavy    = 1000;   // from this mass (and bosses) it cannot be pulled: the yank swings you to it
server bool   rsqwp_whip_lasso    = true;   // the whip weapon: grab it with the off hand (two-hand grip, SteamVR) and whirl it as a lasso
server noarchive int rsqwp_lasso_test = 0;  // test only: 1 acts as if the off hand held the two-hand grip (the rig has no SteamVR latch)

server bool   rsqwp_whip_crackknock = false; // option (owner 11-03): a crack that doesn't kill knocks the monster about -- the engine's damage thrust, and a crack's wrap swung round with your arm (flail). Off: it stays put; a pull back still yanks it, a thrown lasso still flails
server bool   rsqwp_whip_flail    = true;   // a wrapped or lassoed monster is held at the rope's length, swings with your arm and smashes into others
server bool   rsqwp_whip_fireball = false;  // option: a crack whose lash passes through a monster's missile explodes it there
server bool   rsqwp_whip_hitstop  = false;  // option: a crack freezes the monster it hits for a beat (3-6 tics, harder = longer)

server bool   rsqwp_whip_yankgrace = false; // option: a wrap cannot be yanked for 12 tics (~1/3 s), so a crack's follow-through does not yank it at once
server int    rsqwp_whip_fireball_knock = 0; // Crack Fireballs: 0 the missile explodes where it is cracked, 1 it is knocked away along the crack and explodes away from you
// the thong's physics (11-02, on the MIT reference whip: whip.zs)
server float  rsqwp_whip_bend     = 1.0;    // bend stiffness, x the reference whip's leather (1 = the reference)
