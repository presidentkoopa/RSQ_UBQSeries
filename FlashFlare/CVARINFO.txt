// RSQ FlashFlare -- settings. What each does (the menu carries no notes). Server: what the flares do for everyone (light,
// burn, ammo). User: how YOUR hands are read and where your pocket is.
server bool   rsqff_on          = true;   // the whole mod
server int    rsqff_radius      = 140;    // the light's radius, map units (Quest preset 140, PCVR 220)
server int    rsqff_cap         = 3;      // at most this many lit flares give light at once; the oldest goes dark (Quest 3, PCVR 6)
server int    rsqff_flicker     = 4;      // tics between flicker steps: the light's size and colour change this often (Quest 4, PCVR 3)
server bool   rsqff_subdued     = false;  // Subdued: double radius, half brightness (the soft look)
server int    rsqff_burn        = 60;     // seconds a flare burns (saegiru's burned 180)
server bool   rsqff_sparks      = true;   // sparks off the burning end
server bool   rsqff_smoke       = false;  // a thin smoke trail
server bool   rsqff_sound       = true;   // the strike and the burning crackle (stock Doom sounds)
server int    rsqff_start       = 4;      // flares you start a game with (and after dying)
server bool   rsqff_scatter     = true;   // boxes of 6 lie beside a map's items (saegiru's / the UAC Survival Pack's way)
server int    rsqff_boxes       = 2;      // ...at most this many boxes a map
server bool   rsqff_pocket      = false;  // the pocket is topped up to rsqff_pocket_n at each new level
server int    rsqff_pocket_n    = 4;      // ...to this many
user   int    rsqff_light_by    = 2;      // 0 strike (a sharp flick down of the flare hand), 1 that hand's trigger, 2 either
user   float  rsqff_strike      = 1.4;    // m/s downward the flick must reach to strike it
user   float  rsqff_swing       = 2.0;    // m/s: grip or trigger with the hand moving faster than this throws a lit flare
user   float  rsqff_throw       = 1.25;   // throw strength: x the hand's own speed
user   float  rsqff_lift        = 0.25;   // arc: this much of the throw's speed across the ground added upward
user   bool   rsqff_hand        = true;   // draw the hand holding the flare (Brutal Doom 22 VR's FBX hand)
user   float  rsqff_pocket_f    = 0.12;   // m: the pocket, forward of the chest line (body frame)
user   float  rsqff_pocket_l    = 0.15;   // m: ...toward your MAIN hand's side (the off hand reaches across, as for a chest pocket)
user   float  rsqff_pocket_u    = -0.28;  // m: ...under the eyes
user   float  rsqff_pocket_r    = 0.11;   // m: how near the off hand must be to take or put back
user   string rsqff_keys_off    = "443 436"; // the off-hand grip's key codes (QuestZDoom: 443, or 436 while the main grip shifts)
user   bool   rsqff_debug       = false;  // RSQFF lines in the console / log (the lights, the hands)
