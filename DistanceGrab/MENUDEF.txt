// RSQ Distance Grab options (design/MENU_STANDARD.md, owner 11-24): the front page holds the choices a player makes,
// short labels, On/Off; ONE "Test and Debug" page with the rarer switches, every raw number, the grab keys and the debug
// logs; Reset All at the foot covers both (RSQDGResetAll reads the rows). No presets, no help text: each row's
// "// tip:" comment says what it does. Distance Grab has no Gun Bonsai VR gating (reach.zs: the
// grab has no tree). The grab keys are only read with Grab Keys on (0.99a bug 14): their rows are RSQDGKey rows,
// greyed while it is off (zscript/rsqdg/menurows.zs).
//
// EVERY PAGE MUST BE REACHABLE: an orphaned OptionMenu compiles and loads clean and reads in a headset as the feature
// removed (tools/rsmenu/menutool.py check finds orphans).

// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQDG_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQDGRSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "RSQ Distance Grab", "RSQDG_Options"
}
