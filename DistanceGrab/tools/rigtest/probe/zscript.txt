version "4.12"
// TEST ONLY. Never ships. Observes RSQ_DistanceGrab from outside, by its public doors.
class RSQDGT_GrabEventService : Service
{
	override int GetInt(String request, String stringArg, int intArg, double doubleArg, Object objectArg, Name nameArg)
	{
		if (request ~== "grab.event")
		{
			let a = Actor(objectArg);
			Console.Printf("T_EVENT t%d %s hand=%d air=%d thing=%s from=%s", level.maptime, stringArg, intArg, int(doubleArg),
				a ? a.GetClassName() : 'null', nameArg);
		}
		return 0;
	}
}

class RSQDGT_Probe : EventHandler
{
	int setup;
	Array<Actor> spawned;
	override void WorldThingSpawned(WorldEvent e)
	{
		if (e.Thing && e.Thing.GetClassName() == 'RSQDG_Buzz')
			Console.Printf("T_BUZZ t%d missile=%d offhandflag?  pos=(%.0f %.0f %.0f)", level.maptime, e.Thing.bMissile, e.Thing.pos.x, e.Thing.pos.y, e.Thing.pos.z);
	}
	override void WorldTick()
	{
		let p = players[consoleplayer];
		if (!p || !p.mo) return;
		let mo = p.mo;
		int t = level.maptime;
		String scen = CVar.FindCVar("rsqdgt_scene").GetString();
		// SCRIPTED POSES BY GAME TIC: rsqdgt_poses "tic=x y z p yw r;tic=...". The console's
		// `wait` counts frames, not tics, so it cannot time a gesture.
		String poses = CVar.FindCVar("rsqdgt_poses").GetString();
		if (poses.Length())
		{
			Array<String> ps; poses.Split(ps, ";");
			for (int i = 0; i < ps.Size(); i++)
			{
				Array<String> kv; ps[i].Split(kv, "=");
				if (kv.Size() == 2 && kv[0].ToInt() == t)
				{
					let cv = CVar.FindCVar("qz_rhand");
					if (cv) cv.SetString(kv[1]);
					Console.Printf("T_POSE t%d %s", t, kv[1]);
				}
			}
		}
		if (t == 20 && !setup)
		{
			setup = 1;
			// The headset owns the pawn's facing in VR (A_SetAngle does not stick),
			// so objects are placed along the CURRENT facing; clearance is reported.
			double bestA = mo.angle;
			FLineTraceData d;
			mo.LineTrace(bestA, 2000, 0, TRF_THRUACTORS, 30, 0, 0, d);
			double bestD = d.Distance;
			Console.Printf("T_SETUP heading=%.1f clear=%.0f", bestA, bestD);
			Array<String> items; scen.Split(items, ",");
			for (int i = 0; i < items.Size(); i++)
			{
				Array<String> kv; items[i].Split(kv, ":");
				if (kv.Size() < 3) continue;
				double dist = kv[1].ToDouble(); double side = kv[2].ToDouble();
				Vector2 fwd = (cos(bestA), sin(bestA)); Vector2 rt = (sin(bestA), -cos(bestA));
				Vector2 xy = mo.pos.xy + fwd * dist + rt * side;
				// "!Class": spawned and killed at once -- a corpse
				bool kill = kv[0].Left(1) == "!";
				Name cn = kill ? kv[0].Mid(1) : kv[0];
				let a = Actor.Spawn(cn, (xy.x, xy.y, mo.floorz), ALLOW_REPLACE);
				if (a && kill) a.DamageMobj(null, null, a.Health, 'None', DMG_FORCED);
				if (a) spawned.Push(a);
				if (a) Console.Printf("T_SPAWN %s at dist=%.0f side=%.0f  z=%.0f", kv[0], dist, side, a.pos.z);
			}
		}
		// rsqdgt_shoot "tic:Class": that tic, a missile of Class from 300 units ahead, aimed at what hand 0 holds
		String sh = CVar.FindCVar("rsqdgt_shoot").GetString();
		if (sh.Length())
		{
			Array<String> kv; sh.Split(kv, ":");
			let ho = RSQDG_Hold(EventHandler.Find("RSQDG_Hold"));
			Actor hd = ho ? ho.HeldBy(0) : null;
			if (kv.Size() == 2 && kv[0].ToInt() == t && hd)
			{
				Vector3 c = hd.pos + (0, 0, hd.height * 0.5);
				Vector3 from = (mo.pos.x + cos(mo.angle) * 300, mo.pos.y + sin(mo.angle) * 300, c.z);
				Name cn = kv[1];
				let m = Actor.Spawn(cn, from, ALLOW_REPLACE);
				if (m)
				{
					Vector3 d = c - from;
					m.Vel = d.Unit() * m.Speed;
					m.angle = atan2(d.y, d.x);
					Console.Printf("T_SHOOT t%d %s from (%.0f %.0f %.0f) at held (%.0f %.0f %.0f)", t, m.GetClassName(), from.x, from.y, from.z, c.x, c.y, c.z);
				}
			}
		}
		if (t >= 20 && (t % 5) == 0 || CVar.FindCVar("rsqdgt_every").GetBool())
		{
			let fl = RSQDG_Flight(EventHandler.Find("RSQDG_Flight"));
			let gr = RSQDG_Grab(EventHandler.Find("RSQDG_Grab"));
			let aim = RSQDG_Aim(EventHandler.Find("RSQDG_Aim"));
			let ho = RSQDG_Hold(EventHandler.Find("RSQDG_Hold"));
			Actor tg = aim ? aim.TargetFor(0) : null;
			Actor lk = gr ? gr.Locked(0) : null;
			Actor fa = fl ? fl.FlyingActor(0) : null;
			Actor hd = ho ? ho.HeldBy(0) : null;
			for (int k = 0; k < spawned.Size(); k++)
			{
				let a = spawned[k];
				if (!a) { continue; }
				Vector3 o = mo.AttackPos;
				Vector3 d = (a.pos.x, a.pos.y, a.pos.z + a.height * 0.5) - o;
				Vector3 pd = (d.x, d.y, d.z * 1.2);
				Vector3 dir = RSQDG_Reach.Dir(mo, 0);
				Vector3 pdir = (dir.x, dir.y, dir.z * 1.2); pdir = pdir / pdir.Length();
				double c = (pd / pd.Length()) dot pdir;
				let pol = RSQDG_Policy(StaticEventHandler.Find("RSQDG_Policy"));
				let r = pol ? pol.Decide(a, mo, p) : null;
				Console.Printf("T_OBJ %s pos=(%.0f %.0f %.0f) dist=%.0f angle=%.1f rule=%s w=%.2f sight=%d kg=%.2f special=%d thru=%d r=%.0f grav=%d",
					a.GetClassName(), a.pos.x, a.pos.y, a.pos.z, pd.Length(), acos(clamp(c,-1,1)), r ? r.why : "NO", r ? r.weight : 0,
					mo.CheckSight(a), RSQDG_Mass.Kg(a), a.bSPECIAL, a.bTHRUACTORS, a.radius, !a.bNOGRAVITY);
			}
			let fe = RSQDG_Feel(EventHandler.Find("RSQDG_Feel"));
			let bur = RSQDG_Burden(mo.FindInventory("RSQDG_Burden"));
			if (hd || (fe && (fe.Amp(0) > 0 || fe.Amp(1) > 0)) || bur)
				Console.Printf("T_FEEL t%d amp0=%.2f amp1=%.2f strain=%.2f held=(%.0f %.0f %.0f) vel=%.1f burden=%.2f pvel=%.1f palmv=%.1f",
					t, fe ? fe.Amp(0) : 0, fe ? fe.Amp(1) : 0, ho ? ho.Strain(0) : 0,
					hd ? hd.pos.x : 0, hd ? hd.pos.y : 0, hd ? hd.pos.z : 0, hd ? hd.vel.Length() : 0,
					bur ? bur.GetSpeedFactor() : 1.0, mo.vel.xy.Length(), aim ? RSQDG_Reach.Metres(aim.PalmVelocity(0)) * 35 : 0);
			Console.Printf("T_STATE t%d palm=(%.0f %.0f %.0f) yaw=%.0f pit=%.0f rol=%.0f good=%d palmok=%d armed=%d target=%s lock=%s fly=%s prog=%.2f hold=%s",
				t, mo.AttackPos.x, mo.AttackPos.y, mo.AttackPos.z, RSQDG_Reach.Yaw(mo,0), RSQDG_Reach.Pitch(mo,0), RSQDG_Reach.Roll(mo,0),
				aim && aim.PoseGood(), RSQDG_Reach.PalmOk(mo, p, 0), gr && gr.Held(0),
				tg ? tg.GetClassName() : 'none', lk ? lk.GetClassName() : 'none',
				fa ? fa.GetClassName() : 'none', fl ? fl.Progress(0) : 0, hd ? hd.GetClassName() : 'none');
		}
	}
}
