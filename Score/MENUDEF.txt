// RSQ Score options, modelled on Brutal Doom Quest 2.0's options page (the owner, 10-28): ONE front
// page of real choices, named; ONE Tuning and Debug page at the end that holds every number. Every
// cvar keeps its row. Preset choices: tools/rsmenu/presets.py (Standard = today's defaults).

// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQSC_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQSCRSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "RSQ Score", "RSQSC_Options"
}
