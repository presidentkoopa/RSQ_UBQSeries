// Three handlers, on purpose.
//
//   RS_ScoreRegistry is static: the bonus list is decided by which mods are loaded
//     and cannot change mid-session, so it is built once and outlives the map.
//   RS_ScoreHandler is per-level: the census, the threshold and the counted-monster
//     list all belong to one map.
//   RS_ScoreHUD is per-level and only draws.
//
// The registry is listed FIRST so its OnRegister has run before anything asks it
// for the list.

gameinfo
{
    AddEventHandlers = "RS_ScoreRegistry", "RS_ScoreHandler", "RS_ScoreHUD", "RS_ScoreShots", "RS_ComboLedger", "RS_ScoreWrist"
}

// [11-09 melee expansion] the panel rows' watcher (rows_panels.zs): its own block, so the line above stays as it was
gameinfo
{
    AddEventHandlers = "RS_ScorePanels"
}

// [11-09 head shots] the head-shot check (head.zs): its own block. It judges before the combo ledger (SetOrder -100);
// its head bonus is dealt after every handler has heard the blow (RS_ScoreHeadLate, SetOrder 1000).
gameinfo
{
    AddEventHandlers = "RS_ScoreHead", "RS_ScoreHeadLate"
}
