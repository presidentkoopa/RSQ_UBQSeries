version "4.14"
// TEST ONLY (never ships). RSQ Score's scripted scenarios on the headless desktop engine (tools/qzd_hidden_run.ps1):
// one monster per case, blows dealt with the inflictor, damage type and "touch" ids the mods send, then the kill; Score's
// own rs_score_debug line after each kill is the proof (which rows paid, the chain). Every case prints "RSP CASE n" first,
// and "RSP EXPECT ..." says what the next Score line must show. Nothing here reads Score's code: only its service door,
// the same one the mods use.
//
// A dummy that stands in for RSQ Grenade's blast by NAME (Score classifies by class name): never load this probe with
// RSQ_Grenade.pk3.
class RSQGN_HandBlast : Actor { Default { +NOINTERACTION +NOBLOCKMAP } States { Spawn: TNT1 A -1; Stop; } }
class RSQGN_Thrown : Actor { Default { +NOINTERACTION +NOBLOCKMAP } States { Spawn: TNT1 A -1; Stop; } }

class RSPProbe : EventHandler
{
	Actor m, m2;
	int cases;

	PlayerPawn P() { return players[0].mo; }
	Actor Spawn1(String cls, double d = 96)
	{
		let p = P();
		Vector3 at = p.pos + (cos(p.angle) * d, sin(p.angle) * d, 0);
		let a = Actor.Spawn(cls, at, NO_REPLACE);
		if (a) { a.SetOrigin((a.pos.x, a.pos.y, a.floorz), false); a.angle = a.AngleTo(p); a.freezetics = 100000; }   // on its floor: a frozen actor never falls
		return a;
	}
	void Say(String s) { Console.Printf("RSP t%d %s", level.maptime, s); }
	String Nm(Actor a) { String s = "none"; if (a) s = a.GetClassName(); return s; }
	String Ready() { return Nm(P().player.ReadyWeapon); }
	void Scene(int n, String what, String expect)
	{
		cases++;
		Say(String.Format("CASE %d %s", n, what));
		Say("EXPECT " .. expect);
	}
	Service Score() { let it = ServiceIterator.Find("RS_ScoreService"); return it.Next(); }
	void Touch(String id, Actor v)
	{
		let s = Score();
		if (s && v) s.GetInt("touch", id, 0, 0, v);
	}
	void Hit(Actor v, Actor inf, int dmg, Name dt, int flags = 0)
	{
		if (v && v.health > 0) v.DamageMobj(inf, P(), dmg, dt, flags);
	}
	void Pending(String w)
	{
		let p = P();
		if (!p.FindInventory(w)) p.GiveInventory(w, 1);
		p.player.PendingWeapon = Weapon(p.FindInventory(w));
	}
	// a hitscan from the pawn's gun at the monster (the puff is the inflictor, as a real shot)
	void Shoot(Actor v, int dmg)
	{
		let p = P();
		if (!v) return;
		double pitch = -atan2(v.pos.z + v.height * 0.5 - (p.pos.z + p.height * 0.5 + 8), p.Distance2D(v));
		p.LineAttack(p.AngleTo(v), 512, pitch, dmg, 'Hitscan', "BulletPuff");
	}
	void Spawned(String cls = "ZombieMan")
	{
		m = Spawn1(cls);
		Say(String.Format("monster %s hp %d at %.0f", Nm(m), m ? m.health : 0, m ? P().Distance2D(m) : 0));
	}

	override void WorldTick()
	{
		let p = P(); if (!p) return;
		int t = level.maptime;
		let cv = CVar.FindCVar("rs_score_debug");
		if (cv && !cv.GetBool()) cv.SetBool(true);
		if (t == 3) Pending("Fist");
		if (t == 20) Say("ready " .. Ready());

		// 1. a punch kill with the fists up: the old world, unchanged
		if (t == 40) { Scene(1, "punch kill, fists up", "Brawler 0.500 and Point-Blank; chain punch"); Spawned(); }
		if (t == 45) Hit(m, p, 200, 'Melee');

		// 2. the gun-butt shove (Melee names it, then its 'Kick' blow): no Boot, no Brawler; then shove > gun pays as a chain
		if (t == 80) { Scene(2, "shove kill", "NO Boot, NO Brawler; chain shove"); Spawned(); }
		if (t == 85) { Touch("shove", m); Hit(m, p, 200, 'Kick'); }
		if (t == 100) { Scene(3, "shove, then a gun kill", "Combo Chain 2-Tool 0.25 (0.2 x tempo), NOT Gun Kata; chain shove>gun"); Spawned(); }
		if (t == 105) { Touch("shove", m); Hit(m, p, 5, 'Kick'); }
		if (t == 110) Hit(m, p, 200, 'Hitscan');   // the pawn as inflictor: a puffless trace, a shot

		// 4. Melee's Ground Pound: two 'Melee' blows from the pawn, the first named AFTER it lands (the old order: the name
		// still replaces the guess), the killing one named before
		if (t == 140) { Scene(4, "pound kill (two blows)", "NO Brawler, NO Knockdown, NO Counter; chain pound (one run)"); Spawned(); }
		if (t == 145) { Hit(m, p, 5, 'Melee'); Touch("pound", m); }
		if (t == 146) { Touch("pound", m); Hit(m, p, 200, 'Melee'); }

		// 5. a grenade blast (RSQ Grenade's HandBlast by name), then a gun: two tools
		if (t == 180) { Scene(5, "grenade blast, then a gun kill", "Combo Chain[Mop Up 0.30]; chain grenade.frag>gun"); Spawned(); }
		if (t == 185) { let b = Actor.Spawn("RSQGN_Thrown", m.pos, NO_REPLACE); if (b) b.target = p; Hit(m, b, 5, 'None', DMG_EXPLOSION); if (b) b.Destroy(); }
		if (t == 190) Hit(m, p, 200, 'Hitscan');
		if (t == 220) { Scene(6, "grenade blast kill", "NO Boot, NO Brawler, NO Cleaved, NO Juggle, NO Hold My Beer; chain grenade.frag"); Spawned(); }
		if (t == 225) { let b = Actor.Spawn("RSQGN_Thrown", m.pos, NO_REPLACE); if (b) b.target = p; Hit(m, b, 200, 'None', DMG_EXPLOSION); if (b) b.Destroy(); }

		// 7. RSQ Whip's crack ('Whip' from the pawn): Lashed, not Brawler; then crack > gun
		if (t == 260) { Scene(7, "whip crack kill", "Lashed 0.250, NO Brawler; chain whip.crack"); Spawned(); }
		if (t == 265) Hit(m, p, 200, 'Whip');
		if (t == 290) { Scene(8, "whip crack, then a gun kill", "Combo Chain 2-Tool 0.25; chain whip.crack>gun"); Spawned(); }
		if (t == 295) Hit(m, p, 5, 'Whip');
		if (t == 300) Hit(m, p, 200, 'Hitscan');

		// 9. tags ride on the blow: a punch tagged three ways, then a gun: Gun Kata, not a 4-Tool Combo
		if (t == 330) { Scene(9, "punch + tags hit.heavy juggle crit, then a gun kill", "Combo Chain[Gun Kata 0.40] only, NO 4-Tool; chain punch>gun"); Spawned(); }
		if (t == 335) { Hit(m, p, 5, 'Melee'); Touch("hit.heavy", m); Touch("juggle", m); Touch("crit", m); }
		if (t == 340) Hit(m, p, 200, 'Hitscan');

		// 10. the hook's Zip Kick: a 'Melee' blow named zip.kick after it: the hook's, not a punch
		if (t == 370) { Scene(10, "zip kick kill", "NO Brawler, NO Knockdown; chain hook.zip"); Spawned(); }
		if (t == 375) { Touch("zip.kick", m); Hit(m, p, 200, 'Melee'); }

		// 11. gun classes and Switcharoo: a pistol hit, then a shotgun kill, real hitscans
		if (t == 400) { Scene(11, "pistol hit, then shotgun kill", "Switcharoo 0.150; chain gun.pistol>gun.shotgun (one action: no chain pay)"); Spawned("DoomImp"); Pending("Pistol"); }
		if (t == 420) { Say("ready " .. Ready()); Shoot(m, 10); }
		if (t == 425) Pending("Shotgun");
		if (t == 445) { Say("ready " .. Ready()); Shoot(m, 300); }

		// 12. the generic rule: whip > kick > pound > shotgun inside 2 s: Gun Kata names kick+gun, the rest is generic with breadth and tempo
		if (t == 480) { Scene(12, "whip, kick, pound, shotgun kill in 15 tics", "Combo Chain[Door Kick 0.50][4-Tool 0.94 mods4+0.2 tempo] = 1.4375 (Door Kick: kick > gun.shotgun+pb, over Gun Kata), a big moment; chain whip.crack>kick>pound>gun.shotgun"); Spawned("DoomImp"); }
		if (t == 485) Hit(m, p, 5, 'Whip');
		if (t == 490) Hit(m, p, 5, 'Kick');
		if (t == 495) { Hit(m, p, 5, 'Melee'); Touch("pound", m); }
		if (t == 500) Shoot(m, 300);

		// 13. a missile with DamageType Kick (BD22's Grenade Launcher round) is a gun, not a boot
		if (t == 530) { Scene(13, "a 'Kick' missile kill", "NO Boot; chain gun.explosive"); Spawned(); }
		if (t == 535) { let r = Actor.Spawn("Rocket", m.pos + (0, 0, 20), NO_REPLACE); if (r) { r.target = p; r.vel = (0, 0, 0); Hit(m, r, 200, 'Kick'); r.Destroy(); } }

		// 15. a Headlock toss: the thrown monster (alive) is the inflictor of a 'Melee' blow on what it hits: a throw, not a punch
		if (t == 550) { Scene(15, "a thrown monster strikes it (Headlock toss)", "NO Brawler, NO Knockdown; chain headlock.throw"); Spawned(); }
		if (t == 555) { let b = Spawn1("DoomImp", 60); if (b) { Hit(m, b, 200, 'Melee'); b.Destroy(); } }
		// 16. a gun thrown from the rig: the weapon is the inflictor of a 'Melee' blow: a throw, not a punch
		if (t == 580) { Scene(16, "a thrown gun strikes it (rig throw)", "NO Brawler, NO Knockdown; chain throw.gun"); Spawned(); }
		if (t == 585) { let g = Actor.Spawn("Shotgun", m.pos, NO_REPLACE); if (g) { Hit(m, g, 200, 'Melee'); g.Destroy(); } }

		// 14. slow motion (RSQ_SlowMo loaded, rsp_slowmo 1): the key's toggle, then shove > shotgun = F.E.A.R. Effect; Bullet Time shares the rest, not the chain
		// with RSQ_SlowMo loaded (its service answers), the slow-motion case runs too
		let smIt = ServiceIterator.Find("RSQSM_Service");
		if (smIt.Next() != null)
		{
			if (t == 600) { Say("slow motion ON (key)"); EventHandler.SendNetworkEvent("rsm_toggle", 2); }
			if (t == 620) { Scene(14, "in the player's slow motion: shove, then shotgun kill", "Combo Chain[F.E.A.R. Effect 0.60] (over Breach 0.5), Bullet Time 0.050 (half of Point-Blank 0.1, not the chain); chain shove>gun.shotgun"); Spawned(); }
			if (t == 625) { Touch("shove", m); Hit(m, p, 5, 'Kick'); }
			if (t == 630) Shoot(m, 300);
			if (t == 660) { Say("slow motion OFF"); EventHandler.SendNetworkEvent("rsm_toggle", 2); }
		}
		// ---- the 11-03 rows ----
		// 17. the Headlock Slam: Melee names it (slam), then its 'Melee' blow
		if (t == 700) { Scene(17, "headlock slam kill", "Piledriver 0.500 (a big moment), NO Brawler; chain headlock.slam"); Spawned(); }
		if (t == 705) { Touch("slam", m); Hit(m, p, 200, 'Melee'); }
		// 18. a grenade going off in the hand (the HandBlast is the inflictor), the player alive after
		if (t == 730) { Scene(18, "hand blast kill, player alive", "Hold My Beer 0.800; chain grenade.frag"); Spawned(); }
		if (t == 735) { let b = Actor.Spawn("RSQGN_HandBlast", p.pos, NO_REPLACE); if (b) { b.target = p; Hit(m, b, 200, 'None', DMG_EXPLOSION); b.Destroy(); } }
		// 19. one blast, two bodies: the second kill by the same inflictor pays Multi-Kill
		if (t == 760) { Scene(19, "one blast kills two", "second kill: Multi-Kill 0.200; NO Hold My Beer"); Spawned(); m2 = Spawn1("ZombieMan", 140); }
		if (t == 765) { let b = Actor.Spawn("RSQGN_Thrown", m.pos, NO_REPLACE); if (b) { b.target = p; Hit(m, b, 200, 'None', DMG_EXPLOSION); Hit(m2, b, 200, 'None', DMG_EXPLOSION); b.Destroy(); } }
		// 20. Miniboss's flat award arrives on the kill's tic: it pays a share of that kill's base instead (1.0 x 20)
		if (t == 790) { Scene(20, "a kill, then the award 'Miniboss' 500", "RS_Score: +20 more on the Miniboss row (1.0 x base 20), not +500"); Spawned(); }
		if (t == 795) { Hit(m, p, 200, 'Hitscan'); let s = Score(); if (s) Say(String.Format("award Miniboss 500 -> paid %d", s.GetInt("award", "Miniboss", 500, 0, p))); }
		// 21. a melee kill on a Baron: Giant Slayer
		if (t == 820) { Scene(21, "punch kills a Baron (the shotgun is up: no Brawler)", "Giant Slayer 0.500"); Spawned("BaronOfHell"); }
		if (t == 825) Hit(m, p, 2000, 'Melee');
		// 22. a headshot damage type on the killing shot
		if (t == 850) { Scene(22, "a 'Head' hitscan kill", "Headshot 0.250"); Spawned(); }
		if (t == 855) Hit(m, p, 200, 'Head');
		// 23. three whip kills in a row: the third pays Weapon Master
		if (t == 880) { Scene(23, "three whip kills in a row", "third kill: Weapon Master 0.100; NO Multi-Kill (separate cracks)"); Spawned(); }
		if (t == 885) Hit(m, p, 200, 'Whip');
		if (t == 890) Spawned();
		if (t == 895) Hit(m, p, 200, 'Whip');
		if (t == 900) Spawned();
		if (t == 905) Hit(m, p, 200, 'Whip');
		if (t == 930) Say(String.Format("DONE %d cases", cases));
	}
}
