// TEST ONLY (never ships): logs what RSQ Accessibility does, for tools/rigtest/run.sh.
//   AT <tic> | the brace: WF_TWOHANDSTABILIZED, the off-hand gun, vr_two_handed_weapons, rsqa_brace_off / keep
//            | the modifier (swap) | the slow-motion label's edge (smEdge)
//   netevent rsqat_ready <hand>   asks the shared rumble for a "ready" cue on that hand (as every series mod does)
//   rsqat_slowmo 0/1              stands in for RSQ_SlowMo's service: slow motion on / off
//   set rsqat_sel "<row label>"   while a menu is open: selects that row (then the thumbstick changes it)
class RSQAT_Probe : EventHandler
{
	override void WorldTick()
	{
		let ev = CVar.FindCVar("rsqat_every");
		int every = ev ? max(1, ev.GetInt()) : 5;
		if (level.maptime % every) return;
		let p = players[consoleplayer];
		if (!p.mo) return;
		let h = RSQA_Handler.Get();
		Console.Printf("AT %d stab=%d off=%s two_handed=%d brace_off=%d keep=%d swap=%d smEdge=%d quake_haptic=%.2f",
			level.maptime, (p.WeaponState & WF_TWOHANDSTABILIZED) ? 1 : 0, p.OffhandWeapon ? p.OffhandWeapon.GetClassName() : 'none',
			CI("vr_two_handed_weapons"), CI("rsqa_brace_off"), CI("rsqa_brace_keep"),
			h ? int(h.swap[consoleplayer]) : -1, h ? h.smEdge : -1, CF("vr_quake_haptic_level"));
	}
	static int CI(Name n) { let c = CVar.FindCVar(n); return c ? c.GetInt() : -9; }
	static double CF(Name n) { let c = CVar.FindCVar(n); return c ? c.GetFloat() : -9; }

	override void NetworkProcess(ConsoleEvent e)
	{
		// netevent rsqat_offgun: a gun in the off hand, put there as RSQ ShieldSaw does (OffhandWeapon, then raised)
		if (e.Name ~== "rsqat_offgun")
		{
			let p = players[e.Player];
			let w = Weapon(p.mo.FindInventory('RSQ_Off_Rifle'));
			if (!w) { p.mo.GiveInventory('RSQ_Off_Rifle', 1); w = Weapon(p.mo.FindInventory('RSQ_Off_Rifle')); }
			if (!w) { Console.Printf("AT_OFFGUN none"); return; }
			w.bOffhandWeapon = true;
			p.OffhandWeapon = w;
			p.PendingWeapon = w;
			p.mo.BringUpWeapon();
			Console.Printf("AT_OFFGUN %d %s ammo1=%s", level.maptime, w.GetClassName(), w.Ammo1 ? w.Ammo1.GetClassName() : 'none');
			return;
		}
		if (e.Name ~== "rsqat_ready")
		{
			let it = ServiceIterator.Find("RSQA_Service");
			let s = it.Next();
			int r = s ? s.GetInt("ready", "", e.Player, e.Args[0]) : -2;
			Console.Printf("AT_READY %d hand=%d answer=%d", level.maptime, e.Args[0], r);
		}
	}

	override void WorldThingSpawned(WorldEvent e)
	{
		if (e.Thing && (e.Thing.GetClassName() == 'RSQA_PulsePuff' || e.Thing.GetClassName() == 'RSQA_Buzz'))
			Console.Printf("AT_SPAWN %d %s", level.maptime, e.Thing.GetClassName());
	}
}

// stands in for RSQ_SlowMo: the "slowmo.active" its service answers
class RSQSM_Service : Service   // the name RSQ_SlowMo answers to (it is not loaded in these runs)
{
	override int GetInt(String request, String stringArg, int intArg, double doubleArg, Object objectArg, Name nameArg)
	{
		if (request ~== "slowmo.active") { let c = CVar.FindCVar("rsqat_slowmo"); return c && c.GetInt() ? 1 : 0; }
		return -1;
	}
}

class RSQAT_Menu : StaticEventHandler
{
	override void UiTick()
	{
		let c = CVar.FindCVar("rsqat_sel");
		if (!c || c.GetString() == "") return;
		let m = OptionMenu(Menu.GetCurrentMenu());
		if (!m) return;
		for (int i = 0; i < m.mDesc.mItems.Size(); i++)
		{
			if (m.mDesc.mItems[i].mLabel == c.GetString())
			{
				m.mDesc.mSelectedItem = i;
				Console.Printf("AT_SEL %s row %d", c.GetString(), i);
				c.SetString("");
				return;
			}
		}
	}
}

// QoL round (AC1-AC8). netevent rsqat_throw <vx> <vy> <vz>  adds velocity (a knockback stand-in)
//   netevent rsqat_lift <units>       puts the player that far up (a long fall)
//   netevent rsqat_behind <dist>      a dormant ZombieMan that far behind the player
//   netevent rsqat_alert              the player's noise alert (wakes what hears it)
//   netevent rsqat_tele <dist> <deg>  the probe's last monster teleports (Actor.Teleport, fogs) to that spot round you
//   netevent rsqat_cv <name> <tic>    nothing: a marker in the log
class RSQAT_QoLProbe : EventHandler
{
	Actor last;
	override void WorldTick()
	{
		let c = CVar.FindCVar("rsqat_qol");
		let p = players[consoleplayer];
		if (!p.mo) return;
		let it = ThinkerIterator.Create("RSQA_Buzz");
		RSQA_Buzz b;
		while (b = RSQA_Buzz(it.Next())) if (b.tleft == b.total) Console.Printf("AT_BUZZ %d left=%d amp=%.3f tics=%d", level.maptime, b.left, b.amp, b.total);
		if (!c || !c.GetInt()) return;
		let s = ServiceIterator.Find("RSQA_Service").Next();
		let h = RSQA_Handler.Get();
		let mo = p.mo;
		Console.Printf("AT_Q %d raw %.3f %.3f %.3f %.3f %.3f st %.3f %.3f %.3f %.3f %.3f on=%d seat=%d drop=%.3f eye=%.3f stand=%.3f hmd=%.3f vig=%.3f vz=%.1f v=%.1f taps=%.2f rumble=%.2f",
			level.maptime, mo.AttackPos.x - mo.pos.x, mo.AttackPos.y - mo.pos.y, mo.AttackPos.z - mo.pos.z, mo.AttackAngle, mo.AttackPitch,
			s.GetDouble("steady", "x", consoleplayer, 0) - mo.pos.x, s.GetDouble("steady", "y", consoleplayer, 0) - mo.pos.y,
			s.GetDouble("steady", "z", consoleplayer, 0) - mo.pos.z, s.GetDouble("steady", "angle", consoleplayer, 0), s.GetDouble("steady", "pitch", consoleplayer, 0),
			s.GetIntData("steady", "", consoleplayer), s.GetIntData("seated", "", consoleplayer), s.GetDoubleData("seated.drop", "", consoleplayer),
			s.GetDoubleData("seated.eye", "", consoleplayer), s.GetDoubleData("stand.eye", "", consoleplayer), mo.HmdPosition.y,
			h ? h.vig : -1, mo.vel.z, mo.vel.Length(), s.GetDoubleData("taps"), s.GetDoubleData("rumble"));
	}

	override void NetworkProcess(ConsoleEvent e)
	{
		let mo = players[e.Player].mo;
		if (!mo) return;
		if (e.Name ~== "rsqat_throw") { mo.vel += (e.Args[0], e.Args[1], e.Args[2]); Console.Printf("AT_THROW %d", level.maptime); }
		else if (e.Name ~== "rsqat_lift") { mo.SetOrigin(mo.pos + (0, 0, e.Args[0]), false); mo.vel = (0, 0, 0); Console.Printf("AT_LIFT %d z=%.0f floor=%.0f", level.maptime, mo.pos.z, mo.floorz); }
		else if (e.Name ~== "rsqat_behind")
		{
			Vector3 at = mo.Vec3Angle(e.Args[0], mo.angle + 180);
			last = Actor.Spawn("ZombieMan", (at.xy, mo.floorz));
			if (last) { last.angle = mo.angle; last.target = null; }
			Console.Printf("AT_BEHIND %d %s", level.maptime, last ? "spawned" : "none");
		}
		else if (e.Name ~== "rsqat_alert") { mo.SoundAlert(mo); Console.Printf("AT_ALERT %d", level.maptime); }
		else if (e.Name ~== "rsqat_tele" && last)
		{
			Vector3 at = mo.Vec3Angle(e.Args[0], mo.angle + e.Args[1]);
			bool ok = last.Teleport((at.xy, mo.floorz), last.angle, 0);
			Console.Printf("AT_TELE %d ok=%d", level.maptime, ok);
		}
		else if (e.Name ~== "rsqat_mark") Console.Printf("AT_MARK %d %d", level.maptime, e.Args[0]);
	}
}

// "rsqat_key enter|yes": a key to the open menu (the message box takes "yes" as its y key)
class RSQAT_MenuKeys : StaticEventHandler
{
	override void UiTick()
	{
		let c = CVar.FindCVar("rsqat_key");
		if (!c || c.GetString() == "") return;
		let m = Menu.GetCurrentMenu();
		if (!m) return;
		String k = c.GetString();
		c.SetString("");
		bool r = true;
		if (k ~== "yes" && m is "MessageBoxMenu") MessageBoxMenu(m).HandleResult(true);
		else r = m.MenuEvent(Menu.MKEY_Enter, false);
		Console.Printf("AT_KEY %s menu=%s r=%d", k, m.GetClassName(), r);
	}
}
