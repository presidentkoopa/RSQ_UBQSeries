// ==== GESTURES (the series' gesture reader, was RSQ_Gestures until 0.99a; owner 11-05). Names unchanged, so saved
// settings carry over. ====================================================================================
// (the same choice RS_Gestures made). Distances are METRES from your eyes, measured in your own frame
// (forward / sideways / up), so a tall, short or seated player makes the same gesture; the vertical is
// converted with level.pixelstretch (1.2) before any comparison. Angles are degrees as QuestZDoom
// reports them (AttackRoll / AttackPitch).

// ---- INSPECT: the gun held up in front of your face, wrist twisted, held briefly ----------------------
// Answers whenever a mod asks, unless switched off (rsqg_inspect_off).
user bool   rsqg_inspect_off     = false;  // QoL GE5 (Option): 1 = the watch-check pose is off; "inspect" answers -1 (not answering), so RSQ Gun Cards uses its own pose
user bool   rsqg_inspect_tick    = true;   // QoL GE4: a light tick in that hand's controller when the pose is reached (held)
user float  rsqg_tick_strength   = 0.35;   // ...its strength at the peak (0..1; the series' quake rumble, vr_quake_haptic_level scales it)
user float  rsqg_inspect_twist   = 35;     // wrist twist (roll) needed to start, degrees either way (was 45 until 11-06: owner "sometimes I don't see the card when I want to")
user float  rsqg_inspect_untwist = 20;     // ...and below this it drops (less than the start: no flicker; was 30 until 11-06, the same 15 degree gap;
user float  rsqg_inspect_across  = 45;     // the forearm across the body: the least angle between where the hand points and where you face (aiming points forward; owner 11-11: a canted two-handed rifle was a watch check). 0 = not checked
                                           // held at least 5 under the start in code, so the two sliders cannot cross)
user int    rsqg_inspect_way     = 2;      // 2 the back of the wrist to your face, as you check a watch (rig 2026-11-02: right AttackRoll -90 fires, +90 does not; left mirrored); 1 the pulse side; 0 either way
user int    rsqg_inspect_hold    = 6;      // tics the pose must be held before it counts (35 tics = 1 second)
user float  rsqg_inspect_low     = 0.30;   // lowest the hand may be, metres BELOW your eyes (upper chest)
user float  rsqg_inspect_high    = 0.20;   // highest, metres ABOVE your eyes
user float  rsqg_inspect_near    = 0.15;   // nearest in front of your eyes, metres
user float  rsqg_inspect_far     = 0.65;   // furthest in front, metres (an arm's length)
user float  rsqg_inspect_side    = 0.35;   // furthest to either side of your eyes, metres
user float  rsqg_inspect_pitch   = 60;     // the gun may point at most this far up or down (degrees)
user float  rsqg_inspect_still   = 1.2;    // the hand must be slower than this (metres a second) to start
user float  rsqg_inspect_slack   = 0.05;   // once on, the place may drift this far (metres) before it drops

// ---- ONE HAND: stillness ------------------------------------------------------------------------------------
user float  rsqg_still_mps       = 0.3;    // below this a hand counts as still (palms out; the debug telemetry)

// ---- BOTH HANDS: down past the hips, thrown apart, both palms out (RS_Gestures' rsg_twohand) -----------
// Off by default, as in RS_Gestures: the switch buys the rsqg_pair event.
user bool   rsqg_twohand         = false;
// QoL GE6: each two-hand event on its own switch, under the one above (which stays off by default, so nothing
// fires until it is on; then these pick which of the three do)
user bool   rsqg_two_down        = true;   // both hands swept down past the hips
user bool   rsqg_two_apart       = true;   // both hands thrown apart from the chest
user bool   rsqg_two_palms       = true;   // both palms held out
user bool   rsqg_two_thrust      = true;   // both palms thrust forward together (RSQ Melee's Clear Out)
user float  rsqg_two_window_ms   = 120;    // how far apart in time the two hands may move and still be one gesture
user float  rsqg_two_age         = 180;    // each hand's fastest instant must be this fresh (ms; 0 = any in the last quarter second)
user float  rsqg_two_repeat_ms   = 500;    // before the same two-hand gesture may fire again
user float  rsqg_two_down_mps    = 2.0;    // down: each hand's downward speed
user float  rsqg_two_down_frac   = 0.5;    // down: how much of the speed must be downward (0..1)
user float  rsqg_two_hip_m       = 0.60;   // the hip line, metres below your eyes (also the floor for palms out)
user float  rsqg_two_apart_mps   = 2.0;    // apart: how fast the gap between the hands opens
user float  rsqg_two_chest_m     = 0.40;   // apart: the chest, metres below your eyes
user float  rsqg_two_chest_r     = 0.50;   // apart: thrown from within this of the chest
user float  rsqg_two_front_m     = 0.25;   // palms: each hand at least this far in front of your eyes
user float  rsqg_two_hold_ms     = 400;    // palms: held this long
// Palms out on QuestZDoom: there is no finger or grip reading, so "open palm" is read as the hand
// pointing up (fingers up), not pulling a trigger, and the hands side by side, not one behind the other
// on a rifle.
user float  rsqg_two_palms_pitch = 40;     // palms: each hand pointing at least this far up (degrees)
user float  rsqg_two_palms_gap   = 0.25;   // palms: the hands at least this far apart sideways (metres; thrust too)
user float  rsqg_two_thrust_mps  = 2.5;    // thrust: each hand's speed forward, where you face
user float  rsqg_two_thrust_frac = 0.6;    // thrust: how much of the speed must be forward (0..1)
user float  rsqg_two_thrust_pitch = 20;    // thrust: each hand pointing at least this far up (degrees; a palm pushed out)

// ---- PUNCH: jab / hook / uppercut (the gloves' reader, now RSQ Melee's gloves.zs, copied), answered whenever a mod asks

user float  rsqg_punch_mps       = 2.0;    // average speed over the swing for a punch (gloves.zs rsq_punch_min_speed)
user int    rsqg_punch_rehit     = 6;      // tics before the same hand may punch again (rsq_punch_rehit)
user float  rsqg_punch_rise      = 0.5;    // an uppercut: the swing's upward share above this (rsq_punch_uppercut_rise)
user float  rsqg_punch_window    = 300;    // "punch" answers the last punch for this long (ms)
user float  rsqg_punch_together  = 0.6;    // the other hand moving the same way at this share of the speed = a two-hand shove, not a punch
