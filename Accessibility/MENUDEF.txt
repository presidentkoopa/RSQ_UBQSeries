// RSQ_Accessibility -- the series' shared accessibility settings, one page under Radiant Silvergun
// Options (short Title Case labels, no help text; StaticText " " is only a spacer).
//
// The links to each series mod's page that used to sit here (RSQASection rows) moved to Radiant
// Silvergun Options itself (tools/rsmenu), which lists every loaded series mod beside this page.
// RSQASection / RSQA_Menu stay in zscript/rsqa_menu.zs: a Section "Label", "Class1|Class2" row
// still shows its rows only when one of those classes is loaded, if a row ever needs it again.

// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQA_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQARSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "Accessibility", "RSQA_Options"
}
