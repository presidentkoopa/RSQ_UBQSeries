		}
	
	states
		{
		Spawn:
			//"####" "#" 0;
			ICNX "#" 1;
			wait;
		}
	}

class champion_MissileEffect : Actor
	{
	int id;
	bool childmonster;	// (performance, 2026-11-27) one of the leaves it spawned came out a monster: see RemoveMonsterChildren

	// (performance, 2026-11-27) A_RemoveChildren(true) walks every actor in the level, and Tick called it up to
	// twice a tic for each effect while it faded out after its missile. With flags 0 it removes only children
	// that are monsters (DoRemove, p_actionfunctions.cpp), and the only children this effect has are the
	// champion_Leaves it spawns itself with SXF_SETMASTER: the walk can remove something only if one of
	// those was replaced by a monster. Comparator: scratchpad agents/champions/removechildren_compare.txt.
	void RemoveMonsterChildren()
		{
		if(childmonster)
			A_RemoveChildren(true);
		}

	default
		{
		+NOINTERACTION;
		+NOGRAVITY;
		+BRIGHT;
		RenderStyle "Translucent";
		alpha 0.3;
		scale 0.6;
		}
	
	override void PostBeginPlay(void)
		{
		super.PostBeginPlay();
		
		if(master)
			{
			//double scalefactor = master.height * 0.0125;
			
			//scale.x = scale.y = scalefactor;
			}
		
		switch(id)
			{
			default:
				break;
				
			case champ_Cyan:
				for(int i = 0; i < 3; i++)
					{
					bool spawned;
					actor leaf;
					[spawned, leaf] = A_SpawnItemEx("champion_Leaves", angle:i*45, flags:SXF_SETMASTER|SXF_NOCHECKPOSITION);
					if(leaf && leaf.bISMONSTER)
						childmonster = true;
					}
				break;
			}
		}
		
	override void Tick()
		{
		super.Tick();

		if(!master)
			{
			RemoveMonsterChildren();
			SetStateLabel("Death");
			}

		if(master)
			{
			int zmis = (master is "Rocket") ? 8 : 0;
			A_Warp(AAPTR_MASTER, zofs:zmis, flags:WARPF_INTERPOLATE|WARPF_NOCHECKPOSITION);
			}
			
 		if(master && master.InStateSequence(master.CurState,master.FindState("Death")))
			{
			RemoveMonsterChildren();
			SetStateLabel("Death");
			} 
			
		if(master && master.vel.x == 0 && master.vel.y == 0)
			{
			RemoveMonsterChildren();
			SetStateLabel("Death");
			}			
		}
	
	states
		{
		Spawn:
			GLOW "#" 1;
			wait;
		Death:
			GLOW "#" 1 A_FadeOut();
			wait;
		}
	}

class champion_Heart : Actor
	{
	default
		{
		Radius 16;
		Height 24;
		Speed 1;
		Health 50;
		Scale 0.75;
		mass 0x7FFFFFFF;
		PushFactor 0.1;
		Monster;
		-SOLID;
		-COUNTKILL;
		+NOGRAVITY;

		//+BRIGHT;

		+NOTARGET;

		}
		
	name fuckEoA;
	
	override void PostBeginPlay()
		{
		if(master)

			target = master;
			fuckEoA = 'GloryKillCounter';

		{

		}

		{
		super.tick();
		takeInventory(fuckEoA,1);

			{
			target = master;
			}
		if(master)

				{
				//A_SpawnItemEx("TeleportFog");
				SetOrigin((master.pos.x,master.pos.y,master.pos.z+(master.height*0.5)),0);
				//A_SpawnItemEx("TeleportFog");
				}
			}
		}
		
	states
		{
		Spawn:
			HART A 35;
			HART A 1 A_SetScale(scale.x+0.1,scale.y-0.0);
			HART A 1 A_SetScale(scale.x+0.1,scale.y-0.1);
			HART A 1 A_SetScale(scale.x+0.0,scale.y-0.1);
			HART A 0 A_SpawnItemEx("Blood");
			HART A 1 A_SetScale(scale.x-0.1,scale.y+0.0);
			HART A 1 A_SetScale(scale.x-0.1,scale.y+0.1);
			HART A 1 A_SetScale(scale.x+0.0,scale.y+0.1);
			loop;
		Death.Massacre:
			TNT1 A 0 
				{
				master.bBUDDHA = false;				
				A_KillMaster("Massacre",KILS_FOILBUDDHA);
				}
		Death:
			TNT1 A 0 
				{
				if(master)
					{
					master.bBUDDHA = false;
					}
				for(int i = 0; i < 16; i++)
					{
					A_SpawnItemEx("Blood", 	xvel:frandom(2.0,4.0),
											zvel:frandom(4.0,8.0),
											angle:frandom(0.0,359.0));

				}

		GenericFreezeDeath:
			"####" "#" 0 
				{

					{
					master.bNOBLOOD = master.default.bNOBLOOD;
					master.bNODAMAGE = false;
					}

// Titan Champion Aggression buff
class champion_AggroEffect : CustomInventory
	{
	default
		{
		+INVENTORY.ALWAYSPICKUP;
		+INVENTORY.AUTOACTIVATE;
		}
	states
		{
		Use:
			TNT1 A 0
				{
				if(invoker.owner)
					{
					if(invoker.owner.bSPECIAL || invoker.owner.bWEAPONSPAWN)
						return false;

					// (performance, 2026-11-27) a monster that already has the buff only has it refreshed:
					// done in place, without spawning the giver and a second power (see RefreshPower).
					let pw = champion_AggroEffect.RefreshablePower(self);
					if(pw)
						{
						champion_AggroEffect.RefreshPower(pw);
						return true;
						}

					A_GiveInventory("champion_AggroGiver");
					return true;
					}
				return false;
				}
			stop;
		}

	// (performance, 2026-11-27) A Titan gives this to every monster near it every tic, and on each give
	// A_GiveInventory("champion_AggroGiver") spawned two more actors (the giver, then the giver's new
	// champion_AggroPower) only for the new power to be thrown away by the one already held, whose
	// Powerup.HandlePickup resets its clock. That was the bulk of the game's spawns in a big fight.
	// RefreshPower does what that chain leaves behind, read from the engine (QZD 17.3 powerups.zs and
	// inventory.zs): PowerupGiver.Use's copy of the giver's tics, colour and flags onto the new power, then
	// the held power's HandlePickup same-class branch line for line. Nothing else in that chain touches the
	// monster: no item in the series claims a giver or a power of another class (RS Score's crit delayers'
	// age resets already ran, to the same value, when this champion_AggroEffect itself was handed down the
	// chain; champion_GreyShadowSpawner resets only for its own class since decision #48, so it ignores
	// all three), and Actor.CanReceive / HasReceived are not
	// overridden for monsters. Each spawn drew once from the "SpawnActor" RNG (its FloatBobPhase), so the
	// two draws are made here too: the random stream stays the same.
	// The engine path is kept whenever its outcome could differ: no power yet (first give: InitEffect, the
	// pentagram), a giver still in the inventory this tic (it would swallow the new giver), a PlayerPawn,
	// or compat_oldrandom (spawns then draw from Doom's table, not "SpawnActor").
	// Comparator: scratchpad agents/champions/aggro_refresh_compare.txt.
	static Powerup RefreshablePower(Actor mo)
		{
		if(!mo || mo is "PlayerPawn" || (level.compatflags2 & (1 << 18)))	// 1 << 18: COMPATF2_OLD_RANDOM_GENERATOR
			return null;
		if(mo.FindInventory("champion_AggroGiver"))
			return null;
		return Powerup(mo.FindInventory("champion_AggroPower"));
		}

	static void RefreshPower(Powerup pw)
		{
		// the giver's spawn and the new power's spawn, one "SpawnActor" draw each
		int draw = random[SpawnActor](0, 255);
		draw = random[SpawnActor](0, 255);

		// PowerupGiver.Use: what the new power would have carried
		let gdef = GetDefaultByType("champion_AggroGiver");
		let pdef = GetDefaultByType("champion_AggroPower");
		int newtics = pdef.EffectTics;
		if(gdef.EffectTics != 0)
			newtics = gdef.EffectTics;
		color newblend = pdef.BlendColor;
		if(gdef.BlendColor != 0)
			{
			if(gdef.BlendColor != (Powerup.SPECIALCOLORMAP_MASK | 65535))
				newblend = gdef.BlendColor;
			else
				newblend = 0;
			}
		bool newalways		= pdef.bAlwaysPickup || gdef.bAlwaysPickup;
		bool newadditive	= pdef.bAdditiveTime || gdef.bAdditiveTime;

		// Powerup.HandlePickup, the held power's same-class branch
		if(newtics == 0)
			return;
		if(newadditive)
			{
			pw.EffectTics += newtics;
			pw.MaxEffectTics = max(pw.EffectTics, pw.MaxEffectTics);
			pw.BlendColor = newblend;
			}
		else if(pw.EffectTics > BLINKTHRESHOLD && !newalways)
			{
			}
		else if(newtics > pw.EffectTics)
			{
			pw.EffectTics = pw.MaxEffectTics = newtics;
			pw.BlendColor = newblend;
			}
		}
	}
	
class champion_AggroPower : PowerUp
	{
	override void InitEffect()
		{
		Super.InitEffect();
		if(owner)
			{
			if(owner.CountInv("champion_TitanToken"))
				return;
				
			owner.bALWAYSFAST = true;
			owner.A_SpawnItemEx("champion_AggroVisual",flags:SXF_SETMASTER);
			}
		}
	
	override void EndEffect()
		{
		super.EndEffect();
		if(owner)
			{
			if(owner.CountInv("champion_TitanToken"))
				return;
				
			owner.bALWAYSFAST = owner.default.bALWAYSFAST;
			}
		}
	}

class champion_AggroVisual : Actor
	{
	int visdir;
	
	default	
		{
		+NOINTERACTION;
		+FLATSPRITE;
		+ROLLSPRITE;
		Renderstyle "Translucent";
		Scale 0.666;
		}
	override void Tick()
		{
		super.Tick();
		
		if (visdir > 0)
			{
			Alpha += 1/16.;
			if (Alpha >= 1.)
				{
				Alpha = 1.;
				visdir = -1;
				}
			}
		else
			{
			Alpha -= 1/16.;
			if (Alpha <= 0.25)
				{
				Alpha = 0.25;
				visdir = 1;
				}
			}	
			
		if(master && master.CountInv("champion_AggroPower"))
			{
			SetOrigin((master.pos.x,master.pos.y,master.floorz+1),true);
			return;
			}
		destroy();
		}
	states
		{
		Spawn:
			PENT A 1 Bright;
			loop;
		}
	}

class champion_MissileManipulator : Thinker
	{
	actor missile;
	actor missiletarget;
	
	override void PostBeginPlay()
		{
		if(missile)
			{
			missile.bSEEKERMISSILE = true;
			missiletarget = missile.target.target;
			}
		}
	
	override void Tick()
		{
		super.Tick();
		if(missile && missiletarget)
			{
			//double dang = clamp(missile.AngleTo(missiletarget),0,8);
			//double dpit = clamp(missile.
			missile.VelFromAngle(missile.speed, missile.AngleTo(missiletarget));
			}
		
		else
			self.Destroy();
		}
	}
