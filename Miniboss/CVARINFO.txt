server float  rsqmb_bf_speed      = 1.0;    // Meter Speed: x on everything that fills the meter
server bool   rsqmb_bf_callin     = true;   // Call It In: the watch check + Use calls a full meter's Bossfight in (off: it only comes by itself)
server int    rsqmb_bf_auto       = 60;     // seconds after the meter filled that it comes by itself (0: only when called)
server int    rsqmb_bf_first      = 60;     // never in a map's first seconds
server int    rsqmb_bf_cooldown   = 90;     // seconds after one began before the next may come
server int    rsqmb_bf_cap        = 2;      // most Bossfights a map
server int    rsqmb_bf_minhp      = 40;     // it waits while your health is under this
server int    rsqmb_bf_busy       = 3;      // it waits while more awake monsters than this are after you within 1536
server float  rsqmb_bf_near       = 384;    // the arrival spot: at least this far from you (map units; 32 = 1 m)
server float  rsqmb_bf_far        = 1024;   // and at most this far
server float  rsqmb_bf_exit       = 768;    // it waits within this of an exit line
server float  rsqmb_bf_slow       = 3.0;    // seconds of RSQ SlowMo for the landing (0: none)
server int    rsqmb_bf_bits       = 3;      // x more RSQ VisualRewards boss bits for a Bossfight's miniboss
server float  rsqmb_bf_style      = 0.25;   // meter points a RSQ Score style point
server float  rsqmb_bf_xp         = 0.2;    // meter points a Gun Bonsai XP point
server int    rsqmb_bf_tier_score = 4000;   // this map's RSQ Score points a tier
user   bool   rsqmb_bf_hud        = true;   // the meter on the HUD
user   float  rsqmb_bf_hud_size   = 1.5;    // its size
server bool   rsqmb_classic       = false;  // the old pick at level load (rsqmb_per_map, rsqmb_map_chance), off: Bossfights are called in
server int    rsqmb_per_map       = 1;      // classic pick: minibosses chosen per level (0-4)
server int    rsqmb_map_chance    = 100;    // classic pick: % chance a level gets any
server int    rsqmb_thralls       = 3;      // thralls per miniboss (1-8)
server int    rsqmb_radius        = 1024;   // how far from the miniboss a thrall may stand (map units)
server bool   rsqmb_legend_thralls = true;  // with LegenDoom Lite: every thrall is made legendary
server bool   rsqmb_inherit       = true;   // the miniboss inherits its thralls' powers
server int    rsqmb_inherit_max   = 4;      // most distinct powers inherited (legendary not counted)
server int    rsqmb_lives         = 3;      // lives in all (3 = it dies on the third)
server int    rsqmb_shield        = 80;     // % of damage the miniboss shrugs off with every thrall alive (linear per thrall)
server float  rsqmb_invuln        = 5.0;    // seconds invulnerable after a lost life
server float  rsqmb_invuln_bonus  = 1.0;    // + seconds per living thrall
server bool   rsqmb_revive        = true;   // dead thralls rise when the invulnerability ends
server float  rsqmb_slow          = 4.0;    // seconds of slow time a thrall's death buys
server float  rsqmb_slow_chain    = 3.0;    // + seconds for another thrall killed in the window
server float  rsqmb_slow_max      = 12.0;   // the window's longest
server int    rsqmb_slow_depth    = -1;     // world speed % in the window; -1 = RSQ_SlowMo's own (rsm_depth)
server bool   rsqmb_greed         = true;   // the greed rule
server float  rsqmb_rage_speed    = 2.0;    // enraged monsters' speed multiplier
server bool   rsqmb_score         = true;   // push awards to RSQ_Score
// COUNTS AS A BOSS (owner 2026-11-03): a miniboss never gets the engine's BOSS flag; it counts as a boss only here.
server bool   rsqmb_boss_rewards   = true;  // RSQ VisualRewards' Kill Rewards: the boss fountain (x rsqgb_kr_boss) and always drops
server bool   rsqmb_boss_ldhealth  = true;  // made legendary by Miniboss: the HIGHER of LegenDoom's boss and normal legendary health
server bool   rsqmb_boss_roar      = false; // its sight and death sounds at full volume across the map (FULLVOLSEE / FULLVOLDEATH)
server bool   rsqmb_boss_nomorph   = false; // can't be morphed (DONTMORPH)
server bool   rsqmb_boss_noyank    = true;  // RSQ Grapple's and RSQ Whip's yank / throw refuse it (the line pulls you to it, as a boss)
server bool   rsqmb_boss_noheadlock = true; // RSQ Melee's headlock refuses it
user   bool   rsqmb_fx            = true;   // links, lives, invulnerability flames (the original Minibosses! IWAD sprites), rage sparks (particles)
user   int    rsqmb_fx_density    = 0;      // Effects Density: 0 Low (Quest-safe), 1 Medium, 2 High (beads a link, flames, the cap on effects drawn)

server noarchive bool rsqmb_debug = false;  // console lines for every rule as it fires
