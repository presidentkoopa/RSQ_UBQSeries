// RSQ Laser Sight options. The series' menu standard (design/MENU_STANDARD.md, owner 11-24): the front page holds the
// choices a player makes, On / Off, short labels; everything else is on "Test and Debug". Reset All at the foot covers
// both (RSQLSResetAll reads the rows). Each row's "// tip:" says what it does. Rows behind a Gun Bonsai VR upgrade are
// GbOption rows: greyed while RSQ Gun Bonsai VR (with RSQ Tactical Rig) is loaded and the node is unlearned (owner 11-04).
// The page carries no GbLine ("... detected" text): the standard allows no such line (11-24).

// ---- Radiant Silvergun Options (tools/rsmenu) ----------------------------------------------------
// The series' one Options entry. Every series pk3 carries this same block with its own prefix: the
// parent is PROTECTED, so the first definition stays and later ones add nothing (QZD 17.3 merges a
// protected option menu instead of replacing it); the link row removes any earlier copy of itself,
// so Options shows one row however many series pk3s load. Each mod then adds its own row only.
OptionMenu "RSQ_RSOptions" protected
{
	Class "RSQLS_RSMenu"
	Title "RADIANT SILVERGUN OPTIONS"
}

AddOptionMenu "OptionsMenu"
{
	RSQLSRSLink "Radiant Silvergun Options", "RSQ_RSOptions"
}

AddOptionMenu "RSQ_RSOptions"
{
	Submenu "RSQ Laser Sight", "RSQLS_Options"
}
// ----------------------------------------------------------------------------------------------------

	2, "At Dot"

OptionValue "RSQLS_Units"
{
	0, "Metres"
	1, "Feet"
	2, "Map Units"
}

OptionMenu "RSQLS_Options"
{
	Title "RSQ LASER SIGHT"
	// tip: Turns the whole laser sight on or off.
	Option "Laser Sight", "rsqls_on", "OnOff"
	// tip: The laser on the gun in your main hand.
	Option "Main Hand", "rsqls_main", "OnOff", "rsqls_on"
	// tip: The laser on the gun in your off hand.
	Option "Off Hand", "rsqls_off", "OnOff", "rsqls_on"
	StaticText " "
	StaticText "Look", 1
	// tip: Draws the beam from the muzzle.
	Option "Show Beam", "rsqls_beam", "OnOff"
	// tip: Draws the dot where the laser lands.
	Option "Show Dot", "rsqls_pointer", "OnOff"
	// tip: How big the dot is.
	Slider "Dot Size", "rsqls_spot_size", 0.25, 4.0, 0.05, 2, "rsqls_pointer"
	// tip: The laser's colour (the main hand's, or the only one when Colour Mode is Single).
	ColorPicker "Laser Colour", "rsqls_col_main"
	StaticText " "
	StaticText "Targets", 1
	// tip: The dot reacts when it is on a live hostile monster.
	RSQLSGbOption "Target Lock", "rsqls_lock", "OnOff", "RSQLS_SightService", "TargetLock"
	// tip: The dot changes when it is on a monster's head.
	RSQLSGbOption "Headshot Reaction", "rsqls_head", "OnOff", "RSQLS_SightService", "Headshot"
	StaticText " "
	StaticText "Extras", 1
	// tip: A thrown weapon such as a grenade shows its arc once it has been thrown once.
	Option "Grenade Arc", "rsqls_arc", "OnOff"
	// tip: Shows the range to what the laser hits, by the gun or at the dot.
	Option "Show Range", "rsqls_readout", "RSQLS_Readout"
	StaticText " "
	// tip: Every other setting: colours, when it shows, tuning numbers and the debug log.
	Submenu "Test and Debug", "RSQLS_Test"
	// tip: Puts every RSQ Laser Sight setting back to its default.
	RSQLSResetAll "Reset All", "Reset all RSQ Laser Sight settings?"
}
