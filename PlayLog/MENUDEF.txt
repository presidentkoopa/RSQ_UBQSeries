// ---- How To Play (tools/howtoplay, design/HOWTOPLAY_BUILD.md) --------------------------------------
// Every series pk3 carries this block with its own prefix. The help screen (F1 / Read This!) is the LAST loaded pk3's:
// a later ListMenu of one name replaces an earlier one, and every copy of the page is the same code, showing a tab for
// every loaded series mod. RSQ PlayLog has no row in Radiant Silvergun Options, so its How To Play row is in Options;
// it takes itself away whenever Radiant Silvergun Options exists (another series mod is loaded and has the row there).
ListMenu "ReadThisMenu"
{
	Class "RSQPL_HowToMenu"
}

ListMenu "RSQPL_HowTo"
{
	Class "RSQPL_HowToMenu"
}

AddOptionMenu "OptionsMenu"
{
	RSQPLHowToLink "How To Play", "RSQPL_HowTo"
}
// ----------------------------------------------------------------------------------------------------

OptionMenu "RSQPL_Menu"
{
	Title "RSQ PLAY LOG"
	// tip: Writes how you really play (swings, kills, hurts, deaths, maps) to the console for tuning; it changes nothing in the game.
	Option "Record Play Log", "rsqpl_on", "OnOff"
	// tip: Starts writing the console to rsqplay.txt in the game's own folder; starting it again starts the file over.
	SafeCommand "Start Log File", "logfile rsqplay.txt"
	// tip: Stops writing the log file.
	SafeCommand "Stop Log File", "logfile"
	StaticText " "
	// tip: Starting and stopping the log file rsqplay.txt.
	Submenu "Test and Debug", "RSQPL_Test"
	// tip: Puts every RSQ Play Log setting back to its default.
	SafeCommand "Reset All", "resetcvar rsqpl_on", "Reset all RSQ Play Log settings?"
}

OptionMenu "RSQPL_Test"
{
	Title "RSQ PLAY LOG: TEST AND DEBUG"
	StaticText " "
}
